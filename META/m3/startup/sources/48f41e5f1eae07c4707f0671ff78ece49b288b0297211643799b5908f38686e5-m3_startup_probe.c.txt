static int create_reference(char *exe,char *command,char *cwd,STARTUPINFOA *startup,PROCESS_INFORMATION *pi,int bottom_up_off) {

    UpdateAttributesProc update=(UpdateAttributesProc)GetProcAddress(kernel,"UpdateProcThreadAttribute");
    DeleteAttributesProc destroy=(DeleteAttributesProc)GetProcAddress(kernel,"DeleteProcThreadAttributeList");
    if(!bottom_up_off)return CreateProcessA(exe,command,NULL,NULL,FALSE,DEBUG_ONLY_THIS_PROCESS,NULL,cwd,startup,pi);
    if(!init || !update || !destroy)return 0;
    memset(&ex,0,sizeof(ex));ex.startup=*startup;ex.startup.cb=sizeof(ex);
    init(NULL,1,0,&size);if(!size)return 0;

    if(!update(ex.attributes,0,0x20007,policy,sizeof(policy),NULL,NULL)){destroy(ex.attributes);free(ex.attributes);return 0;}
    result=CreateProcessA(exe,command,NULL,NULL,FALSE,DEBUG_ONLY_THIS_PROCESS|0x80000,NULL,cwd,&ex.startup,pi);

static int code_byte(DWORD at,unsigned char byte) {
    DWORD old,unused;int ok;
    if(!VirtualProtectEx(child,(LPVOID)at,1,PAGE_EXECUTE_READWRITE,&old))return 0;
    ok=write_bytes(at,&byte,1) && FlushInstructionCache(child,(LPCVOID)at,1);
    if(!VirtualProtectEx(child,(LPVOID)at,1,old,&unused))ok=0;return ok;
}
static int add_point(DWORD at,int kind) {
    Breakpoint *p;if(point_count==MAX_BP)return 0;p=points+point_count;
    if(!read_bytes(at,&p->original,1) || p->original==0xcc || !code_byte(at,0xcc))return 0;
    p->address=at;p->kind=kind;p->active=1;++point_count;return 1;
}
static int restore(Breakpoint *p) {if(!code_byte(p->address,p->original))return 0;p->active=0;return 1;}
static int restore_all(void) {

        if(!read_bytes(points[i].address,&byte,1) || byte!=points[i].original)ok=0;
    }
    return ok;
}
