#include <math.h>
#include <float.h>
static DWORD flight_owner;static int flight_stage;
static const DWORD flight_ordinals[]={4939,5055,5064},flight_events[]={2010,1100,2011};
static float flight_bounds[6],flight_reference[3],flight_axis[3];
static UINT flight_keys[6];static int flight_held[6],flight_yaw_sign,flight_pitch_sign;
static int flight_finite(float x) {return x==x && x<FLT_MAX && x>-FLT_MAX;}
/* Target evidence is read-only. All offsets belong to the reviewed virtual-car
   family, not the player layout. Native code computes sampled positions. */
static DWORD training_target,training_owner,training_render_count,training_render_callback;
static int training_finished,training_launched,training_handoff,training_damage_count,training_kill,training_released;
static DWORD training_damage_packet[3];
static DWORD training_launch_owner(void) {
    DWORD head=word(0x4df254),node=head,owner;int i;
    if(!head)return 0;
    for(i=0;i<10000;i++) {
        if(word(node+0x118)==4827){owner=word(node+0x1c8);return owner?owner:node;}
        if(!read_bytes(node+0x124,&node,4) || !node || node==head)return 0;
    }return 0;
}
static int training_snapshot(const char *phase,DWORD count) {
    static const DWORD offsets[]={0xfc,0x100,0x11c,0x188,0xcc,0xd0,0x1a4,0x1a8,0x1e8,0x180,0x184,0x194,0x198,0x19c,0x1ac};
    DWORD fields[15],curve[15],displacement[3];int i;
    if(!training_target)return 0;
    for(i=0;i<15;i++)if(!read_bytes(training_target+offsets[i],fields+i,4))return 0;
    if(!read_bytes(training_target+0x120,curve,sizeof(curve)) || !read_bytes(training_target+0x15c,displacement,sizeof(displacement)))return 0;
    printf("{\"event\":\"training-target-state\",\"phase\":\"%s\",\"completed_simulation_callbacks\":%lu,\"entity\":%lu,\"raw_fields\":[",phase,count,training_target);
    for(i=0;i<15;i++){if(i)putchar(',');printf("%lu",fields[i]);}printf("],\"raw_curve_words\":[");
    for(i=0;i<15;i++){if(i)putchar(',');printf("%lu",curve[i]);}printf("],\"raw_displacement_words\":[%lu,%lu,%lu]}\n",displacement[0],displacement[1],displacement[2]);
    return fields[0]==6 && fields[1]==5000 && fields[2]==2013 && fields[3]==0x41200000 && fields[6]==training_owner && fields[7]==training_owner;
}
static int flight_target(void) {
    DWORD head=word(0x4df254),node=head,owner;int i;unsigned char keys[6];
    if(!head)return 0;
    for(i=0;i<10000;i++) {
        if(word(node+0x118)==flight_ordinals[flight_stage])break;
        if(!read_bytes(node+0x124,&node,4) || !node || node==head)return 0;
    }
    if(i==10000)return 0;flight_owner=node;owner=word(node+0x1c8);if(!owner)owner=node;
    if(!read_bytes(owner+0x1b0,flight_bounds,sizeof(flight_bounds)) || !read_bytes(word(0x4df240)+0x1c,keys,6))return 0;
    for(i=0;i<6;i++){if(!flight_finite(flight_bounds[i]) || !keys[i] || keys[i]==255)return 0;flight_keys[i]=keys[i];}
    for(i=0;i<3;i++)if(flight_bounds[i]>=flight_bounds[i+3])return 0;
    printf("{\"event\":\"intro-flight-target\",\"placement\":%lu,\"owner\":%lu,\"bounds_owner\":%lu,\"bounds\":[%g,%g,%g,%g,%g,%g]}\n",flight_ordinals[flight_stage],node,owner,flight_bounds[0],flight_bounds[1],flight_bounds[2],flight_bounds[3],flight_bounds[4],flight_bounds[5]);return 1;
}
static int flight_set_keys(DWORD pid,DWORD count,const int *wanted) {
    int i;for(i=0;i<6;i++)if(wanted[i]!=flight_held[i]) {
        if(!post_child_key(pid,count,flight_keys[i],wanted[i]))return 0;flight_held[i]=wanted[i];
    }return 1;
}
static int flight_tick(DWORD pid,DWORD count) {
    DWORD player=word(0x4dfab8);float p[3],b[9],v[3],d[3],aim[3],distance=0,speed=0,right=0,up=0,forward=0,slope=0,desired;int i,wanted[6]={0,0,0,0,0,0};
    if(!player || word(0x48f958)!=0x416120 || !read_bytes(player+0xc,p,12) || !read_bytes(player+0x18,b,36) || !read_bytes(player+0x48,v,12))return 0;
    for(i=0;i<3;i++)if(!flight_finite(p[i]) || !flight_finite(v[i]))return 0;
    for(i=0;i<9;i++)if(!flight_finite(b[i]))return 0;
    if(count==410) {
        if(!flight_target())return 0;
        for(i=0;i<3;i++){flight_reference[i]=-b[6+i];flight_axis[i]=b[i];}wanted[3]=1;
    } else if(count==416) {
        for(i=0;i<3;i++)slope+=(-b[6+i]-flight_reference[i])*flight_axis[i];
        if(fabs(slope)<0.00001)return 0;flight_yaw_sign=slope>0?1:-1;
        printf("{\"event\":\"flight-calibration\",\"axis\":\"yaw\",\"response\":%g,\"sign\":%d}\n",slope,flight_yaw_sign);
    } else if(count==422) {
        for(i=0;i<3;i++){flight_reference[i]=-b[6+i];flight_axis[i]=b[3+i];}wanted[1]=1;
    } else if(count==428) {
        for(i=0;i<3;i++)slope+=(-b[6+i]-flight_reference[i])*flight_axis[i];
        if(fabs(slope)<0.00001)return 0;flight_pitch_sign=slope>0?1:-1;
        printf("{\"event\":\"flight-calibration\",\"axis\":\"pitch\",\"response\":%g,\"sign\":%d}\n",slope,flight_pitch_sign);
    }
    for(i=0;i<3;i++){d[i]=(flight_bounds[i]+flight_bounds[i+3])*0.5f-p[i];distance+=d[i]*d[i];speed+=v[i]*v[i];}
    distance=(float)sqrt(distance);speed=(float)sqrt(speed);
    if(count>=434) {
        float norm=0;
        for(i=0;i<3;i++){aim[i]=d[i]-v[i]*0.7f;norm+=aim[i]*aim[i];}norm=(float)sqrt(norm);
        if(norm<0.001f)norm=1;
        for(i=0;i<3;i++){right+=aim[i]*b[i]/norm;up+=aim[i]*b[3+i]/norm;forward-=aim[i]*b[6+i]/norm;}
        if(fabs(right)>0.04)wanted[right*flight_yaw_sign>0?3:2]=1;
        if(fabs(up)>0.04)wanted[up*flight_pitch_sign>0?1:0]=1;
        desired=distance*0.35f;if(desired>16)desired=16;if(desired<2)desired=2;
        if((speed>desired+2 || (forward<0.8f && speed>4)))wanted[5]=1;
        else if(forward>0.8f && speed<desired && distance>1)wanted[4]=1;
    }
    printf("{\"event\":\"flight-observation\",\"completed_simulation_callbacks\":%lu,\"position\":[%g,%g,%g],\"velocity\":[%g,%g,%g],\"distance\":%g,\"aim_local\":[%g,%g,%g],\"keys\":[%d,%d,%d,%d,%d,%d]}\n",count,p[0],p[1],p[2],v[0],v[1],v[2],distance,right,up,forward,wanted[0],wanted[1],wanted[2],wanted[3],wanted[4],wanted[5]);
    return flight_set_keys(pid,count,wanted);
}
