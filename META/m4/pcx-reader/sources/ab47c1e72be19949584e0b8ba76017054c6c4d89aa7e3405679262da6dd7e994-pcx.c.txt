static void real_assets(const char *archive) {
    BHPcx image;BHPcxPalette palette;BHVFile *file;FILE *raw;unsigned char *bytes,*mapped;unsigned long backend,i,j,k,decoded=0,total=0;
    raw=fopen(archive,"rb");CHECK(raw!=NULL,"independent raw archive for PCX");if(!raw)return;bh_pcx_init(&image,NULL);
    for(backend=0;backend<3;backend++) {
        CHECK(bh_vfile_mount(archive),"mount PCX archive");
        for(i=0;i<sizeof(manifest)/sizeof(manifest[0]);i++) {
            const Manifest *m=manifest+i;
            bytes=(unsigned char*)malloc(m->size);CHECK(bytes!=NULL,"allocate independent raw PCX");if(!bytes)break;
            CHECK(!fseek(raw,m->offset,SEEK_SET) && fread(bytes,1,m->size,raw)==m->size,"raw PCX span");
            if(backend==1)CHECK(bh_vfile_preload(m->name,"rb"),"preload real PCX");
            if(backend==2)CHECK(write_file("bhpcx_real.tmp",bytes,m->size),"write private loose PCX");
            file=bh_vfile_open(backend==2?"bhpcx_real.tmp":m->name,"rb");CHECK(file!=NULL,"open real image");
            if(file) {
                CHECK(bh_pcx_read(&image,file),"source image decode");
                if(image.error==BH_PCX_OK) {
                    CHECK(image.width==m->width && image.height==m->height && image.pixel_bytes==m->pixels && crc32(image.pixels,image.pixel_bytes)==m->pixels_crc,"every decoded pixel matches independent RLE control");
                    CHECK(crc32(image.palette,768)==m->palette_crc && !memcmp(image.palette,bytes+m->size-768,768),"entire raw palette unchanged");
                    mapped=(unsigned char*)malloc(image.pixel_bytes);CHECK(mapped!=NULL,"allocate mapped-index comparison");
                    if(mapped) {
                        for(j=0;j<4;j++) {
                            const Transform *t=m->transforms+j;
                            CHECK(bh_pcx_palette(&palette,image.palette,(int)t->mode,t->color) && crc32(palette.rgb,768)==t->palette_crc && crc32(palette.indices,256)==t->indices_crc,"real palette and index map match independent instruction model");
                            for(k=0;k<image.pixel_bytes;k++)mapped[k]=palette.indices[image.pixels[k]];
                            CHECK(crc32(mapped,image.pixel_bytes)==t->mapped_crc,"entire mapped image in all four palette/color settings");
                        }
                        free(mapped);
                    }
                    ++decoded;total+=image.pixel_bytes;
                }
                CHECK(!bh_vfile_close(file),"caller closes image handle");
            }
            free(bytes);
        }
        bh_vfile_shutdown();
    }
    CHECK(decoded==1377 && total==24442368UL,"all 459 images through three backends");
    printf("{\"scenario\":\"pcx-real-assets\",\"files\":459,\"backend_decodes\":%lu,\"pixel_bytes_compared\":%lu,\"palette_settings\":4,\"surface_objects_created\":false}\n",decoded,total);
    bh_pcx_dispose(&image);fclose(raw);remove("bhpcx_real.tmp");
}
