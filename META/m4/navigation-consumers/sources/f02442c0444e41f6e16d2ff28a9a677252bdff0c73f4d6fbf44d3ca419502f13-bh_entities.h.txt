#ifndef BH_ENTITIES_H
#define BH_ENTITIES_H
#include "bh_math.h"
#include "bh_scene.h"

typedef struct BHEntity BHEntity;
typedef void (__cdecl *BHEntityStep)(BHEntity *entity);
typedef void (__cdecl *BHEntityCallback)(BHEntity *entity,void *packet);
struct BHEntity {
    unsigned long flags,generation,kind;
    BHVec3 position;
    unsigned char opaque0[48];
    BHVec3 origin;
    unsigned char opaque1[96];
    float lifetime,scale;
    unsigned char opaque2[12];
    BHEntityCallback refresh;
    BHEntityStep step;
    BHEntityCallback query;
    BHEntity *query_next,*query_previous,*previous,*next;
    unsigned char payload[636];
};
typedef struct BHEntityPositionPacket {long event;BHVec3 position;} BHEntityPositionPacket;
typedef enum BHEntityError {
    BH_ENTITY_OK,BH_ENTITY_ARGUMENT,BH_ENTITY_MEMORY,BH_ENTITY_LIMIT,
    BH_ENTITY_BUSY,BH_ENTITY_CONTAINER,BH_ENTITY_MUTATION,BH_ENTITY_NUMERICAL
} BHEntityError;
typedef struct BHEntityPool {
    BHEntity *active,*free_slots,*query_head,*focus;
    unsigned long capacity,used;
    BHVec3 origin;
    BHEntityError error;
    BHSceneAllocator allocator;
    void *chunks; /* source-private ownership, not original game globals */
    unsigned long dispatch_depth;
    int focused;
} BHEntityPool;

/* Complete original pool/dispatch behaviors with explicit context and NULL
   representation of verified 0047d680 RET callback. No game callbacks supplied
   by this module. Initial slots zeroed, recycled/growth payload not cleared.
   Initialize a fresh context; dispose before reinitialization. Max 65535 slots.
   Origin is an explicit caller-supplied equivalent of the shared origin vector.
   Generation, links, callbacks and private ownership fields are engine-owned. */
int __cdecl bh_entities_init(BHEntityPool *pool,unsigned long count,const BHSceneAllocator *allocator); /* 00476da0 */
BHEntity *__cdecl bh_entity_acquire(BHEntityPool *pool); /* 00476f20 */
int __cdecl bh_entity_release(BHEntityPool *pool,BHEntity *entity); /* 00477070 */
int __cdecl bh_entity_set_query(BHEntityPool *pool,BHEntity *entity,BHEntityCallback callback); /* 00476e90 */
int __cdecl bh_entities_query(BHEntityPool *pool,unsigned long mask,unsigned long value,
                            const BHVec3 *center,float radius,void *packet); /* 004770f0 */
int __cdecl bh_entities_step(BHEntityPool *pool); /* 004772f0 */
int __cdecl bh_entities_refresh(BHEntityPool *pool); /* 00477330 */
/* Source ownership/focus/validation adapters. Dispose rejects active dispatch.
   Callbacks may release the current slot or add new head slots. Changing a
   cached successor's lifetime stops dispatch explicitly with MUTATION. Nested
   queries are allowed to depth 16; nested step/refresh are rejected. */
int __cdecl bh_entities_dispose(BHEntityPool *pool);
int __cdecl bh_entities_focus(BHEntityPool *pool,BHEntity *entity);
int __cdecl bh_entities_valid(const BHEntityPool *pool);
#endif
