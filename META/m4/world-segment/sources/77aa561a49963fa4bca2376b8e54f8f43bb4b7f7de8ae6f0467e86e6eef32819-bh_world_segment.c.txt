typedef char actor_basis_offset[offsetof(BHEntity,opaque0)==0x18 ? 1:-1];

static BHWorldSegmentRoom *room(BHWorldSegmentContext *q,const BHWorldGeometry *w) {
    unsigned long i;for(i=0;i<q->room_count;++i)if(q->rooms[i].world==w)return q->rooms+i;
    return NULL;
}
static int fail(BHWorldSegmentContext *q,BHWorldSegmentError e) {q->error=e;return -1;}
static int finite_vec(const BHVec3 *v) {return v && _finite(v->x) && _finite(v->y) && _finite(v->z);}
static int contains(const BHWorldGeometry *w,const BHVec3 *p) {
    return w->minimum.x<=p->x && p->x<=w->maximum.x && w->minimum.y<=p->y &&
        p->y<=w->maximum.y && w->minimum.z<=p->z && p->z<=w->maximum.z;
}
static int excluded(const BHWorldHit *self,BHWorldGeometry *w,BHEntity *e) {
    return self && ((w && self->kind==BH_WORLD_HIT_GEOMETRY && self->object.geometry==w) ||
                   (e && self->kind==BH_WORLD_HIT_ACTOR && self->object.actor==e));
}
int __cdecl bh_world_segment_init(BHWorldSegmentContext *q,BHWorldGeometry *first,
    unsigned long count,BHWorldSegmentRoom *rooms,unsigned long room_count,
    const BHWorldSegmentActor *actors,unsigned long actor_count,unsigned long max_room_actors) {
    BHWorldSegmentContext next;BHWorldGeometry *p;unsigned long i,j;
    if(!q || !first || !count || count>65535 || !rooms || room_count!=count ||
       actor_count>65535 || (actor_count && !actors) || !max_room_actors || max_room_actors>65535)return 0;
    memset(&next,0,sizeof(next));next.first=first;next.world_count=count;
    next.rooms=rooms;next.room_count=room_count;next.actors=actors;
    next.actor_count=actor_count;next.max_room_actors=max_room_actors;
    p=first;
    for(i=0;i<count;++i) {
        if(!p || (i && p==first) || !room(&next,p) || p->navigation.count>6)return 0;
        for(j=0;j<i;++j)if(rooms[j].world==rooms[i].world)return 0;
        p=(BHWorldGeometry*)p->navigation.next;
    }
    if(p!=first)return 0;
    for(i=0;i<actor_count;++i)if(!actors[i].entity || actors[i].entity->kind)return 0;
    for(i=0;i<count;++i) {
        p=rooms[i].world;
        if(p->owner && !room(&next,p->owner))return 0;
        for(j=0;j<p->navigation.count;++j)if(p->navigation.portals[j].neighbor &&
            !room(&next,(BHWorldGeometry*)p->navigation.portals[j].neighbor))return 0;
    }
    *q=next;return 1;
}
static double projection(const BHVec3 *point,const BHVec3 *start,const BHVec3 *direction) {
    return (((double)point->x-start->x)*direction->x+
            ((double)point->z-start->z)*direction->z)+((double)point->y-start->y)*direction->y;
}
static double distance_squared(double x,double y,double z) {
    float sx,sy;bh_round_float(&sx,x);bh_round_float(&sy,y);
    return (x*sx+y*sy)+z*z;
}
static long material(unsigned long bits) {
    long signed_bits;float stored;memcpy(&signed_bits,&bits,4);
    bh_round_float(&stored,(double)signed_bits);
    /* Original CRT __ftol produces64 bits; only EAX is stored. The sole
       positive overflow of this signed32 input roundtrip is exactly2^31. */
    if(stored>=2147483648.0)return (long)0x80000000u;
    return (long)stored;
}
/* First strict crossing in slot order; aperture (+8), not matched overlap. */
static BHWorldGeometry *crossing(BHWorldGeometry *w,const BHVec3 *a,const BHVec3 *b,BHVec3 *point) {
    unsigned long i,axis,u,v;const float *start=(const float*)a,*end=(const float*)b;
    BHPortal *p;double plane,t,s,du,dv,upper_u;float tt,ss,plane_float,stored_u,stored_v;BHVec3 at;
    for(i=0;i<w->navigation.count;++i) {
        p=w->navigation.portals+i;if(!p->neighbor || p->direction>6 || p->direction==3)continue;
        axis=p->direction&3;bh_round_float(&plane_float,(double)p->plane);plane=plane_float;
        if(p->direction<4) {if(!(end[axis]<plane && plane<start[axis]))continue;}
        else if(!(start[axis]<plane && plane<end[axis]))continue;
        t=(plane-start[axis])/((double)end[axis]-start[axis]);
        s=((double)end[axis]-plane)/((double)end[axis]-start[axis]);
        bh_round_float(&tt,t);bh_round_float(&ss,s);
        u=axis==0?1:0;v=axis==2?1:2;
        /* X crossings reload both binary32 ratios. Y/Z retain both ratios
           for X, then reload them for their other aperture coordinate. */
        du=axis==0 ? (double)end[u]*tt+(double)start[u]*ss : (double)end[u]*t+(double)start[u]*s;
        dv=(double)end[v]*tt+(double)start[v]*ss;
        bh_round_float(&stored_u,du);bh_round_float(&stored_v,dv);
        upper_u=axis==0?stored_u:du;
        if(axis==2)dv=stored_v; /* Z crossing stores Y before both tests. */
        if(p->aperture[0]<du && upper_u<p->aperture[1] && p->aperture[2]<dv && dv<p->aperture[3]) {
            ((float*)&at)[axis]=plane_float;((float*)&at)[u]=stored_u;((float*)&at)[v]=stored_v;
            *point=at;return (BHWorldGeometry*)p->neighbor;
        }
    }
    return NULL;
}
static int geometry(BHWorldSegmentContext *q,BHWorldGeometry *w,const BHVec3 *a,
    BHVec3 *end,BHWorldSegmentHit *out) {
    BHWorldSegmentRoom *binding=room(q,w);BHSegmentHit hit;int result;
    if(!binding)return fail(q,BH_WORLD_SEGMENT_TOPOLOGY);
    if(!binding->query)return 0;
    result=bh_segment_query_run(binding->query,a,end,&w->transform,&w->inverse,&hit);
    if(result<0)return fail(q,BH_WORLD_SEGMENT_DEPENDENCY);
    if(result) {
        *end=hit.point;out->point=hit.point;out->normal=hit.normal;
        out->object.kind=BH_WORLD_HIT_GEOMETRY;out->object.object.geometry=w;
        out->material_flags=material(hit.flags);
    }
    return result;
}
static BHEntity *actor_next(BHEntity *e) {BHEntity *next;memcpy(&next,e->payload+16,sizeof(next));return next;}
static int bh_world_segment_room_trace(BHWorldSegmentContext *q,const BHWorldHit *self,BHWorldGeometry *w,
    const BHVec3 *start,const BHVec3 *end,float radius,const BHVec3 *direction,BHWorldSegmentHit *out) {
    BHVec3 a=*start,limit,hit_point,normal;BHWorldGeometry *neighbor,*child;
    BHWorldSegmentRoom *binding;BHEntity *actor;BHEntityPositionPacket packet;
    float length,along,rr;double dot,dx,dy,dz,ds,hit_length,px,py,pz;unsigned long steps,rooms=0;int result,found;
    while(w) {
        if(++rooms>q->world_count+1)return fail(q,BH_WORLD_SEGMENT_TOPOLOGY);
        binding=room(q,w);if(!binding)return fail(q,BH_WORLD_SEGMENT_TOPOLOGY);
        limit=*end;neighbor=crossing(w,&a,end,&limit);hit_point=limit;found=0;
        bh_round_float(&length,sqrt(distance_squared((double)limit.x-a.x,(double)limit.y-a.y,(double)limit.z-a.z)));
        if(radius>0) {
            bh_round_float(&rr,(double)radius*radius);actor=binding->actors;steps=0;packet.event=3;
            while(actor) {
                if(++steps>q->max_room_actors)return fail(q,BH_WORLD_SEGMENT_TOPOLOGY);
                if(actor->kind)return fail(q,BH_WORLD_SEGMENT_ARGUMENT);
                if(!excluded(self,NULL,actor)) {
                    if(!actor->query)return fail(q,BH_WORLD_SEGMENT_DEPENDENCY);
                    actor->query(actor,&packet);
                    if(!finite_vec(&packet.position))return fail(q,BH_WORLD_SEGMENT_DEPENDENCY);
                    dot=projection(&packet.position,&a,direction);bh_round_float(&along,dot);
                    if(dot>=0 && along<=length) {
                        dx=((double)along*direction->x+a.x)-packet.position.x;
                        dy=((double)along*direction->y+a.y)-packet.position.y;
                        dz=((double)along*direction->z+a.z)-packet.position.z;
                        ds=distance_squared(dx,dy,dz);
                        if(ds<rr) {
                            hit_length=(double)along-sqrt((double)rr-ds);bh_round_float(&length,hit_length);
                            px=hit_length*direction->x+a.x;py=hit_length*direction->y+a.y;pz=hit_length*direction->z+a.z;
                            bh_round_float(&hit_point.x,px);bh_round_float(&hit_point.y,py);bh_round_float(&hit_point.z,pz);
                            bh_round_float(&normal.x,px-packet.position.x);
                            bh_round_float(&normal.y,py-packet.position.y);
                            bh_round_float(&normal.z,pz-packet.position.z);
                            bh_vec3_normalize(&normal);out->point=hit_point;out->normal=normal;
                            out->object.kind=BH_WORLD_HIT_ACTOR;out->object.object.actor=actor;
                            out->material_flags=0;found=1;
                        }
                    }
                }
                actor=actor_next(actor);
            }
        }
        if((w->query_flags&2) && !excluded(self,w,NULL)) {
            result=geometry(q,w,&a,&hit_point,out);if(result<0)return -1;if(result)found=1;
        }
        child=(BHWorldGeometry*)w->navigation.next;steps=0;
        while(child && child->owner) {
            if(++steps>q->world_count || !room(q,child))return fail(q,BH_WORLD_SEGMENT_TOPOLOGY);
            if((child->query_flags&2) && !excluded(self,child,NULL) &&
               (child->minimum.x<=a.x || child->minimum.x<=hit_point.x) &&
               (a.x<=child->maximum.x || hit_point.x<=child->maximum.x) &&
               (child->minimum.y<=a.y || child->minimum.y<=hit_point.y) &&
               (a.y<=child->maximum.y || hit_point.y<=child->maximum.y) &&
               (child->minimum.z<=a.z || child->minimum.z<=hit_point.z) &&
               (a.z<=child->maximum.z || hit_point.z<=child->maximum.z)) {
                result=geometry(q,child,&a,&hit_point,out);if(result<0)return -1;if(result)found=1;
            }
            child=(BHWorldGeometry*)child->navigation.next;
        }
        if(!child)return fail(q,BH_WORLD_SEGMENT_TOPOLOGY);
        if(found)return 1;
        if(!neighbor)return 0;
        a=limit;w=neighbor;
    }
    return 0;
}
static void actor_pose(const BHEntity *e,BHMat34 *pose) {
    unsigned long i,j;float value;
    for(i=0;i<3;++i)for(j=0;j<3;++j) {
        memcpy(&value,e->opaque0+(i*3+j)*4,4);pose->m[j][i]=value;
    }
    pose->m[0][3]=e->position.x;pose->m[1][3]=e->position.y;pose->m[2][3]=e->position.z;
}
static int bh_world_segment_actor_trace(BHWorldSegmentContext *q,const BHWorldHit *self,const BHVec3 *a,
    const BHVec3 *end,const BHVec3 *direction,BHWorldSegmentHit *out) {
    unsigned long i;BHEntity *e;BHMat34 pose,inverse;BHVec3 limit=*end;
    BHSegmentHit hit;float length,along,stored_y,stored_z;double dot,dx,dy,dz;int found=0,result;
    bh_round_float(&length,projection(end,a,direction));
    for(i=0;i<q->actor_count;++i) {
        e=q->actors[i].entity;
        if(!e)return fail(q,BH_WORLD_SEGMENT_TOPOLOGY);
        if(e->kind)return fail(q,BH_WORLD_SEGMENT_ARGUMENT);
        if(excluded(self,NULL,e))continue;
        dot=projection(&e->position,a,direction);
        bh_round_float(&along,dot);
        if(!(-(double)e->collision_radius<dot && along<(double)length+e->collision_radius))continue;
        dx=((double)along*direction->x+a->x)-e->position.x;
        dy=((double)along*direction->y+a->y)-e->position.y;
        dz=((double)along*direction->z+a->z)-e->position.z;
        bh_round_float(&stored_y,dy);bh_round_float(&stored_z,dz);
        if(!((dx*dx+dy*stored_y)+dz*stored_z<(double)e->collision_radius*e->collision_radius))continue;
        if(!q->actors[i].query)continue;
        actor_pose(e,&pose);bh_mat34_inverse_rigid(&pose,&inverse);
        result=bh_segment_query_run(q->actors[i].query,a,&limit,&pose,&inverse,&hit);
        if(result<0)return fail(q,BH_WORLD_SEGMENT_DEPENDENCY);
        if(result) {
            dot=(((double)hit.point.y-a->y)*direction->y+((double)hit.point.x-a->x)*direction->x)+
                 ((double)hit.point.z-a->z)*direction->z;
            bh_round_float(&along,dot);
            if(dot<length) {
                length=along;limit=hit.point;out->point=hit.point;out->normal=hit.normal;
                out->object.kind=BH_WORLD_HIT_ACTOR;out->object.object.actor=e;found=1;
                /* Original global actor pass does not overwrite output+2c. */
            }
        }
    }
    return found;
}
int __cdecl bh_world_segment_run(BHWorldSegmentContext *q,const BHWorldHit *self,
    BHWorldGeometry **cached,const BHVec3 *start,const BHVec3 *end,float radius,BHWorldSegmentHit *out) {
    BHWorldGeometry *owner,*old;BHWorldSegmentHit hit,original;BHVec3 direction,limit;int a,b;
    double denominator,numerator;
    if(!q)return -1;
    if(!cached || !out)return fail(q,BH_WORLD_SEGMENT_ARGUMENT);
    if(q->busy)return fail(q,BH_WORLD_SEGMENT_BUSY);
    if(!finite_vec(start) || !finite_vec(end) || !_finite(radius) || !bh_math_environment_supported())return fail(q,BH_WORLD_SEGMENT_ARGUMENT);
    old=*cached;if(old && !room(q,old))return fail(q,BH_WORLD_SEGMENT_TOPOLOGY);
    original=*out;hit=original;hit.material_flags=0;owner=old;
    if(!owner || !contains(owner,start)) {
        owner=bh_world_locate(q->first,q->world_count,old,start,0);
        if(!owner)owner=old;
    }
    if(!owner) {out->material_flags=0;q->error=BH_WORLD_SEGMENT_OK;return 0;}
    bh_round_float(&direction.x,(double)end->x-start->x);
    bh_round_float(&direction.y,(double)end->y-start->y);
    bh_round_float(&direction.z,(double)end->z-start->z);bh_vec3_normalize(&direction);
    *cached=owner;out->material_flags=0;
    q->busy=1;q->error=BH_WORLD_SEGMENT_OK;
    a=bh_world_segment_room_trace(q,self,owner,start,end,radius,&direction,&hit);
    if(a<0) {q->busy=0;*cached=old;*out=original;return -1;}
    limit=a?hit.point:*end;b=bh_world_segment_actor_trace(q,self,start,&limit,&direction,&hit);
    q->busy=0;if(b<0) {*cached=old;*out=original;return -1;}
    if(a || b) {
        numerator=projection(&hit.point,start,&direction);
        denominator=(((double)end->z-start->z)*direction.z+((double)end->y-start->y)*direction.y)+
                     ((double)end->x-start->x)*direction.x;
        bh_round_float(&hit.fraction,numerator/denominator);*out=hit;
    } else out->material_flags=0;
    return a || b;
}
