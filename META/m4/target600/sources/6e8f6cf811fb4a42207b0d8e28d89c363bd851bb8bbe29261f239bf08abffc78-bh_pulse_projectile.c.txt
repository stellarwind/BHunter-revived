static const BHVec3 zero={0,0,0},acceleration={0,-.002f,0};
static void get(const BHEntity *e,unsigned long a,void *p,unsigned long n){memcpy(p,(const unsigned char*)e+a,n);}
static void put(BHEntity *e,unsigned long a,const void *p,unsigned long n){memcpy((unsigned char*)e+a,p,n);}
static float scalar(const BHEntity *e,unsigned long a){float f;get(e,a,&f,4);return f;}
static float stored(double x){float f;bh_round_float(&f,x);return f;}
static BHPulseProjectileContext *context(BHEntity *e){BHPulseProjectileContext *c;memcpy(&c,e->payload+600,4);return c;}
static int live(BHPulseProjectileContext *c,BHEntity *e,unsigned long gen){BHEntity *p;unsigned long i=0;for(p=c->effects->pool->active;p&&i++<c->effects->pool->capacity;p=p->next)if(p==e)return e->generation==gen;return 0;}
static void __cdecl draw(BHEntity *e,void *unused){(void)unused;bh_pulse_visual_draw(context(e)->draw,e);}
static void announce(BHPulseProjectileContext *c,BHEntity *e){unsigned long sync,i;unsigned char payload[2];BHWorldGeometry *world;BHPacketWriter writer;get(e,0x120,&sync,4);if(c->peers<=1||!sync)return;get(e,0x124,&world,4);if(!world)return;for(i=0;i<c->world_count;i++)if(c->world_indices[i].world==world)break;if(i==c->world_count){c->error=2;return;}
 if(!bh_packet_begin(&writer,c->packet,sizeof(c->packet),9,(unsigned short)sync)||!bh_packet_write_position(&writer,world,c->world_indices[i].index,&e->position)){c->error=2;return;}payload[0]=(unsigned char)(sync>>16);payload[1]=3;if(!bh_packet_append(&writer,payload,2)||!bh_packet_write_vector(&writer,&zero,1)||!bh_packet_finish(&writer,&c->packet_bytes)){c->error=2;return;}bh_network_send(c->transport,0,c->packet,c->packet_bytes,0);++c->packets;
}
void __cdecl bh_pulse_projectile_tick(BHEntity *e){BHPulseProjectileContext *c=context(e);BHHomingProjectileContext *b;float life,phase,spin,angle,drift;BHVec3 velocity,base,random_velocity,end,point;BHWorldGeometry *cached;BHEntity *ignore,*source,*contact;BHWorldHit self;BHWorldSegmentHit hit;BHImpactState *impact;unsigned long identity,sound,generation,i;double dot,power,offset,raw_life;int r,result;struct{long event;float amount;BHEntity *source;}damage;
 if(!c||!bh_math_environment_supported())return;b=c->effects;c->error=0;generation=e->generation;raw_life=(double)scalar(e,0xe8)-1.0/30;life=stored(raw_life);phase=stored((double)scalar(e,0xec)+scalar(e,0xf0));put(e,0xec,&phase,4);put(e,0xe8,&life,4);if(raw_life<3){drift=stored((double)scalar(e,0xf0)*.92);put(e,0xf0,&drift,4);}
 if(raw_life<.2&&raw_life>=.16666666666666669){announce(c,e);if(c->error)return;bh_script_debris_create(c->debris,30,&e->position,4,50,1);get(e,0x11c,&identity,4);*c->identity=identity;c->debris->origin_tag=identity;get(e,0x130,&source,4);bh_script_explosion_create(c->explosion,&e->position,1.5f,27,1,(unsigned long)source);*c->identity=c->debris->origin_tag;
  if(!live(c,e,generation)){c->error=3;return;}if(!identity){damage.event=0;damage.amount=1;damage.source=source;bh_entities_query(b->pool,0,0,&e->position,30,&damage);if(!live(c,e,generation)){c->error=3;return;}}++c->explosions;
 }
 for(i=0;i<3;i++){r=rand();((float*)&random_velocity)[i]=stored(((double)r-16384)*.000002);}
 if(life>3){r=rand();spin=stored(((double)r-16384)*(1.0/16384));r=rand();angle=stored((double)r*(1.0/512));bh_script_particle_create(b->particles,b->first_image,3,&e->position,&random_velocity,&acceleration,.2f,angle,5,spin,.9f,.95f,.5f,.15f);}
 if(life<=0){bh_entity_release(b->pool,e);return;}
 get(e,0xf4,&velocity,12);get(e,0x10c,&base,12);for(i=0;i<3;i++){((float*)&e->origin)[i]=stored((double)((float*)&base)[i]+((float*)&velocity)[i]);((float*)&velocity)[i]=stored((double)((float*)&velocity)[i]*.95);}put(e,0xf4,&velocity,12);
 power=pow((double)life,.25);offset=(1-power)*cos((double)life*3)*(-1.0/6);end.x=stored((double)e->position.x+(double)e->origin.x/30);end.y=stored((double)e->position.y+((double)e->origin.y/30-offset));end.z=stored((double)e->position.z+(double)e->origin.z/30);
 get(e,0x124,&cached,4);get(e,0x128,&ignore,4);self.kind=BH_WORLD_HIT_ACTOR;self.object.actor=ignore;result=bh_world_segment_run(b->segment,ignore?&self:NULL,&cached,&e->position,&end,-1,&hit);if(result<0){c->error=2;return;}put(e,0x124,&cached,4);get(e,0x118,&sound,4);
 if(!result)e->position=end;else if(hit.object.kind==BH_WORLD_HIT_GEOMETRY&&(hit.material_flags&4)){point.x=stored((double)hit.point.x+(double)hit.normal.x*.5);point.y=stored((double)hit.point.y+(double)hit.normal.y*.5);point.z=stored((double)hit.point.z+(double)hit.normal.z*.5);contact=bh_contact_impact_create(b->contacts,&point,&hit.normal,4);if(contact){impact=bh_contact_impact_state(b->contacts,contact);impact->duration=1;impact->size=10;bh_sound_request(b->sound,b->liquid_sound,stored(10.0/6),1,&point,&zero);}bh_sound_stop(b->sound,sound);bh_entity_release(b->pool,e);++c->liquid_hits;return;
 }else{get(e,0xf4,&velocity,12);dot=(((double)velocity.x*hit.normal.x+(double)velocity.y*hit.normal.y)+(double)velocity.z*hit.normal.z)*(-2);for(i=0;i<3;i++)((float*)&velocity)[i]=stored((double)((float*)&hit.normal)[i]*dot+((float*)&velocity)[i]);put(e,0xf4,&velocity,12);dot=(((double)base.x*hit.normal.x+(double)base.y*hit.normal.y)+(double)base.z*hit.normal.z)*(-2);for(i=0;i<3;i++)((float*)&base)[i]=stored((double)((float*)&hit.normal)[i]*dot+((float*)&base)[i]);ignore=NULL;put(e,0x128,&ignore,4);e->position=hit.point;for(i=0;i<3;i++){((float*)&velocity)[i]=stored((double)((float*)&velocity)[i]*.9);((float*)&base)[i]=stored((double)((float*)&base)[i]*.9);}put(e,0xf4,&velocity,12);put(e,0x10c,&base,12);bh_sound_request(b->sound,c->bounce_sound,1,1,&e->position,&zero);bh_contact_spark_create(b->contacts,&e->position,.75f,3);bh_contact_spark_create(b->contacts,&e->position,1,2);++c->bounces;}
 bh_sound_update(b->sound,sound,1,1,&e->position,&e->origin);
}
BHEntity *__cdecl bh_pulse_projectile_create(BHPulseProjectileContext *c,BHWorldGeometry *world,BHEntity *source,const BHVec3 *position,const BHVec3 *base,const BHVec3 *velocity){BHEntity *e;BHProjectileQueryState state;float f;unsigned long sync,i;BHHomingProjectileContext *b;
 if(!c||!position||!base||!velocity||!c->effects||!c->lifetime||!c->draw||!c->draw->queue||!c->identity||!c->sync_identity||!c->debris||!c->explosion||!c->transport||!bh_math_environment_supported())return NULL;b=c->effects;if(!b->pool||!b->sound||!b->segment||!b->particles||!b->contacts||c->lifetime->pool!=b->pool||c->lifetime->sound!=b->sound||c->debris->pool!=b->pool||c->explosion->pool!=b->pool||c->explosion->debris!=c->debris)return NULL;
 e=bh_entity_acquire(b->pool);if(!e)return NULL;bh_projectile_query_state(e,&state);state.countdown=4;state.identity=*c->identity;*c->identity=0;state.secondary=source;e->position=*position;f=stored((double)rand()*(1.0/5120));put(e,0xec,&f,4);f=1.0f/30;put(e,0xf0,&f,4);put(e,0xf4,velocity,12);put(e,0x10c,base,12);for(i=0;i<3;i++)((float*)&e->origin)[i]=stored((double)((const float*)velocity)[i]+((const float*)base)[i]);e->origin=*velocity;memcpy(e->payload+600,&c,4);e->step=bh_pulse_projectile_tick;e->refresh=draw;bh_projectile_lifetime_bind(c->lifetime,e,&state);state.sound=bh_sound_request(b->sound,c->loop_sound,1,1,position,&e->origin);put(e,0x118,&state.sound,4);put(e,0x124,&world,4);put(e,0x128,&source,4);sync=*c->sync_identity;*c->sync_identity=0;put(e,0x120,&sync,4);if(state.identity){f=stored((double)state.countdown-(-.3));put(e,0xe8,&f,4);}return e;
}
