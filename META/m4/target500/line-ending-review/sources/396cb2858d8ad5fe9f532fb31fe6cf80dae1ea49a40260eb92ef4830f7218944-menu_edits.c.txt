static unsigned long checks, failures;
#define CHECK(x, s)                                                            \
  do {                                                                         \
    ++checks;                                                                  \
    if (!(x)) {                                                                \
      ++failures;                                                              \
      if (failures < 20)                                                       \
        printf("FAIL menu-edits %d %s\n", __LINE__, s);                        \
    }                                                                          \
  } while (0)
typedef struct Fixture {
  BHMenuEdits context;
  BHConfig config;
  BHInputProfile profile;
  BHMenuSettings settings;
  BHMenuNetworkRecord network;
  BHMenuRows rows;
  BHMenuRow row[16], *slots[16];
  BHAtlasPage page[16];
  BHTextQueue queue;
  BHMenuSettingsMenu menu, mission_menu;
  BHMenuSettingsRow settings_row, *settings_slots[16];
  BHMenuEditBridge bridge;
  BHCdClose cd;
  BHCdNotify notify;
  BHCdAudio audio;
  BHNetworkSessions sessions;
  BHInputKeyCapture capture;
  BHPlayerInputOwner input_owner;
  BHPlayerInputList *input;
  BHInputSnapshot snapshot;
  BHPlayerInputState state;
  BHEntityPool pool;
  BHSoundQueue sound;
  unsigned char result[2];
  BHMenuMissionRow missions[9];
  long mode, mission, ship;
} Fixture;
static int setup(Fixture *f) {
  unsigned long i;
  memset(f, 0, sizeof(*f));
  if (!bh_config_init(&f->config, NULL) ||
      !bh_entities_init(&f->pool, 4, NULL) ||
      !bh_sound_queue_init(&f->sound, NULL, NULL, NULL) ||
      !bh_player_input_owner_init(&f->input_owner, NULL))
    return 0;
  f->input = bh_player_input_create(&f->input_owner, 0);
  if (!f->input)
    return 0;
  for (i = 0; i < 255; ++i)
    bh_player_input_bind_key(f->input, i, "", 0, 0, (unsigned short)i);
  bh_text_queue_init(&f->queue);
  for (i = 0; i < 16; ++i) {
    f->slots[i] = f->row + i;
    f->row[i].page = f->page + i;
    f->page[i].size = 128;
    f->settings_slots[i] = &f->settings_row;
  }
  f->rows.rows = f->slots;
  f->rows.count = f->rows.capacity = 16;
  f->rows.text = &f->queue;
  f->settings.input_profile = &f->profile;
  f->settings.selected_network = &f->network;
  f->menu.rows = f->settings_slots;
  f->menu.row_count = 16;
  f->menu.selection = 1;
  f->menu.user = &f->bridge;
  f->settings_row.refresh = bh_menu_edit_row_callback;
  f->bridge.context = &f->context;
  f->cd.last_track = -1;
  f->notify.cd = &f->cd;
  f->audio.notify = &f->notify;
  f->audio.wave_device = (HWAVEOUT)-1;
  f->capture.pool = &f->pool;
  f->capture.input = f->input;
  f->capture.snapshot = &f->snapshot;
  f->capture.state = &f->state;
  f->capture.settings = &f->settings;
  f->capture.menu = &f->menu;
  f->capture.sound = &f->sound;
  f->capture.result = f->result;
  f->context.config = &f->config;
  f->context.settings = &f->settings;
  f->context.rows = &f->rows;
  f->context.audio = &f->audio;
  f->context.sessions = &f->sessions;
  f->context.capture = &f->capture;
  f->context.menu = &f->menu;
  f->context.mission_menu = &f->mission_menu;
  f->context.mode = &f->mode;
  f->context.mission = &f->mission;
  f->context.ship = &f->ship;
  for (i = 0; i < 9; ++i)
    f->context.mission_rows[i] = f->missions + i;
  return 1;
}
static void finish(Fixture *f) {
  if (f->capture.entity)
    bh_input_key_capture_cancel(&f->capture);
  CHECK(bh_sound_queue_dispose(&f->sound) && bh_entities_dispose(&f->pool),
        "real pool/sound ownership disposed");
  bh_player_input_owner_dispose(&f->input_owner);
  bh_config_dispose(&f->config);
}
static void reset(Fixture *f) {
  bh_text_queue_init(&f->queue);
  f->context.error = 0;
}
static void body(Fixture *f, long row, const char *text) {
  unsigned long n = f->queue.write & 255;
  if (!text) {
    CHECK(n == 1 && f->queue.commands[0].kind == BH_TEXT_COMMAND_CLEAR,
          "NULL original serial label gives clear-only row");
    return;
  }
  CHECK(n >= 2 && f->queue.commands[n - 2].kind == BH_TEXT_COMMAND_CLEAR &&
            f->queue.commands[n - 1].kind == BH_TEXT_COMMAND_TEXT &&
            f->queue.commands[n - 1].page == f->page + row &&
            !strcmp((char *)f->queue.commands[n - 1].text, text),
        "whole refresh reaches actual row clear/text queue with exact body and "
        "selected page");
}
static void labels(void) {
  Fixture f;
  long value;
  unsigned long i;
  CHECK(setup(&f), "real row, input, config and media fixture");
  value = 0;
  memcpy(f.config.settings + 4, &value, 4);
  bh_menu_edit_video(&f.context, 0);
  body(&f, 0, "Disable");
  reset(&f);
  value = 1;
  memcpy(f.config.settings + 4, &value, 4);
  bh_menu_edit_video(&f.context, 0);
  body(&f, 0, "Allow");
  reset(&f);
  memcpy(f.config.settings + 12, "1234567890123456789012345", 26);
  bh_menu_edit_video(&f.context, 1);
  body(&f, 1, "12345678901234567890");
  reset(&f);
  bh_menu_edit_video(&f.context, 2);
  body(&f, 2, "320 x 240");
  reset(&f);
  value = 1;
  memcpy(f.config.settings + 8, &value, 4);
  bh_menu_edit_video(&f.context, 2);
  body(&f, 2, "640 x 480");
  reset(&f);
  value = 7;
  memcpy(f.config.settings + 8, &value, 4);
  bh_menu_edit_video(&f.context, 2);
  CHECK(!f.queue.write, "unknown video enum preserves row");
  reset(&f);
  bh_menu_edit_audio(&f.context, 1);
  CHECK(f.queue.write == 4 && !f.audio.wave_reads,
        "wave missing-endpoint getter executes without native reads");
  body(&f, 1, "100% Volume");
  CHECK(!strcmp((char *)f.queue.commands[1].text, "100% Volume"),
        "audio refreshes paired first row before selected second row");
  reset(&f);
  bh_menu_edit_audio(&f.context, 3);
  CHECK(f.queue.write == 4 && !f.audio.auxiliary_reads,
        "CD missing-endpoint getter executes without native reads");
  body(&f, 3, "100% Volume");
  reset(&f);
  bh_menu_edit_audio(&f.context, 2);
  CHECK(!f.queue.write, "unselected audio row ignored");
  strcpy((char *)&f.profile + 12, "CurrentProfile");
  reset(&f);
  bh_menu_edit_profile(&f.context, 0);
  body(&f, 0, "CurrentProfile");
  reset(&f);
  bh_menu_edit_profile(&f.context, 1);
  CHECK(f.queue.write == 1 && f.queue.commands[0].kind == BH_TEXT_COMMAND_CLEAR,
        "NEW profile row has no body");
  for (i = 1; i <= 12; ++i) {
    reset(&f);
    ((unsigned char *)&f.profile)[0x1b + i] = 0;
    bh_menu_edit_motion(&f.context, i);
    CHECK(f.queue.write == 2 && !f.context.error,
          "all12 motion refresh cases use actual corresponding original table");
  }
  reset(&f);
  ((unsigned char *)&f.profile)[0x1c] = 'A';
  bh_menu_edit_motion(&f.context, 1);
  body(&f, 1, "Key 'A'");
  for (i = 1; i <= 9; ++i) {
    reset(&f);
    ((unsigned char *)&f.profile)[0x27 + i] = 'B';
    bh_menu_edit_primary(&f.context, i);
    body(&f, i, "Key 'B'");
    reset(&f);
    ((unsigned char *)&f.profile)[0x30 + i] = 'C';
    bh_menu_edit_secondary(&f.context, i);
    body(&f, i, "Key 'C'");
  }
  for (i = 1; i <= 14; ++i) {
    reset(&f);
    memset((char *)&f.profile + 0x3a, 0, 14);
    bh_menu_edit_camera(&f.context, i);
    CHECK(f.queue.write == 2 && !f.context.error,
          "all14 camera rows including swapped axes have exact source table "
          "lookup");
  }
  reset(&f);
  ((unsigned char *)&f.profile)[0x3a] = 5;
  bh_menu_edit_camera(&f.context, 1);
  CHECK(f.queue.write == 1,
        "original NULL key table entry retains clear-only body behavior");
  reset(&f);
  f.sessions.count = 1;
  strcpy(f.sessions.records[0].name, "123456789012345678901234");
  bh_menu_edit_sessions(&f.context, 2);
  body(&f, 2, "12345678901234567890");
  reset(&f);
  bh_menu_edit_sessions(&f.context, 3);
  body(&f, 3, "<Searching>");
  reset(&f);
  bh_menu_edit_sessions(&f.context, 1);
  CHECK(f.context.error && !f.queue.write,
        "original negative session array index rejected by source guard");
  reset(&f);
  strcpy((char *)f.network.bytes + 0x1c, "Some modem");
  bh_menu_edit_modem(&f.context, 1);
  body(&f, 1, "Some modem");
  reset(&f);
  bh_menu_edit_modem(&f.context, 2);
  body(&f, 2, "<Wait for Call>");
  reset(&f);
  strcpy((char *)f.network.bytes + 0x7c, "55512345");
  bh_menu_edit_modem(&f.context, 2);
  body(&f, 2, "55512345");
  reset(&f);
  f.network.bytes[0x1c] = 0;
  bh_menu_edit_tcpip(&f.context, 1);
  body(&f, 1, "<Auto Search>");
  reset(&f);
  strcpy((char *)f.network.bytes + 0x1c, "localhost");
  bh_menu_edit_tcpip(&f.context, 1);
  body(&f, 1, "localhost");
  {
    long values[5] = {3, 19200, 1, 2, 3};
    const char *expected[5] = {"3", NULL, NULL, NULL, NULL};
    expected[1] = bh_serial_baud_label(values[1]);
    expected[2] = bh_serial_stop_label(values[2]);
    expected[3] = bh_serial_parity_label(values[3]);
    expected[4] = bh_serial_flow_label(values[4]);
    for (i = 1; i <= 5; ++i) {
      reset(&f);
      memcpy(f.network.bytes + 0x18 + i * 4, values + i - 1, 4);
      bh_menu_edit_serial(&f.context, i);
      body(&f, i, expected[i - 1]);
    }
  }
  reset(&f);
  bh_menu_edit_motion(&f.context, 0);
  bh_menu_edit_motion(&f.context, 13);
  bh_menu_edit_primary(&f.context, 10);
  bh_menu_edit_secondary(&f.context, -1);
  bh_menu_edit_camera(&f.context, 15);
  CHECK(!f.queue.write, "switch defaults preserve rows");
  finish(&f);
}
static void actions(void) {
  Fixture f;
  unsigned long i, offset, before;
  short unlocked;
  BHMenuKeyCommit commits[4] = {bh_menu_keys_motion, bh_menu_keys_weapons,
                                bh_menu_keys_camera, bh_menu_keys_extra};
  void(__cdecl * actions[4])(BHMenuEdits *, long) = {
      bh_menu_action_motion, bh_menu_action_primary, bh_menu_action_secondary,
      bh_menu_action_camera};
  CHECK(setup(&f), "real menu action ownership fixture");
  f.bridge.kind = BH_MENU_EDIT_MOTION;
  for (i = 7; i <= 12; ++i) {
    reset(&f);
    offset = 0x1b + i;
    ((unsigned char *)&f.profile)[offset] = (unsigned char)(i <= 9 ? 16 : 4);
    before = f.pool.used;
    bh_menu_action_motion(&f.context, i);
    CHECK(!((unsigned char *)&f.profile)[offset] && f.pool.used == before,
          "whole motion axis action wraps correct17/5 enum without capture "
          "allocation");
    CHECK(f.queue.write == 2 && !f.context.error,
          "axis action invokes actual registered row refresh");
  }
  f.bridge.kind = BH_MENU_EDIT_CAMERA;
  for (i = 11; i <= 14; ++i) {
    reset(&f);
    offset = i == 11 ? 0x45 : i == 12 ? 0x44 : i == 13 ? 0x47 : 0x46;
    ((unsigned char *)&f.profile)[offset] = (unsigned char)(i <= 12 ? 16 : 4);
    bh_menu_action_camera(&f.context, i);
    CHECK(!((unsigned char *)&f.profile)[offset] && f.queue.write == 2,
          "whole camera axis action preserves original swapped field order and "
          "refresh");
  }
  for (i = 0; i < 4; ++i) {
    f.context.error = 0;
    before = f.pool.used;
    actions[i](&f.context, 1);
    CHECK(f.capture.entity && f.capture.commit == commits[i] &&
              f.pool.used == before + 1 && f.input->flags == 2 &&
              !f.context.error,
          "whole action allocates actual keycapture and enables cached mode1 "
          "with correct real commit");
    CHECK(bh_input_key_capture_cancel(&f.capture) && f.pool.used == before &&
              !f.input->flags,
          "real cancellation releases source capture without physical input");
  }
  bh_menu_action_mission(&f.context, -3);
  CHECK(f.mode == 1 && f.mission == -3,
        "whole mission action preserves original unrestricted signed index and "
        "mode1");
  unlocked = 3;
  memcpy(f.config.settings + 2, &unlocked, 2);
  bh_menu_action_ship(&f.context, 2);
  CHECK(f.ship == 2 && f.mission_menu.selection == 3 &&
            f.mission_menu.display_selection == 3,
        "whole ship action publishes actual ship and progress selection");
  for (i = 0; i < 9; ++i)
    CHECK(f.missions[i].action == (i <= 3 ? bh_menu_action_mission : NULL),
          "all9 mission callback identities reflect signed progress threshold");
  f.missions[2].action(&f.context, 2);
  CHECK(f.mode == 1 && f.mission == 2,
        "actual installed mission action is callable source");
  unlocked = -1;
  memcpy(f.config.settings + 2, &unlocked, 2);
  bh_menu_action_ship(&f.context, 0);
  CHECK(f.mission_menu.selection == -1 &&
            f.mission_menu.display_selection == -1,
        "negative signed progress retains original selection");
  for (i = 0; i < 9; ++i)
    CHECK(!f.missions[i].action,
          "negative progress disables every actual mission row");
  finish(&f);
}
int __cdecl bh_menu_edits_checkpoint(unsigned long *count) {
  BHMathEnvironment fp;
  checks = failures = 0;
  if (!bh_math_environment_enter(&fp))
    return 2;
  labels();
  actions();
  bh_math_environment_leave(&fp);
  if (count)
    *count = checks;
  printf(
      "{\"scenario\":\"menu-edits\",\"checks\":%lu,\"failures\":%lu,\"original_"
      "code_required\":false,\"play_calls\":0,\"physical_input_calls\":0}\n",
      checks, failures);
  return failures != 0;
}
