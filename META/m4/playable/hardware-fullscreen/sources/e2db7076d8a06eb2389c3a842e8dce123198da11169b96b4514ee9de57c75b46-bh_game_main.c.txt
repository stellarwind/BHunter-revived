#include "bh_game_app.h"
#include <string.h>
int WINAPI WinMain(HINSTANCE instance,HINSTANCE previous,LPSTR command,int show){char path[MAX_PATH],*end,*option;DWORD length;int result,check=0,hardware=0,fullscreen=0;
 (void)previous;for(option=strtok(command," \t");option;option=strtok(0," \t")){if(!strcmp(option,"--startup-check"))check=1;else if(!strcmp(option,"--hardware"))hardware=1;else if(!strcmp(option,"--fullscreen"))fullscreen=1;else return 2;}
 if(hardware)fullscreen=1;
 length=GetModuleFileNameA(instance,path,sizeof(path));if(!length||length>=sizeof(path))return 2;
 end=strrchr(path,'\\');if(!end)return 2;*end=0;if(!SetCurrentDirectoryA(path))return 2;
 /* Private preview option bits: bounded startup1, HAL2, fullscreen4. */
 result=bh_game_app_run(instance,"..\\..\\GAME_DATA",check|hardware*2|fullscreen*4,check?SW_HIDE:show);
 if(result&&!check)MessageBoxA(0,"The source preview stopped at an explicit dependency error. See build\\game\\startup.log for details.","BHunter source preview",MB_OK|MB_ICONERROR);
 return result;
}
