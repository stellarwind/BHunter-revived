/* Windowed presentation adapter: the recovered producer, queues, rasterizer
   and present owners run synchronously. The original worker/Flip path remains
   separately available; no fullscreen surface is pretended here. */
static int manual_loop(BHGameApp *a){MSG message;DWORD began,elapsed,rendered,presented;RECT destination;POINT origin;unsigned long i;float controls[3];char profile[512];
 if(a->startup_check)return failure(a,12,"Startup check cannot enter application frames");
 if(!bh_draw_queue_dispose(&a->queue)||!bh_frame_buffers_start(&a->frames,source_produce,a,30,3500*sizeof(BHDrawCommand),source_message,a))return failure(a,12,"Source frame ownership initialization failed");
 a->service.profile=1;a->service.input=&a->collection;a->service.entities=&a->pool;a->service.sound=&a->sound;a->service.frames=&a->frames;a->service.draw=&a->queue;for(i=0;i<3;++i)a->service.slot_draw[i]=a->slots+i;
 a->service.player_input=&a->global_state;a->service.paused=(const volatile LONG*)&a->player_state.paused;a->service.frame_counter=&a->frame;a->service.stop=&a->worker.stop;a->service.worker_done=&a->worker.done;
 a->text_capture.input=a->bootstrap.all_keys;a->key_capture.input=a->bootstrap.all_keys;
 stage(a,"Manual preview running; keyboard/mouse belong to the active game window");
 while(!a->quit&&!a->error){
  began=GetTickCount();while(PeekMessageA(&message,0,0,0,PM_REMOVE)){if(message.message==WM_QUIT){a->quit=1;break;}TranslateMessage(&message);DispatchMessageA(&message);}if(a->quit)break;
  if(GetForegroundWindow()!=a->window||IsIconic(a->window)){a->focus_sample=0;Sleep(20);continue;}
  if(!a->focus_sample){if(!game_recenter(160,100))return failure(a,12,"Windowed input recenter failed");memset(a->snapshot.mouse_axes,0,sizeof(a->snapshot.mouse_axes));memset(a->snapshot.pair4,0,sizeof(a->snapshot.pair4));a->collection.delta[0]=a->collection.delta[1]=0;a->focus_sample=1;}
  a->flyby.input_profile=a->profile;a->menu_settings.input_profile=a->profile;a->player_create.profile=a->profile;
  a->actor_route.tick=(unsigned short)a->player_state.tick;a->vehicle_draw.tick=a->actor_route.tick;a->script_draw.tick=a->actor_route.tick;a->actor_launch.tick=a->actor_route.tick;a->actor_launch.mode=a->actor_damage.mode=a->player_state.mode;
  a->weapon_target.actors=a->vehicle_lifetime.head;a->weapon_arc.head=a->vehicle_lifetime.head;a->pursuit.actor_head=a->vehicle_lifetime.head;
  a->eye=a->flyby_state.camera.position;a->angles=a->flyby_state.camera.angles;a->contacts.has_camera=1;a->contacts.camera_yaw=a->yaw=a->angles.x;
  a->frames.produce(a->frames.produce_user);
  if(a->error)break;
  if(a->player_tick.error||a->player_query.error||a->vm.error||a->lifetime.timers.error||a->menu.error||a->flyby.error||a->actor_construct.error||a->actor_response.error){char text[256];sprintf(text,"Source dependency failure: player=%d query=%d vm=%d timer=%d menu=%d flyby=%d actor=%d response=%d",a->player_tick.error,a->player_query.error,a->vm.error,a->lifetime.timers.error,a->menu.error,a->flyby.error,a->actor_construct.error,a->actor_response.error);failure(a,12,text);break;}
  rendered=GetTickCount();if(!render_frame(a,a->slots+a->frames.latest))break;rendered=GetTickCount()-rendered;
  GetClientRect(a->window,&destination);origin.x=origin.y=0;ClientToScreen(a->window,&origin);OffsetRect(&destination,origin.x,origin.y);
  presented=GetTickCount();if(IDirectDrawSurface_Blt(a->primary,&destination,a->target,0,DDBLT_WAIT,0)!=DD_OK){failure(a,13,"Windowed presentation failed");break;}presented=GetTickCount()-presented;
  if(a->frame<=5||a->frame%30==0){memcpy(controls,(char*)a->player+0x18c,12);sprintf(profile,"Frame profile: frame=%lu mode=%ld total=%lu input=%lu step=%lu refresh=%lu sound=%lu render=%lu blit=%lu commands=%lu mouse=%ld,%ld filtered=%.5g,%.5g controls=%.5g,%.5g,%.5g angular=%.5g,%.5g,%.5g keys_LR=%u,%u panel=%d arena=%lu ring=%ld",a->frame,a->player_state.mode,GetTickCount()-began,a->service.input_ms,a->service.step_ms,a->service.refresh_ms,a->service.sound_ms,rendered,presented,a->queue.count,a->collection.delta[0],a->collection.delta[1],a->snapshot.mouse_axes[0],a->snapshot.mouse_axes[1],controls[0],controls[1],controls[2],a->player_motion.angular_velocity.x,a->player_motion.angular_velocity.y,a->player_motion.angular_velocity.z,a->snapshot.current[37],a->snapshot.current[39],a->panel.error,a->messages.arena_used,a->messages.ring_count);stage(a,profile);}
  elapsed=GetTickCount()-began;if(elapsed<(DWORD)a->frames.delay_ms)Sleep((DWORD)a->frames.delay_ms-elapsed);
 }
 return !a->error;
}
static void cleanup(BHGameApp *a){unsigned long i;
 bh_loading_progress_remove(&a->loading_progress);bh_loading_images_dispose(&a->loading_images);
 /* No callback may survive its context. Quiesce all callbacks before releasing
    metadata; shared models/textures are released after their borrowers. */
 if(a->panel.entity)bh_message_panel_close(&a->panel);bh_flyby_scene_unbind(&a->flyby);bh_player_tick_unbind(&a->player_tick);bh_player_query_unbind(&a->player_query);
 if(a->physics.pool)bh_physics_step_dispose(&a->physics);
 bh_scene_bindings_dispose(&a->scene_bindings);
 for(i=0;i<N(a->block_queries);++i)if(a->block_queries[i].geometry)bh_segment_query_dispose(a->block_queries+i);
 if(a->pool.chunks){bh_editor_world_dispose(&a->editor_world);while(a->pool.active)if(!bh_entity_release(&a->pool,a->pool.active))break;bh_weapon_factory_close(&a->weapons);bh_entities_dispose(&a->pool);}
 if(a->player_collision.bsp.allocator.release)bh_collision_model_dispose(&a->player_collision);
 bh_script_blocks_dispose(&a->blocks);bh_navigation_dispose(&a->navigation);bh_rail_paths_dispose(&a->rails);
 bh_text_queue_init(&a->text);for(i=0;i<BH_BOOTSTRAP_MENUS;++i)bh_menu_rows_dispose(&a->bootstrap.menus[i].object.rows);
 if(a->bootstrap.confirmation)bh_atlas_page_dispose(&a->atlas,a->bootstrap.confirmation);
 if(a->bootstrap.menu_font)bh_atlas_page_dispose(&a->atlas,a->bootstrap.menu_font);
 if(a->bootstrap.graffiti)bh_atlas_page_dispose(&a->atlas,a->bootstrap.graffiti);
 if(a->panel.page)bh_atlas_page_dispose(&a->atlas,a->panel.page);
 bh_atlas_context_dispose(&a->atlas);bh_render_resources_dispose(&a->resources);bh_horizon_color_dispose(&a->horizon_colors);
 bh_input_devices_dispose(&a->devices);bh_player_input_owner_dispose(&a->inputs);bh_config_dispose(&a->config);bh_sound_queue_dispose(&a->sound);
 bh_network_game_dispose(&a->network_game);if(a->provider_startup.com_initialized)CoUninitialize();bh_provider_catalog_dispose(&a->providers);bh_network_transport_dispose(&a->transport);if(!a->queue.allocator.release)memset(&a->queue,0,sizeof(a->queue));bh_frame_buffers_dispose(&a->frames);bh_draw_queue_dispose(&a->queue);
 bh_render_startup_dispose(&a->renderer);if(a->target)IDirectDrawSurface_Release(a->target);if(a->primary)IDirectDrawSurface_Release(a->primary);if(a->window_clipper)IDirectDrawClipper_Release(a->window_clipper);if(a->draw)IDirectDraw_Release(a->draw);
 if(a->window)DestroyWindow(a->window);UnregisterClassA("BHunterRestoredSource",a->instance);input_window=0;bh_vfile_shutdown();
 free(a->motion_bindings);free(a->world_responses);free(a->pulse_indices);free(a->actor_rooms);free(a->city_ids);free(a->rooms);free(a->scheduled);free(a->scene_inputs);free(a->segment_rooms);free(a->physics_worlds);free(a->reset_nodes);
}
int __cdecl bh_game_app_run(HINSTANCE instance,const char *directory,int startup_check,int show){BHGameApp *a;unsigned long bytes,crc;int result;FILE *summary;
 a=(BHGameApp*)calloc(1,sizeof(*a));if(!a)return 2;a->instance=instance;a->startup_check=startup_check;a->width=640;a->height=480;a->log=fopen(startup_check?"startup-check.log":"startup.log","w");
 if(!directory||strlen(directory)>450){if(a->log)fclose(a->log);free(a);return 2;}
 sprintf(a->archive,"%s\\Bhunter.bin",directory);strcpy(a->config_path,"source-preview.cfg");
 if(!bh_math_environment_enter(&a->math)){if(a->log)fclose(a->log);free(a);return 2;}
 stage(a,"Restored source startup; no original executable loaded");
 if(!bh_game_assets_verify(a->archive,&bytes,&crc)){failure(a,2,"Archive missing or differs from pinned data");goto done;}
 if(!platform(a,show)||!source_owners(a))goto done;
 if(!bh_vfile_mount(a->archive)){failure(a,2,"Archive mount failed");goto done;}
 if(!loading_begin(a))goto done;
 if(!bh_sound_startup(&a->sound,a->window,&a->sound_stats)){failure(a,6,"Native sound initialization failed");goto done;}
 bh_provider_startup(&a->provider_startup);
 if(!bh_texture_set_path(&a->renderer.textures,"textures",0)||!bh_bootstrap_resources(&a->bootstrap,5)||!bh_bootstrap_resources(&a->bootstrap,6)){char message[256];sprintf(message,"Engine resources failed: game=%d sound=%d texture=%d loaded=%lu",a->bootstrap.error,a->sounds.error,a->renderer.textures.error,a->bootstrap.resources_loaded);failure(a,6,message);goto done;}
 if(!bh_game_bootstrap(&a->bootstrap)){char message[256];sprintf(message,"Bootstrap failed: game=%d blocks=%d parser=%d world=%d vm=%d player=%d flyby=%d",a->bootstrap.error,a->blocks.error,a->parser.error,a->world_load.error,a->vm.error,a->player_create.error,a->flyby_construct.error);failure(a,a->error?a->error:10,message);goto done;}
 if(startup_check){stage(a,"Checking initial scene presentation without input or entity steps");if(!bh_draw_queue_begin(&a->queue)||!bh_flyby_scene_draw(&a->flyby)){failure(a,13,"Initial scene composition failed");goto done;}bh_menu_runtime_refresh_entity(a->menu.root->entity,0);if(!render_frame(a,&a->queue))goto done;
  if(a->view.near_z!=a->screen_clip.near_z||a->view.near_z!=a->primitive.raster.near_z||a->schedule.viewport==&a->view){failure(a,13,"Scene traversal corrupted render clipping state");goto done;}
  stage(a,"Checking static source beam/boost/logo/panel bindings after scene composition");if(!startup_effect_checks(a))goto done;}
 if(!loading_finish(a))goto done;
 a->initialized=1;stage(a,"Real-data startup complete; no input collection or application frame executed");
 summary=fopen(startup_check?"startup-check-summary.json":"startup-summary.json","w");if(summary){fprintf(summary,"{\"passed\":true,\"blocks\":%lu,\"placements\":%lu,\"navigation_nodes\":%lu,\"pool_entities\":%lu,\"menus\":%lu,\"rows\":%lu,\"resources\":%lu,\"physics_bodies\":%lu,\"flyby_constructor_settling_calls\":%lu,\"input_collection_calls\":%lu,\"entity_step_calls\":%lu,\"entity_refresh_calls\":%lu,\"application_frames\":%lu,\"native_primary_sound_play_calls\":%lu}\n",a->blocks.count,a->editor_world.count,a->navigation.node_count,a->pool.used,a->bootstrap.menus_created,a->bootstrap.rows_created,a->bootstrap.resources_loaded,a->physics.body_count,a->flyby_construct.settled,a->service.input_calls,a->service.step_calls,a->service.refresh_calls,a->frame,a->sound_stats.play_calls);fclose(summary);}
 if(!startup_check){bh_force_feedback_init(&a->force_init,instance,a->window);manual_loop(a);}
done:
 result=a->error;cleanup(a);stage(a,"Source owners released");bh_math_environment_leave(&a->math);if(a->log)fclose(a->log);free(a);return result;
}
