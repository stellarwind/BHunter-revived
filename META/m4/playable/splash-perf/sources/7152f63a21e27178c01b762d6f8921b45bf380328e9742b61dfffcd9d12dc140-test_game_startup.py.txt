"""Run only the explicit bounded source-game startup check. No game input."""
import hashlib
import json
from pathlib import Path
import subprocess
import time

ROOT = Path(__file__).resolve().parents[1]


def main():
    out = ROOT / 'build/game'
    exe = out / 'BHunter-source.exe'
    build = json.loads((out / 'build.json').read_text())
    digest = hashlib.sha256(exe.read_bytes()).hexdigest()
    assert build['passed'] and build['executable_sha256'] == digest
    for source, expected in build['sources'].items():
        assert hashlib.sha256((ROOT / source).read_bytes()).hexdigest() == expected, source
    summary_path = out / 'startup-check-summary.json'
    if summary_path.exists():
        summary_path.unlink()
    began = time.monotonic()
    result = dict(passed=False, executable_sha256=digest,
                  command=[str(exe), '--startup-check'], timeout_seconds=45,
                  original_code_required=False, input_automation=False,
                  gameplay_automation=False)
    try:
        child = subprocess.run(result['command'], cwd=out, capture_output=True,
                               text=True, timeout=45)
        result.update(exit_code=child.returncode, stdout=child.stdout, stderr=child.stderr)
        if summary_path.exists():
            summary = json.loads(summary_path.read_text())
            result['summary'] = summary
            result['passed'] = child.returncode == 0 and summary['passed'] and all(
                summary[key] == 0 for key in ('input_collection_calls', 'entity_step_calls',
                                             'entity_refresh_calls', 'application_frames'))
    except subprocess.TimeoutExpired:
        result.update(timeout=True, reason='Startup child terminated at the explicit bound')
    result['seconds'] = time.monotonic() - began
    result['log'] = (out / 'startup-check.log').read_text() if (out / 'startup-check.log').exists() else ''
    (out / 'startup-check.json').write_text(json.dumps(result, indent=2) + '\n')
    print(json.dumps(result, indent=2))
    return 0 if result['passed'] else 1


if __name__ == '__main__':
    raise SystemExit(main())
