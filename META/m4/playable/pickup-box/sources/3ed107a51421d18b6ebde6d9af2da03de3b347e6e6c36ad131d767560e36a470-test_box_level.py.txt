"""Bounded hardware-only playthrough of the separate box diagnostic."""
import csv,hashlib,json,math,shutil,subprocess,time
from pathlib import Path
ROOT=Path(__file__).resolve().parents[1]
OUT=ROOT/'build/box-test'
def run_once():
    start=time.monotonic();result=dict(passed=False,hardware_only=True,software_renderer_executed=False,original_code_executed=False,automatic_gameplay=True,scope='user-authorized isolated box level',timeout_seconds=45)
    exe=OUT/'BHunter-box.exe';result['executable_sha256']=hashlib.sha256(exe.read_bytes()).hexdigest()
    trace=OUT/'box-trace.csv'
    if trace.exists():trace.replace(OUT/'box-trace-previous.csv')
    try:
        p=subprocess.run([str(exe),'--box-auto'],cwd=OUT,timeout=45,capture_output=True)
        result['exit_code']=p.returncode
    except subprocess.TimeoutExpired:result['timeout']=True
    result['seconds']=time.monotonic()-start
    rows=list(csv.DictReader(trace.open())) if trace.exists() else []
    result['ticks']=len(rows)
    checks={}
    if rows:
        last=rows[-1]
        checks['finite_state']=all(math.isfinite(float(v)) for r in rows for v in r.values())
        checks['inside_box']=all(-40<=float(r['x'])<=40 and -20<=float(r['y'])<=20 and -80<=float(r['z'])<=80 for r in rows)
        checks['outside_solid_post']=all(not(16<float(r['x'])<24 and -20<float(r['y'])<8 and -34<float(r['z'])<-26) for r in rows)
        checks['pickup_removed']=any(r['pickup_alive']=='0' for r in rows)
        checks['pickup_increased_inventory']=max(int(r['ammo']) for r in rows)>int(rows[0]['ammo'])
        checks['collision_contact']=int(last['contacts'])>0
        checks['shots_emitted']=int(last['shots'])>0
        checks['damage_delivered']=int(last['damage_packets'])>0
        checks['ai_moved']=any(sum((float(r['ai_'+k])-float(rows[0]['ai_'+k]))**2 for k in 'xyz')>1 for r in rows if r['ai_alive']=='1')
        checks['target_destroyed']=any(r['ai_alive']=='0' for r in rows)
        checks['objective_completed']=last['objective']=='1'
        checks['wreck_spawned_pickup']=last['drop_seen']=='1'
        checks['wreck_pickup_collected']=last['drop_taken']=='1'
        result['last']=last
    result['checks']=checks
    result['passed']=result.get('exit_code')==0 and len(rows)==900 and bool(checks) and all(checks.values())
    result['limits']=['Source playthrough; original numerical/PC24 equivalence is not implied by scenario success.']
    result['trace_sha256']=hashlib.sha256(trace.read_bytes()).hexdigest() if trace.exists() else None
    return result

def main():
    results=[]
    for index in range(2):
        result=run_once();results.append(result)
        if (OUT/'box-trace.csv').exists():shutil.copyfile(OUT/'box-trace.csv',OUT/('box-trace-run%d.csv'%(index+1)))
        if not result['passed']:break
    result=dict(passed=len(results)==2 and all(r['passed'] for r in results) and results[0]['trace_sha256']==results[1]['trace_sha256'],
                deterministic_trace_match=len(results)==2 and results[0]['trace_sha256']==results[1]['trace_sha256'],
                runs=results,limits=['Repeatability of recorded source state is not original-game equivalence or a complete state/RNG comparison.'])
    (OUT/'playthrough.json').write_text(json.dumps(result,indent=2)+'\n');print(json.dumps(result,indent=2))
    return 0 if result['passed'] else 1
if __name__=='__main__':raise SystemExit(main())
