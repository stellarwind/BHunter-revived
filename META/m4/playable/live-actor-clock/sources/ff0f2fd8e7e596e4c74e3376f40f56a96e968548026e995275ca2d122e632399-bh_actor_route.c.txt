static void read_field(const BHEntity *a,unsigned long offset,void *out,unsigned long n){memcpy(out,(const unsigned char*)a+offset,n);}
static void write_field(BHEntity *a,unsigned long offset,const void *in,unsigned long n){memcpy((unsigned char*)a+offset,in,n);}
static float stored(double value){float out;bh_round_float(&out,value);return out;}
#define READ(member,offset) read_field(a,offset,&s->member,sizeof(s->member))
#define WRITE(member,offset) write_field(a,offset,&s->member,sizeof(s->member))
unsigned short __cdecl bh_actor_route_now(const BHActorRouteContext *c) {
    return c->live_tick?(unsigned short)*c->live_tick:c->tick;
}
void __cdecl bh_actor_route_set_tick(BHActorRouteContext *c,unsigned short tick) {
    if(c->live_tick)*c->live_tick=(*c->live_tick&0xffff0000UL)|tick;
    else c->tick=tick;
}
int __cdecl bh_actor_route_read(const BHEntity *a,BHActorRouteState *s) {
    if(!a || !s)return 0;
    READ(current,0x1a4);READ(next,0x1a8);
    READ(controls[0],0x120);READ(controls[1],0x144);READ(controls[2],0x150);READ(controls[3],0x12c);
    READ(portal_center,0x138);READ(phase,0x180);READ(duration,0x184);READ(speed,0x188);
    READ(u,0x18c);READ(v,0x190);READ(speed_scale,0x198);READ(jitter,0x1b0);
    READ(sound,0x104);READ(sound_frames,0x108);READ(random,0x1e8);READ(tick,0x1e4);READ(duration_ticks,0x1e6);return 1;
}
int __cdecl bh_actor_route_write(BHEntity *a,const BHActorRouteState *s) {
    if(!a || !s)return 0;
    WRITE(current,0x1a4);WRITE(next,0x1a8);
    WRITE(controls[0],0x120);WRITE(controls[1],0x144);WRITE(controls[2],0x150);WRITE(controls[3],0x12c);
    WRITE(portal_center,0x138);WRITE(phase,0x180);WRITE(duration,0x184);WRITE(speed,0x188);
    WRITE(u,0x18c);WRITE(v,0x190);WRITE(speed_scale,0x198);WRITE(jitter,0x1b0);
    WRITE(sound,0x104);WRITE(sound_frames,0x108);WRITE(random,0x1e8);WRITE(tick,0x1e4);WRITE(duration_ticks,0x1e6);return 1;
}
#undef READ
#undef WRITE
static int fail(BHActorRouteContext *c,BHActorRouteError e){if(c)c->error=e;return 0;}
static BHPickupRoom *room_binding(BHActorRouteContext *c,const BHWorldGeometry *world) {
    unsigned long i;for(i=0;i<c->room_count;++i)if(c->rooms[i].world==world)return c->rooms+i;return NULL;
}
static BHEntity *actor_link(const BHEntity *a,unsigned long offset) {BHEntity *out;read_field(a,offset,&out,4);return out;}
static void set_link(BHEntity *a,unsigned long offset,BHEntity *value){write_field(a,offset,&value,4);}
static int vector_finite(const BHVec3 *v){return _finite(v->x) && _finite(v->y) && _finite(v->z);}
static void difference(BHVec3 *v,const BHVec3 *a,const BHVec3 *b) {
    v->x=stored((double)a->x-b->x);v->y=stored((double)a->y-b->y);v->z=stored((double)a->z-b->z);
}
static void tangent(BHVec3 *out,const BHVec3 *point,const BHVec3 *delta) {
    out->x=stored((double)point->x-(double)delta->x*(-1.3));
    out->y=stored((double)point->y-(double)delta->y*(-1.3));
    out->z=stored((double)point->z-(double)delta->z*(-1.3));
}
static int portal_valid(const BHPortal *p) {
    unsigned long i;if(p->direction>6 || p->direction==3)return 0;
    for(i=0;i<4;++i)if(!_finite(p->scratch[i]))return 0;
    return p->scratch[0]<=p->scratch[1] && p->scratch[2]<=p->scratch[3];
}
static int duration_ticks(double duration,unsigned short *out) {
    double ticks=duration*30.0;
    if(!_finite(ticks) || ticks< -9223372036854775808.0 || ticks>=9223372036854775808.0)return 0;
    *out=(unsigned short)(__int64)ticks;return 1;
}
int __cdecl bh_actor_route_promote(BHActorRouteContext *c,BHEntity *a,long incoming,long outgoing) {
    BHActorRouteState s;BHPickupRoom *old,*current;BHWorldGeometry *world;
    const BHPortal *in,*out;BHEntity *next,*previous,*at;unsigned long n;
    BHVec3 delta,center;float u,v,cu,cv,plane,first_length,width;
    double length,duration,test_width;
    if(!c || !a || !c->rooms || !c->room_count || !c->actor_limit)return fail(c,BH_ACTOR_ROUTE_ARGUMENT);
    if(!bh_math_environment_supported())return fail(c,BH_ACTOR_ROUTE_NUMERICAL);
    bh_actor_route_read(a,&s);old=room_binding(c,s.current);current=room_binding(c,s.next);world=s.next;
    if(!old || !current || !world->model || world->model->portal_count>6 ||
       world->navigation.count!=world->model->portal_count || incoming<0 || outgoing<0 ||
       (unsigned long)incoming>=world->navigation.count || (unsigned long)outgoing>=world->navigation.count)
        return fail(c,BH_ACTOR_ROUTE_TOPOLOGY);
    in=world->navigation.portals+incoming;out=world->navigation.portals+outgoing;
    if(!out->neighbor || !room_binding(c,(BHWorldGeometry*)out->neighbor) || !portal_valid(in) || !portal_valid(out))return fail(c,BH_ACTOR_ROUTE_TOPOLOGY);
    next=actor_link(a,0xf4);previous=actor_link(a,0xf8);
    if((previous?actor_link(previous,0xf4)!=a:old->actors!=a) || (next && actor_link(next,0xf8)!=a))return fail(c,BH_ACTOR_ROUTE_TOPOLOGY);
    at=old->actors;n=0;while(at && at!=a && n<c->actor_limit){at=actor_link(at,0xf4);++n;}
    if(at!=a || n>=c->actor_limit)return fail(c,BH_ACTOR_ROUTE_TOPOLOGY);
    if(!vector_finite(&s.controls[3]) || !vector_finite(&world->minimum) || !vector_finite(&world->maximum) ||
       !_finite(s.u) || !_finite(s.v) || !_finite(s.speed) || s.speed<=0 || !_finite(s.speed_scale) || s.speed_scale<=0)
        return fail(c,BH_ACTOR_ROUTE_NUMERICAL);
    s.current=world;s.next=(BHWorldGeometry*)out->neighbor;s.controls[0]=s.controls[3];
    u=stored((1.0-(double)s.u)*out->scratch[1]+(double)s.u*out->scratch[0]);
    v=stored((1.0-(double)s.v)*out->scratch[3]+(double)s.v*out->scratch[2]);
    cu=stored(((double)out->scratch[1]+out->scratch[0])*0.5);
    cv=stored(((double)out->scratch[3]+out->scratch[2])*0.5);plane=stored((double)out->plane);
    switch(out->direction&3UL) {
    case 0:s.controls[3].x=plane;s.controls[3].y=u;s.controls[3].z=v;s.portal_center.x=plane;s.portal_center.y=cu;s.portal_center.z=cv;break;
    case 1:s.controls[3].x=u;s.controls[3].y=plane;s.controls[3].z=v;s.portal_center.x=cu;s.portal_center.y=plane;s.portal_center.z=cv;break;
    default:s.controls[3].x=u;s.controls[3].y=v;s.controls[3].z=plane;s.portal_center.x=cu;s.portal_center.y=cv;s.portal_center.z=plane;break;
    }
    if(out->direction==(in->direction^4UL) && (world->query_flags&8)) {
        s.controls[3]=s.controls[0];((float*)&s.controls[3])[out->direction&3UL]=plane;
        difference(&delta,s.controls+3,s.controls);s.controls[1]=s.controls[3];s.controls[2]=s.controls[0];
        length=bh_vec3_snap_axis(&delta);length+=length;
    } else {
        center.x=stored(((double)world->minimum.x+world->maximum.x)*0.5);
        center.z=stored(((double)world->minimum.z+world->maximum.z)*0.5);
        center.y=stored(((double)s.controls[3].y+s.controls[0].y)*0.5);
        difference(&delta,&center,s.controls);first_length=stored(bh_vec3_snap_axis(&delta));
        tangent(s.controls+1,s.controls,&delta);
        difference(&delta,&center,s.controls+3);length=bh_vec3_snap_axis(&delta)+first_length;
        tangent(s.controls+2,s.controls+3,&delta);
        if(incoming==outgoing) {
            s.controls[1].x=stored((double)s.controls[1].x-delta.z);s.controls[1].y=stored((double)s.controls[1].y-2);
            s.controls[1].z=stored((double)s.controls[1].z+delta.x);
            s.controls[2].x=stored((double)s.controls[2].x+delta.z);s.controls[2].y=stored((double)s.controls[2].y-(-2));
            s.controls[2].z=stored((double)s.controls[2].z-delta.x);
        }
    }
    s.phase=0;s.duration=stored(length/((double)s.speed*s.speed_scale));duration=s.duration;
    if(!duration_ticks(duration,&s.duration_ticks))return fail(c,BH_ACTOR_ROUTE_NUMERICAL);
    s.tick=bh_actor_route_now(c);
    test_width=(double)out->scratch[1]-out->scratch[0];width=stored(test_width);
    if(test_width<30) {
        test_width=(double)out->scratch[3]-out->scratch[2];if(test_width>width)width=stored(test_width);
        if(width<30) {
            test_width=(double)in->scratch[1]-in->scratch[0];if(test_width>width)width=stored(test_width);
            if(width<30) {
                test_width=(double)in->scratch[3]-in->scratch[2];if(test_width>width)width=stored(test_width);
                if(width<30) {
                    duration=(double)s.duration/(((double)width-(-10.0))*0.025);
                    s.duration=stored(duration);
                    if(!duration_ticks(duration,&s.duration_ticks))return fail(c,BH_ACTOR_ROUTE_NUMERICAL);
                }
            }
        }
    }
    if(!_finite(s.duration))return fail(c,BH_ACTOR_ROUTE_NUMERICAL);
    if(previous)set_link(previous,0xf4,next);else old->actors=next;
    if(next)set_link(next,0xf8,previous);
    next=current->actors;set_link(a,0xf4,next);current->actors=a;set_link(a,0xf8,NULL);
    if(next)set_link(next,0xf8,a);
    bh_actor_route_write(a,&s);write_field(a,0xc4,&s.current,4);c->error=BH_ACTOR_ROUTE_OK;return 1;
}
static unsigned long random_next(unsigned long state){return ((state*0xe3570000UL)|((state*0xe357UL)>>16))^state;}
int __cdecl bh_actor_route_advance(BHActorRouteContext *c,BHEntity *a) {
    BHActorRouteState s;BHWorldGeometry *room;long incoming=-1,outgoing;unsigned long i,first,second;
    double factor;
    if(!c || !a || !c->rooms || !c->room_count)return fail(c,BH_ACTOR_ROUTE_ARGUMENT);
    if(!bh_math_environment_supported())return fail(c,BH_ACTOR_ROUTE_NUMERICAL);
    bh_actor_route_read(a,&s);
    if(s.sound_frames==1 && s.sound!=BH_SOUND_INVALID && !c->sound)return fail(c,BH_ACTOR_ROUTE_ARGUMENT);
    if(s.sound_frames) {
        --s.sound_frames;write_field(a,0x108,&s.sound_frames,4);
        if(!s.sound_frames && s.sound!=BH_SOUND_INVALID){bh_sound_stop(c->sound,s.sound);s.sound=BH_SOUND_INVALID;write_field(a,0x104,&s.sound,4);}
    }
    room=s.next;
    if(!room || !room->model || room->model->portal_count>6 || room->navigation.count!=room->model->portal_count)return fail(c,BH_ACTOR_ROUTE_TOPOLOGY);
    for(i=0;i<room->navigation.count;++i)if((BHWorldGeometry*)room->navigation.portals[i].neighbor==s.current || incoming<0)incoming=(long)i;
    if(room->query_flags&8){if(!bh_route_guided_exit(a,room,incoming,&outgoing))return fail(c,BH_ACTOR_ROUTE_TOPOLOGY);}
    else if(!bh_route_choose_exit(a,room,NULL,incoming,&outgoing))return fail(c,BH_ACTOR_ROUTE_TOPOLOGY);
    read_field(a,0x1e8,&s.random,4);first=random_next(s.random);second=random_next(first);
    factor=1.0-(double)s.jitter;
    s.u=stored((double)s.u*factor-((double)(first&0x7fff)*s.jitter)*(-1.0/32768.0));
    s.v=stored((double)s.v*factor-((double)(second&0x7fff)*s.jitter)*(-1.0/32768.0));s.random=second;
    write_field(a,0x18c,&s.u,4);write_field(a,0x190,&s.v,4);write_field(a,0x1e8,&s.random,4);
    return bh_actor_route_promote(c,a,incoming,outgoing);
}
