#include "bh_actor_vehicle_draw.h"
#include "bh_draw_submit.h"
#include "bh_camera_curve.h"
#include "bh_rotation.h"
#include "bh_round.h"
#include "bh_fp.h"
#include <math.h>
#include <float.h>
#include <stdlib.h>
#include <string.h>
static float stored(double x){float f;bh_round_float(&f,x);return f;}
int __cdecl bh_actor_vehicle_draw(BHActorVehicleDrawContext *c,BHEntity *e){
 BHActorRouteState s;BHVec3 before,p,after,offset,angles,velocity,local,glow;
 BHMat34 matrix;unsigned short elapsed;short se,sd;float phase,dx,dz,px,pz,size,alpha,angle;
 double bank,clock;float fade;unsigned long resource,sound_resource,damage_frame,i;static const float xs[4]={.6f,.2f,-.2f,-.6f};static const double rates[4]={25.5,18.5,28.5,23.5};
 if(!c||!c->queue||!c->sound||!e){if(c)c->error=BH_ACTOR_ROUTE_ARGUMENT;return 0;}
 if(!bh_math_environment_supported()){c->error=BH_ACTOR_ROUTE_NUMERICAL;return 0;}
 bh_actor_route_read(e,&s);elapsed=(unsigned short)((c->clock?bh_actor_route_now(c->clock):c->tick)-s.tick);memcpy(&se,&elapsed,2);memcpy(&sd,&s.duration_ticks,2);
 if(!sd){c->error=BH_ACTOR_ROUTE_NUMERICAL;return 0;}
 phase=stored((double)se/sd);
 bh_camera_curve(stored((double)phase-1.0/30.0),&before,s.controls,s.controls+1,s.controls+2,s.controls+3);
 bh_camera_curve(phase,&p,s.controls,s.controls+1,s.controls+2,s.controls+3);
 bh_camera_curve(stored((double)phase-(-1.0/30.0)),&after,s.controls,s.controls+1,s.controls+2,s.controls+3);
 memcpy(&offset,(char*)e+0x15c,12);
 for(i=0;i<3;++i){((float*)&before)[i]=stored((double)((float*)&offset)[i]+((float*)&before)[i]);((float*)&p)[i]=stored((double)((float*)&offset)[i]+((float*)&p)[i]);((float*)&after)[i]=stored((double)((float*)&offset)[i]+((float*)&after)[i]);}
 dx=stored((double)after.x-p.x);dz=stored((double)after.z-p.z);px=stored((double)p.x-before.x);pz=stored((double)p.z-before.z);
 bank=(((double)pz*dx-(double)px*dz)/sqrt((double)dz*dz+(double)dx*dx))/sqrt((double)pz*pz+(double)px*px);
 angles.x=stored(atan2(dx,dz));angles.y=0;angles.z=stored(bank*3.141592653589793);
 if(c->clock&&c->clock->nonfinite_curve&&(!_finite(angles.x)||!_finite(angles.y)||!_finite(angles.z)))
  c->clock->nonfinite_curve(c->clock->nonfinite_user,0x434920,e,&s,bh_actor_route_now(c->clock),&before,&p,&after,&angles);
 memcpy(&resource,(char*)e+0x114,4);bh_draw_submit_resource(c->queue,0,resource,&p,&angles,1);
 memcpy(&fade,(char*)e+0x19c,4);
 if(fade<1){size=stored(4.0-(double)rand()*(-1.0/16384.0));alpha=stored(1.0-fade);damage_frame=(unsigned long)rand()&7;angle=stored((double)rand()*(1.0/4096.0));bh_draw_billboard(c->queue,3,c->damage_resource,&p,size,size,angle,damage_frame,alpha);}
 if(resource==c->glowing_model){
  memset(&matrix,0,sizeof(matrix));matrix.m[0][0]=matrix.m[1][1]=matrix.m[2][2]=1;
  bh_mat34_translate_world(p.x,p.y,p.z,&matrix);bh_mat34_rotate_y(angles.x,&matrix);bh_mat34_rotate_x(angles.y,&matrix);bh_mat34_rotate_z(angles.z,&matrix);
  clock=(double)c->animation_clock*.03125;
  for(i=0;i<4;++i){local.x=xs[i];local.y=-.95f;local.z=-.65f;bh_mat34_transform_point(&matrix,&local,&glow);alpha=stored((sin(clock*rates[i])-(-1.0))*.5);bh_draw_billboard(c->queue,3,i<2?c->glow_first:c->glow_second,&glow,2,2,0,0,alpha);}
 }
 velocity.x=stored((double)fade*((double)after.x-p.x)*30);velocity.y=stored((double)fade*((double)after.y-p.y)*30);velocity.z=stored(((double)after.z-p.z)*fade*30);
 memcpy(&sound_resource,(char*)e+0x10c,4);s.sound_frames=2;if(s.sound==BH_SOUND_INVALID)s.sound=bh_sound_request(c->sound,sound_resource,1,1,&p,&velocity);else bh_sound_update(c->sound,s.sound,1,1,&p,&velocity);
 memcpy((char*)e+0x104,&s.sound,4);memcpy((char*)e+0x108,&s.sound_frames,4);c->error=BH_ACTOR_ROUTE_OK;return 1;
}
