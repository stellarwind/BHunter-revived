static unsigned long checks;static int failures;
static void check(int ok,const char *name,int line){++checks;if(!ok){++failures;printf("FAIL pickup draw %d: %s\n",line,name);}}
#define CHECK(x,n) check((x),(n),__LINE__)
static float val(const unsigned long *word){float out;memcpy(&out,word,4);return out;}
static void setl(BHEntity *e,unsigned long offset,long v){memcpy(e->payload+offset,&v,4);}
static void setf(BHEntity *e,unsigned long offset,float v){memcpy(e->payload+offset,&v,4);}
static int closef(float a,float b){return fabs((double)a-b)<.00005;}
static int emitted;static float corners[4][4];static unsigned long color;
static int __cdecl capture(BHClipVertex *p,BHTextureImage *image,unsigned long shade,void *user){
    unsigned long i;BHClipVertex *at=p;(void)image;(void)user;
    for(i=0;i<4;++i){corners[i][0]=at->sx;corners[i][1]=at->sy;corners[i][2]=at->u;corners[i][3]=at->v;at=at->next;}
    if(at!=p)return 0;color=shade;emitted++;return 1;
}
static BHTextureImage image;
static void *__cdecl allocation(unsigned long n,void *user){(void)user;return malloc(n);}
static void __cdecl release(void *p,void *user){(void)user;free(p);}
static BHTextureImage *__cdecl lookup(unsigned long handle,void *user){(void)user;return handle==17?&image:0;}
static void fixture(BHPickupDrawContext *c,BHDrawQueue *q,BHMainCamera *cam,BHViewConfig *view,BHScreenClipContext *clip){
    memset(c,0,sizeof(*c));memset(cam,0,sizeof(*cam));memset(view,0,sizeof(*view));
    cam->projected.m[0][0]=100;cam->projected.m[1][1]=100;cam->projected.m[2][2]=1;cam->focal=100;
    view->near_z=1;view->far_z=100;view->left=0;view->right=100;view->top=0;view->bottom=100;
    bh_screen_clip_init(clip,1,100,0,100,0,100);
    c->queue=q;c->camera=cam;c->view=view;c->clip=clip;c->emit=capture;
    c->atlas_image=17;c->glow_image=19;c->display_depth=32;
}
static void producer(void){
    BHDrawQueue q;BHDrawCommand *p;unsigned long i;
    CHECK(bh_draw_queue_init(&q,15,0),"queue init");
    CHECK(bh_pickup_rect_queue(&q,8,17,12,13,14,15,16,7,.5f,.25f,.125f,.0625f,.03125f,18)==1,"rectangle producer append");
    p=q.commands+9;CHECK(p->kind==BH_DRAW_RECT && p->resource==17 && p->next==BH_DRAW_END && q.commands[8].next==9,"bucket 8 link and rect kind");
    for(i=0;i<5;++i)CHECK(closef(val(p->words+i),(float)(12+i)),"ordered rectangle geometry");
    CHECK(closef(val(p->words+5),.5f) && closef(val(p->words+6),.25f) && closef(val(p->words+7),.125f) &&
          closef(val(p->words+8),.0625f) && closef(val(p->words+9),.03125f) &&
          closef(val(p->words+10),18) && p->words[11]==7,"rectangle UV/depth/flag offsets");
    CHECK(bh_pickup_rect_queue(&q,99,18,1,2,3,4,5,0,1,0,0,1,1,9)==1 && q.commands[0].next==10,"out of range bucket uses original bucket zero");
    CHECK(bh_draw_queue_dispose(&q),"queue disposal");
}
static void projection_and_rect(void){
    BHDrawQueue q;BHPickupDrawContext c;BHMainCamera cam;BHViewConfig view;BHScreenClipContext clip;
    BHDrawCommand *cmd;unsigned long next=0;float x,y,z,s;
    CHECK(bh_draw_queue_init(&q,32,0),"rect fixture queue");fixture(&c,&q,&cam,&view,&clip);
    x=5;y=5;z=.5f;s=123;
    CHECK(!bh_pickup_project(&c,&x,&y,&z,&s) && closef(x,5) && closef(s,123),"projector near rejection keeps input");
    z=10;CHECK(bh_pickup_project(&c,&x,&y,&z,&s) && closef(x,50) && closef(y,50) && closef(z,10) && closef(s,10),"projected coordinates and reciprocal scale");
    CHECK(bh_pickup_rect_queue(&q,8,17,50,50,20,10,0,0,1,.25f,.125f,.125f,.0625f,10)==1,"rect enqueue for consumer");
    cmd=q.commands+9;emitted=0;
    CHECK(bh_pickup_rect_execute(&c,lookup,0,cmd,&next) && emitted==1 && next==BH_DRAW_END,"rect consumer emits and publishes next");
    CHECK(closef(corners[0][0],40) && closef(corners[0][1],45) && closef(corners[0][2],.25f) && closef(corners[0][3],.125f),"upper left and UV exact");
    CHECK(closef(corners[2][0],60) && closef(corners[2][1],55) && closef(corners[2][2],.375f) && closef(corners[2][3],.1875f),"lower right and UV exact");
    CHECK(color==0x80ffffffUL,"original signed grayscale color");
    CHECK(bh_pickup_rect_draw(&c,&image,50,50,20,10,0,7,.5f,.25f,.125f,.125f,.0625f,10) && emitted==2,
          "rectangle flags traverse the real quad and clip owner");
    CHECK(closef(corners[0][2],.75f) && closef(corners[0][3],.875f) &&
          closef(corners[1][2],.625f) && closef(corners[1][3],.875f),
          "flip U/V and rotated UV corner assignment");
    CHECK(color==0x807f7f7fUL,"half alpha truncates grayscale toward zero");
    cmd->resource=99;next=0;CHECK(!bh_pickup_rect_execute(&c,lookup,0,cmd,&next) && next==BH_DRAW_END && emitted==2,"missing image publishes next without geometry");
    CHECK(bh_pickup_rect_draw(&c,&image,-50,50,20,10,0,0,1,0,0,1,1,10) && emitted==2 && c.culled==1,"screen cull");
    CHECK(bh_draw_queue_dispose(&q),"rect queue disposal");
}
static void native_rectangle(void){
    BHDrawQueue queue;BHPickupDrawContext c;BHMainCamera cam;BHViewConfig view;BHScreenClipContext clip;
    BHTextureContext textures;BHTextureImage native_image;BHNativePrimitive native;
    memset(&textures,0,sizeof(textures));memset(&native_image,0,sizeof(native_image));memset(&native,0,sizeof(native));
    textures.allocator.allocate=allocation;textures.allocator.release=release;
    native.textures=&textures;native.near_z=1;
    CHECK(bh_draw_queue_init(&queue,16,0),"native rectangle queue");fixture(&c,&queue,&cam,&view,&clip);
    c.emit=bh_native_primitive_emit;c.emit_user=&native;
    CHECK(bh_pickup_rect_draw(&c,&native_image,50,50,20,10,0,0,1,.25f,.125f,.125f,.0625f,10) &&
          native.polygons==1 && native.vertices==4 && native_image.run_count==1 && native_image.vertex_count==4,
          "real native vertex queue consumes projected rectangle");
    CHECK(closef(native_image.vertices[0].sz,.1f) && closef(native_image.vertices[0].rhw,.1f) &&
          native_image.vertices[0].color==0x80ffffffUL && closef(native_image.vertices[0].tu,.25f),
          "native queue preserves depth, shade and atlas UV");
    textures.software=1;
    CHECK(!bh_pickup_rect_draw(&c,&native_image,50,50,20,10,0,0,1,0,0,1,1,10) &&
          textures.error==BH_TEXTURE_BACKEND && native_image.vertex_count==4,
          "software backend is explicitly rejected before queue publication");
    free(native_image.runs);free(native_image.vertices);bh_draw_queue_dispose(&queue);
}
static void labels(void){
    BHDrawQueue q;BHPickupDrawContext c;BHMainCamera cam;BHViewConfig view;BHScreenClipContext clip;BHEntity e;
    memset(&e,0,sizeof(e));e.position.x=5;e.position.y=5;e.position.z=10;
    CHECK(bh_draw_queue_init(&q,120,0),"label queue");fixture(&c,&q,&cam,&view,&clip);
    setl(&e,0x1c,19);
    CHECK(bh_pickup_label(&c,&e,123,2)==1 && q.count==15,"two layers include separator, digits and unit");
    CHECK(closef(val(q.commands[9].words+6),.08203125f) && closef(val(q.commands[9].words+7),.0703125f),"first layer separator UV");
    CHECK(closef(val(q.commands[10].words+6),(.5f+8.0f)*.0078125f) &&
          closef(val(q.commands[12].words+6),(.5f+24.0f)*.0078125f),"first layer digits 1..3 use atlas stride");
    CHECK(closef(val(q.commands[9].words+10),8) && closef(val(q.commands[9].words+5),.42f),"depth offset and fade");
    CHECK(bh_draw_queue_begin(&q),"reset queue");setl(&e,0x1c,21);
    CHECK(bh_pickup_label(&c,&e,999,8)==1 && q.count==11,"special type renders one icon per layer");
    CHECK(closef(val(q.commands[9].words+2),1.1875f) && closef(val(q.commands[10].words+2),.71875f),"special icon widths");
    CHECK(bh_draw_queue_begin(&q),"reset queue for type20");setl(&e,0x1c,20);
    CHECK(bh_pickup_label(&c,&e,0,1)==1 && q.count==14,"zero top number and type20 unit path");
    CHECK(bh_draw_queue_begin(&q),"reset queue for depth");e.position.z=.5f;
    CHECK(bh_pickup_label(&c,&e,5,1)==1 && q.count==9,"near projected labels produce no queue records");
    CHECK(bh_draw_queue_begin(&q),"reset queue for signed digits");e.position.z=10;setl(&e,0x1c,19);
    CHECK(bh_pickup_label(&c,&e,-12,-3)==1 && q.count==14,"signed quotients preserve two top digits and one lower digit");
    CHECK(closef(val(q.commands[10].words+6),-.05859375f) &&
          closef(val(q.commands[11].words+6),-.12109375f) &&
          closef(val(q.commands[12].words+6),-.18359375f),
          "negative decimal slots retain signed atlas UV instead of fabricating sign glyph");
    CHECK(bh_draw_queue_dispose(&q),"label queue disposal");
}
static void owner(void){
    BHDrawQueue q;BHPickupDrawContext c;BHMainCamera cam;BHViewConfig view;BHScreenClipContext clip;BHEntity e;float phase=.2f;
    CHECK(bh_draw_queue_init(&q,80,0),"owner queue");fixture(&c,&q,&cam,&view,&clip);
    memset(&e,0,sizeof(e));e.position.x=5;e.position.y=5;e.position.z=10;
    setf(&e,0x18,phase);setl(&e,0x1c,21);setl(&e,0x28,3);setl(&e,0x2c,2);setl(&e,0x30,71);
    CHECK(bh_pickup_draw(&c,&e)==1 && q.count==13,"default owner queues model glow and two text layers");
    CHECK(q.commands[9].kind==BH_DRAW_MODEL_RESOURCE && q.commands[9].resource==71 &&
          q.commands[10].kind==BH_DRAW_BILLBOARD && q.commands[11].kind==BH_DRAW_RECT,"default owner command sequence");
    CHECK(closef(val((const unsigned long *)(e.opaque0+0x24)),.4f),"draw writes double phase into pose");
    CHECK(bh_draw_queue_begin(&q),"reset owner queue");setl(&e,0x2c,0);
    CHECK(bh_pickup_draw(&c,&e)==1 && q.count==11,"zero token skips both numeric labels");
    CHECK(bh_draw_queue_dispose(&q),"owner queue disposal");
}
static void constructor_bridge(void){
    BHDrawQueue q;BHPickupDrawContext draw;BHEntityResourceDrawContext special;
    BHPickupRenderBinding binding;BHPickupContext owner;BHEntityPool pool;
    BHWorldGeometry world;BHPickupRoom room;BHMainCamera cam;BHViewConfig view;BHScreenClipContext clip;
    BHEntity *regular,*rare;BHVec3 position={5,5,5};unsigned long resources[10];BHEntityStep callback;
    memset(&owner,0,sizeof(owner));memset(&world,0,sizeof(world));memset(&room,0,sizeof(room));
    memset(&special,0,sizeof(special));memset(resources,0,sizeof(resources));
    world.navigation.next=(BHPortalNode*)&world;
    world.minimum.x=world.minimum.y=world.minimum.z=-10;
    world.maximum.x=world.maximum.y=world.maximum.z=10;
    room.world=&world;owner.first_world=&world;owner.world_count=1;
    owner.rooms=&room;owner.room_count=1;owner.resources=resources;owner.resource_count=10;
    resources[2]=71;resources[9]=72;
    CHECK(bh_entities_init(&pool,4,0),"constructor fixture pool");owner.pool=&pool;
    CHECK(bh_draw_queue_init(&q,80,0),"constructor fixture queue");fixture(&draw,&q,&cam,&view,&clip);
    special.queue=&q;special.glow_image=19;
    binding.default_draw=&draw;binding.special_draw=&special;
    CHECK(bh_pickup_render_bind(&owner,&binding) && owner.draw_user==&binding &&
          owner.default_draw==bh_pickup_default_step && owner.special_draw==bh_pickup_special_step,
          "typed render binding installs both source callback adapters");
    regular=bh_pickup_spawn(&owner,&position,2,0,3,3,7,0);
    CHECK(regular && bh_pickup_context(regular)==&owner,"constructor retains renderer owner");
    if(regular){memcpy(&callback,regular->payload+4,4);
        CHECK(callback==bh_pickup_default_step,"constructor stores default draw at payload +e8");
        if(callback)callback(regular);
        CHECK(q.count==16 && q.commands[9].kind==BH_DRAW_MODEL_RESOURCE &&
              q.commands[10].kind==BH_DRAW_BILLBOARD && q.commands[11].kind==BH_DRAW_RECT,
              "constructor-created pickup invokes real source model, glow and label producers");
    }
    CHECK(bh_draw_queue_begin(&q),"reset constructor queue");
    rare=bh_pickup_spawn(&owner,&position,9,0,1,1,1,0);
    CHECK(rare && bh_pickup_context(rare)==&owner,"special pickup constructor");
    if(rare){memcpy(&callback,rare->payload+4,4);
        CHECK(callback==bh_pickup_special_step,"constructor stores special draw at payload +e8");
        if(callback)callback(rare);
        CHECK(q.count==12 && q.commands[9].kind==BH_DRAW_MODEL_RESOURCE &&
              q.commands[10].kind==BH_DRAW_MODEL_RESOURCE && q.commands[11].kind==BH_DRAW_BILLBOARD,
              "special constructor callback invokes both model draws and glow");
    }
    bh_entities_dispose(&pool);bh_draw_queue_dispose(&q);
}
int bh_pickup_draw_checkpoint(unsigned long *count){checks=0;failures=0;producer();projection_and_rect();native_rectangle();labels();owner();constructor_bridge();if(count)*count=checks;
    printf("{\"scenario\":\"pickup-draw\",\"checks\":%lu,\"failures\":%d,\"original_code_required\":false}\n",checks,failures);return failures?1:0;}
