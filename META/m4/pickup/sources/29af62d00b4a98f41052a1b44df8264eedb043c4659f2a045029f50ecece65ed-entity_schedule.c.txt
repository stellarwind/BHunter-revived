static unsigned long checks,failures,draws;
static BHEntity *mutated_next,*mutated_pickup_next;
#define CHECK(x,s) do {++checks;if(!(x)){++failures;printf("FAIL entity-schedule: %s\n",s);}} while(0)
static float number(const unsigned long *word) {float value;memcpy(&value,word,4);return value;}
static void actor_link(BHEntity *e,BHEntity *next) {memcpy(e->payload+16,&next,sizeof(next));}
static void __cdecl draw(BHEntity *e) {++draws;(void)e;}
static void __cdecl draw_mutate(BHEntity *e) {++draws;actor_link(e,mutated_next);}
static void __cdecl draw_mutate_pickup(BHEntity *e) {++draws;bh_pickup_state(e)->room_next=mutated_pickup_next;}
static void method(BHEntity *e) {BHEntityStep p=draw;memcpy(e->payload+4,&p,sizeof(p));}
static void identity(BHMat34 *m) {memset(m,0,sizeof(*m));m->m[0][0]=m->m[1][1]=m->m[2][2]=1;}
static void setup(BHEntityScheduleContext *c,BHDrawQueue *q,BHViewConfig *v,BHViewCamera *camera) {
    memset(c,0,sizeof(*c));memset(v,0,sizeof(*v));memset(camera,0,sizeof(*camera));
    v->near_z=1;v->far_z=10;v->left=v->top=-1;v->right=v->bottom=1;
    identity(&camera->projected);
    c->queue=q;c->viewport=v;c->camera=camera;c->frame=42;
    c->max_rooms=8;c->max_entities=16;c->max_depth=4;
}
static void root_and_lists(void) {
    BHDrawQueue q;BHEntityScheduleContext c;BHViewConfig v;BHViewCamera camera;
    BHWorldGeometry w;BHPickupRoom room;BHEntityScheduleRoom schedule;
    BHEntity actors[2],pickup;
    memset(&w,0,sizeof(w));memset(&room,0,sizeof(room));memset(&schedule,0,sizeof(schedule));
    memset(actors,0,sizeof(actors));memset(&pickup,0,sizeof(pickup));
    CHECK(bh_draw_queue_init(&q,64,NULL),"root queue initialized");setup(&c,&q,&v,&camera);
    room.world=&w;room.actors=actors;room.pickups=&pickup;w.query_flags=4;
    schedule.room=&room;schedule.resource=37;method(actors);method(actors+1);method(&pickup);
    actor_link(actors,actors+1);draws=0;
    CHECK(bh_entity_schedule(&c,&schedule),"normal root schedules");
    CHECK(draws==3 && q.count==10 && q.commands[9].kind==BH_DRAW_MODEL_RESOURCE &&
          q.commands[9].resource==37,"root resource then actor chain and pickup draw");
    CHECK(schedule.frame_active && schedule.frame_remaining==30 && schedule.last_frame==42,
          "root installs finite frame callback");
    CHECK(bh_entity_schedule(&c,&schedule) && draws==3 && q.count==11,
          "same frame suppresses both root child lists while resource queues");
    c.frame++;CHECK(bh_entity_schedule(&c,&schedule) && draws==6,
                    "new frame redraws root children");
    {unsigned long i;for(i=0;i<29;++i)bh_entity_schedule_tick(&schedule);}
    CHECK(schedule.frame_active && schedule.frame_remaining==1,"tick retains callback before zero");
    bh_entity_schedule_tick(&schedule);
    CHECK(!schedule.frame_active && !schedule.frame_remaining,"thirtieth tick removes callback");
    CHECK(bh_draw_queue_dispose(&q),"root queue disposed");
}
static void linked_and_ghost(void) {
    BHDrawQueue q;BHEntityScheduleContext c;BHViewConfig v;BHViewCamera camera;
    BHWorldGeometry worlds[2];BHPickupRoom rooms[2];BHEntityScheduleRoom scheduled[2];
    BHEntity pickup;
    memset(worlds,0,sizeof(worlds));memset(rooms,0,sizeof(rooms));
    memset(scheduled,0,sizeof(scheduled));memset(&pickup,0,sizeof(pickup));
    CHECK(bh_draw_queue_init(&q,64,NULL),"linked queue initialized");setup(&c,&q,&v,&camera);
    rooms[0].world=worlds;rooms[1].world=worlds+1;rooms[1].pickups=&pickup;
    scheduled[0].room=rooms;scheduled[1].room=rooms+1;scheduled[0].next=scheduled+1;
    scheduled[1].linked_active=1;
    worlds[0].query_flags=4;worlds[1].query_flags=4|0x100;
    worlds[1].minimum.x=worlds[1].maximum.x=40;
    scheduled[0].resource=11;scheduled[1].resource=12;
    scheduled[0].resource_flags=scheduled[1].resource_flags=0x40;
    method(&pickup);draws=0;
    CHECK(bh_entity_schedule(&c,scheduled),"special root and linked room schedule");
    CHECK(q.count==23,"root dual draw; linked dual+default+ghost and eight state commands");
    CHECK(draws==1,"linked pickup list executes");
    CHECK(q.commands[9].kind==BH_DRAW_TEXTURE_MASK &&
          q.commands[10].kind==BH_DRAW_MODEL_RESOURCE &&
          q.commands[10].resource==11,"root state precedes first resource");
    CHECK(q.commands[21].kind==BH_DRAW_MODEL_RESOURCE && q.commands[21].resource==12,
          "linked special emits extra default resource");
    CHECK(q.commands[22].kind==BH_DRAW_MODEL_RESOURCE &&
          q.commands[22].resource==12,"ghost overlay queues linked room resource");
    CHECK(fabs(number(q.commands[22].words)+10.0f)<.0001f &&
          fabs(number(q.commands[22].words+6)-1.0f/3.0f)<.0001f,
          "ghost overlay uses camera blend and near fade");
    CHECK(bh_draw_queue_begin(&q),"linked queue reset for sentinel");
    scheduled[1].linked_active=0;scheduled[1].room=NULL;
    CHECK(bh_entity_schedule(&c,scheduled) && q.count==15 && draws==1,
          "inactive +1c8 sentinel stops before dereferencing room or drawing");
    CHECK(bh_draw_queue_dispose(&q),"linked queue disposed");
}
static void portal_and_clip(void) {
    BHDrawQueue q;BHEntityScheduleContext c;BHViewConfig v;BHViewCamera camera;
    BHWorldGeometry worlds[2];BHPickupRoom rooms[2];BHEntityScheduleRoom scheduled[2];
    BHPortal *portal;BHClipContext clip;BHClipVertex vertices[4],*out;
    BHVec3 square[4]={{-.5f,-.5f,2},{.5f,-.5f,2},{.5f,.5f,2},{-.5f,.5f,2}};
    unsigned long i;
    memset(worlds,0,sizeof(worlds));memset(rooms,0,sizeof(rooms));memset(scheduled,0,sizeof(scheduled));
    CHECK(bh_draw_queue_init(&q,64,NULL),"portal queue initialized");setup(&c,&q,&v,&camera);
    camera.position.z=3;
    for(i=0;i<2;++i){rooms[i].world=worlds+i;scheduled[i].room=rooms+i;}
    worlds[0].navigation.count=1;portal=worlds[0].navigation.portals;
    portal->direction=2;portal->plane=2;portal->overlap[0]=portal->overlap[2]=-.5f;
    portal->overlap[1]=portal->overlap[3]=.5f;
    portal->neighbor=&worlds[1].navigation;scheduled[0].portal_targets[0]=scheduled+1;
    CHECK(bh_entity_schedule(&c,scheduled),"visible portal recurses to bound target");
    CHECK(v.left==-1 && v.right==1 && v.top==-1 && v.bottom==1,
          "recursive portal restores source clip rectangle");
    CHECK(q.count==11 && q.commands[9].kind==BH_DRAW_CLIP_RECT &&
          q.commands[10].kind==BH_DRAW_CLIP_RECT,"both renderer buckets receive portal rectangle");
    CHECK(bh_draw_queue_begin(&q),"portal queue reset");camera.position.z=1;
    CHECK(bh_entity_schedule(&c,scheduled) && q.count==9,"back-facing portal culled");
    CHECK(bh_draw_queue_begin(&q),"portal queue reset again");camera.position.z=3;
    portal->plane=20;CHECK(bh_entity_schedule(&c,scheduled) && q.count==9,
                           "beyond far depth portal culled");
    CHECK(bh_draw_queue_begin(&q),"integer portal precision queue reset");
    portal->direction=4;portal->plane=16777217;portal->overlap[0]=-.5f;
    portal->overlap[1]=.5f;portal->overlap[2]=2;portal->overlap[3]=3;
    camera.position.x=16777216.0f;camera.projected.m[0][3]=-16777216.0f;
    CHECK(bh_entity_schedule(&c,scheduled) && q.count==11,
          "camera-side comparison retains original integer plane precision");
    CHECK(bh_clip_init(&clip,1,10,-1,1,-1,1),"portal clip context initialized");
    out=bh_entity_schedule_clip_portal(&clip,vertices,square);
    CHECK(out && clip.error==BH_CLIP_OK && clip.used==0,"inside four-vertex portal remains intact");
    for(i=0;i<4;++i)square[i].z=.5f;
    out=bh_entity_schedule_clip_portal(&clip,vertices,square);
    CHECK(!out && clip.error==BH_CLIP_OK,"near depth rejects fully behind portal");
    for(i=0;i<4;++i)square[i].z=2;
    square[0].x=square[3].x=-3;
    out=bh_entity_schedule_clip_portal(&clip,vertices,square);
    CHECK(out && clip.used>=2,"portal edge crossing inserts clipped pair");
    CHECK(bh_draw_queue_dispose(&q),"portal queue disposed");
}
static void mutation_and_zero_frame(void) {
    BHDrawQueue q;BHEntityScheduleContext c;BHViewConfig v;BHViewCamera camera;
    BHWorldGeometry w;BHPickupRoom room;BHEntityScheduleRoom schedule;
    BHEntity actors[3],pickups[3];BHEntityStep p=draw_mutate;
    memset(&w,0,sizeof(w));memset(&room,0,sizeof(room));
    memset(&schedule,0,sizeof(schedule));memset(actors,0,sizeof(actors));
    memset(pickups,0,sizeof(pickups));
    CHECK(bh_draw_queue_init(&q,32,NULL),"mutation queue initialized");
    setup(&c,&q,&v,&camera);c.frame=0;room.world=&w;room.actors=actors;
    room.pickups=pickups;w.query_flags=4;
    schedule.room=&room;method(actors);method(actors+1);method(actors+2);
    method(pickups);method(pickups+1);method(pickups+2);
    memcpy(actors[0].payload+4,&p,sizeof(p));actor_link(actors,actors+1);
    actor_link(actors+1,actors+2);
    p=draw_mutate_pickup;memcpy(pickups[0].payload+4,&p,sizeof(p));
    bh_pickup_state(pickups)->room_next=pickups+1;
    bh_pickup_state(pickups+1)->room_next=pickups+2;
    mutated_next=actors+2;mutated_pickup_next=pickups+2;draws=0;
    CHECK(bh_entity_schedule(&c,&schedule) && draws==4,
          "first frame zero draws, then observes post-callback successor mutation");
    CHECK(bh_pickup_state(pickups)->room_next==pickups+2,
          "pickup draw sees live post-callback +ec link");
    CHECK(schedule.frame_seen && schedule.last_frame==0,"zero frame marked seen");
    CHECK(bh_entity_schedule(&c,&schedule) && draws==4,"repeat zero frame suppresses children");
    CHECK(bh_draw_queue_dispose(&q),"mutation queue disposed");
}
static void clip_branches(void) {
    BHClipContext clip;BHClipVertex vertices[4],*out;
    BHVec3 square[4]={{-.5f,-.5f,2},{.5f,-.5f,2},{.5f,.5f,2},{-.5f,.5f,2}};
    BHVec3 control[4];unsigned long branch,i;
    CHECK(bh_clip_init(&clip,1,10,-1,1,-1,1),"six-plane portal clip initialized");
    for(branch=0;branch<6;++branch) {
        memcpy(control,square,sizeof(control));
        for(i=0;i<4;++i) {
            if(branch==0 && i<2)control[i].z=.5f;
            if(branch==1 && i>=2)control[i].z=11.0f;
            if(branch==2 && (i==0 || i==3))control[i].x=-3;
            if(branch==3 && (i==1 || i==2))control[i].x=3;
            if(branch==4 && i<2)control[i].y=-3;
            if(branch==5 && i>=2)control[i].y=3;
        }
        out=bh_entity_schedule_clip_portal(&clip,vertices,control);
        CHECK(out && clip.error==BH_CLIP_OK && clip.used>=2,
              "portal convex ring crosses one near/far/left/right/top/bottom plane");
    }
}
static void ghost_thresholds(void) {
    BHDrawQueue q;BHEntityScheduleContext c;BHViewConfig v;BHViewCamera camera;
    BHWorldGeometry worlds[2];BHPickupRoom rooms[2];BHEntityScheduleRoom schedule[2];
    const float distances[6]={29.0f,30.0f,60.0f,150.0f,180.0f,181.0f};
    unsigned long i;
    memset(worlds,0,sizeof(worlds));memset(rooms,0,sizeof(rooms));
    memset(schedule,0,sizeof(schedule));
    CHECK(bh_draw_queue_init(&q,32,NULL),"ghost threshold queue initialized");
    setup(&c,&q,&v,&camera);
    rooms[0].world=worlds;rooms[1].world=worlds+1;
    schedule[0].room=rooms;schedule[1].room=rooms+1;
    schedule[0].next=schedule+1;schedule[1].linked_active=1;
    worlds[1].query_flags=4|0x100;schedule[1].resource=91;
    for(i=0;i<6;++i) {
        CHECK(bh_draw_queue_begin(&q),"ghost threshold queue reset");
        worlds[1].minimum.x=worlds[1].maximum.x=distances[i];
        CHECK(bh_entity_schedule(&c,schedule) &&
              q.count==((i==2 || i==3)?11UL:10UL),
              "ghost strict near/far threshold and fade plateau");
        if(i==2 || i==3)
            CHECK(fabs(number(q.commands[10].words+6)-1.0f)<.0001f,
                  "ghost full alpha at exact 60 and 150 boundaries");
    }
    CHECK(bh_draw_queue_dispose(&q),"ghost threshold queue disposed");
}
int bh_entity_schedule_checkpoint(unsigned long *count) {
    root_and_lists();linked_and_ghost();portal_and_clip();mutation_and_zero_frame();
    clip_branches();ghost_thresholds();
    if(count)*count=checks;
    printf("{\"scenario\":\"entity-schedule\",\"checks\":%lu,\"failures\":%lu}\n",checks,failures);
    return failures!=0;
}

int main(void) {return bh_entity_schedule_checkpoint(NULL);}
#endif
