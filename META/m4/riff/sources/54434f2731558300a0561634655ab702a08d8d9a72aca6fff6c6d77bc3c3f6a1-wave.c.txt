static void file_tests(void) {
    BHWave wave;BHVFile *file;unsigned char prefix[65],*old;unsigned long i;Memory memory={0,0,0};BHSceneAllocator hooks={allocate,release,&memory};
    make_fixture();memcpy(prefix,"abc",3);memcpy(prefix+3,fixture,62);CHECK(write_file("bhwave_prefix.tmp",prefix,65),"write private prefixed RIFF");
    CHECK(bh_wave_init(&wave,&hooks) && bh_wave_decode(&wave,fixture,62),"initial owned data before I/O");
    file=bh_vfile_open("bhwave_prefix.tmp","rb");CHECK(file!=NULL,"open borrowed loose handle");
    if(file) {
        CHECK(!bh_vfile_seek(file,3,SEEK_SET) && bh_wave_read(&wave,file) && bh_vfile_tell(file)==65,"read remaining payload from nonzero cursor");
        old=wave.samples;
        CHECK(!bh_vfile_seek(file,2,SEEK_SET) && !bh_wave_read(&wave,file) && bh_vfile_tell(file)==2 && wave.samples==old,"malformed read restores cursor and old samples");
        for(i=1;i<=2;i++) {
            CHECK(!bh_vfile_seek(file,3,SEEK_SET),"restore valid start");memory.fail_at=memory.calls+i;
            CHECK(!bh_wave_read(&wave,file) && wave.error==BH_WAVE_MEMORY && bh_vfile_tell(file)==3 && wave.samples==old && memory.live==1,"each temporary/output allocation fails with rollback");memory.fail_at=0;
        }
        CHECK(bh_wave_read(&wave,file) && memory.live==1,"reuse borrowed handle after failures");
        CHECK(!bh_wave_read(&wave,file) && wave.error==BH_WAVE_TRUNCATED && bh_vfile_tell(file)==65,"empty remaining payload rejected");
        CHECK(!bh_vfile_close(file),"caller closes borrowed handle");
    }
    bh_wave_dispose(&wave);CHECK(!memory.live,"I/O allocations balanced");remove("bhwave_prefix.tmp");bh_vfile_shutdown();
}

static void real_assets(const char *archive) {
    FILE *raw;BHWave wave;BHVFile *file;unsigned char *bytes;unsigned long backend,i,at,n,decoded=0,total=0;const unsigned char *pcm;
    raw=fopen(archive,"rb");CHECK(raw!=NULL,"open independent raw archive");if(!raw)return;bh_wave_init(&wave,NULL);
    for(backend=0;backend<3;backend++) {

            const Manifest *m=manifest+i;
            bytes=(unsigned char*)malloc(m->size);CHECK(bytes!=NULL,"allocate raw RIFF control");if(!bytes)break;

            CHECK(pcm!=NULL && crc32(pcm,m->samples)==m->crc,"independent PCM fingerprint");
            if(backend==1)CHECK(bh_vfile_preload(m->name,"rb"),"preload real sound");
            if(backend==2)CHECK(write_file("bhwave_real.tmp",bytes,m->size),"write private loose sound");
            file=bh_vfile_open(backend==2?"bhwave_real.tmp":m->name,"rb");CHECK(file!=NULL,"open real sound asset");
            if(file) {
                CHECK(bh_wave_read(&wave,file),"bounded source decode real sound");
                if(wave.error==BH_WAVE_OK && pcm) {
                    CHECK(wave.sample_bytes==m->samples && wave.sample_rate==m->rate && wave.byte_rate==m->byte_rate && wave.channels==m->channels && wave.bits==m->bits && wave.block_align==m->align && wave.frames==m->samples/m->align,"all real PCM metadata matches independent struct controls");
                    CHECK(crc32(wave.samples,wave.sample_bytes)==m->crc && !memcmp(wave.samples,pcm,m->samples),"every sample byte matches raw archive and independent CRC");++decoded;total+=wave.sample_bytes;
                }
                CHECK(!bh_vfile_close(file),"caller closes real sound handle");

        }
        bh_vfile_shutdown();
    }
    CHECK(decoded==180 && total==5762313UL,"all 60 WAVs through three backends");
    printf("{\"scenario\":\"wave-real-assets\",\"files\":60,\"backend_decodes\":%lu,\"sample_bytes_compared\":%lu,\"playback_objects_created\":false}\n",decoded,total);
