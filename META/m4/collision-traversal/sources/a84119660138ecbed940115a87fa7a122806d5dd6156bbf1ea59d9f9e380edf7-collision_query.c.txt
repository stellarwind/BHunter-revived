static unsigned long checks;
static int failed;
#define CHECK(x,label) do {++checks;if(!(x)){if(++failed<8)printf("FAIL collision traversal %d: %s\n",__LINE__,label);}} while(0)
static void section(unsigned char *data,unsigned long *at,unsigned long count,const void *p,unsigned long size) {
    memcpy(data+*at,&count,4);*at+=4;
    memcpy(data+*at,p,count*size);*at+=count*size;
}
static unsigned long fixture(unsigned char *data,int destination) {
    BHBspMaterial material;
    BHBspVertex vertices[4];BHBspEdge edges[4];BHBspFace faces[3];
    BHBspNode nodes[3];BHBspLeaf leaves[3];
    unsigned long at=0,i;
    memset(&material,0,sizeof(material));strcpy(material.texture,"STONE.PCX");
    memset(vertices,0,sizeof(vertices));memset(edges,0,sizeof(edges));
    memset(faces,0,sizeof(faces));memset(nodes,0,sizeof(nodes));
    for(i=0;i<4;++i) {
        vertices[i].y=(i==0 || i==3) ? -2 : 2;
        if(destination) vertices[i].z=i<2 ? -1 : 1;
        else vertices[i].x=i<2 ? -1 : 1;
        edges[i].vertex=i;edges[i].next=i==3 ? -1 : i+1;
    }
    for(i=0;i<3;++i) {
        faces[i].first_edge=0;
        nodes[i].minimum[0]=destination ? 0 : -1;
        nodes[i].maximum[0]=destination ? 0 : 1;
        nodes[i].minimum[1]=-2;nodes[i].maximum[1]=2;
        nodes[i].minimum[2]=destination ? -1 : 0;
        nodes[i].maximum[2]=destination ? 1 : 0;
        nodes[i].normal[destination ? 0 : 2]=1;
        nodes[i].front_leaf=i;nodes[i].back_leaf=-1;
        nodes[i].front_node=nodes[i].back_node=-1;
        leaves[i].face=i;leaves[i].next=-1;
    }
    section(data,&at,1,&material,sizeof(material));
    section(data,&at,4,vertices,sizeof(vertices[0]));
    section(data,&at,4,edges,sizeof(edges[0]));
    section(data,&at,3,faces,sizeof(faces[0]));
    section(data,&at,3,nodes,sizeof(nodes[0]));
    section(data,&at,3,leaves,sizeof(leaves[0]));
    return at;
}
static void identity(BHMat34 *m) {memset(m,0,sizeof(*m));m->m[0][0]=m->m[1][1]=m->m[2][2]=1;}
static void cases(void) {
    unsigned char bytes[2048];unsigned long size,i;
    BHCollisionModel a,b;BHCollisionQuery q;
    BHMat34 relative,world;
    BHCollisionHit hit,saved;
    BHBspNode *an,*bn; BHBspLeaf *al,*bl;
    identity(&relative);identity(&world);
    CHECK(bh_collision_model_init(&a,NULL) && bh_collision_model_init(&b,NULL),"real model owners");
    size=fixture(bytes,0);CHECK(bh_collision_model_decode(&a,bytes,size),"decode source geometry");
    size=fixture(bytes,1);CHECK(bh_collision_model_decode(&b,bytes,size),"decode destination geometry");
    an=(BHBspNode *)a.bsp.nodes;bn=(BHBspNode *)b.bsp.nodes;
    al=(BHBspLeaf *)a.bsp.leaves;bl=(BHBspLeaf *)b.bsp.leaves;
    CHECK(bh_collision_query_init(&q,&a.geometry,&b.geometry),"real owned query workspace");
    CHECK(bh_collision_query_run(&q,0,0,&relative,&world,&hit)==1 &&
          hit.a_face==0 && hit.b_face==0 && hit.normal.x==1 && hit.normal.y==0 &&
          hit.contact.raw_point.x==0 && hit.contact.raw_point.y==0 && hit.contact.raw_point.z==0,
          "selected face pair and normal from real decoded models");
    saved=hit;
    CHECK(bh_geometry_set_flags(&a.geometry,0,1) &&
          bh_collision_query_run(&q,0,0,&relative,&world,&hit)==0 && !memcmp(&hit,&saved,sizeof(hit)),
          "source disabled flag filters all face work");
    bh_geometry_set_flags(&a.geometry,0,0);
    CHECK(bh_geometry_set_flags(&b.geometry,0,1) &&
          bh_collision_query_run(&q,0,0,&relative,&world,&hit)==0,"destination disabled face");
    bh_geometry_set_flags(&b.geometry,0,0);
    bn[0].front_leaf=-1;bn[0].back_leaf=1;
    CHECK(bh_collision_query_node(&q,0,0,&relative,&world,&hit)==1 &&
          hit.b_face==1 && hit.normal.x==-1 && hit.contact.point.x<0,"back list reverses normal");
    bn[0].front_leaf=0;
    CHECK(bh_collision_query_node(&q,0,0,&relative,&world,&hit)==1 && hit.b_face==0,
          "front face wins before back face");
    bn[0].back_leaf=-1;bn[0].front_leaf=-1;bn[0].front_node=1;bn[0].back_node=2;
    CHECK(bh_collision_query_node(&q,0,0,&relative,&world,&hit)==1 && hit.b_node==1,
          "destination front subtree precedes back subtree");
    bh_geometry_set_flags(&b.geometry,1,1);
    CHECK(bh_collision_query_node(&q,0,0,&relative,&world,&hit)==1 && hit.b_node==2,
          "destination back subtree follows miss");
    bh_geometry_set_flags(&b.geometry,1,0);
    bn[0].front_leaf=0;bn[0].front_node=bn[0].back_node=-1;
    an[0].front_leaf=-1;an[0].front_node=1;an[0].back_node=2;
    CHECK(bh_collision_query_run(&q,0,0,&relative,&world,&hit)==1 && hit.a_node==1,
          "source front subtree first");
    bh_geometry_set_flags(&a.geometry,1,1);
    CHECK(bh_collision_query_run(&q,0,0,&relative,&world,&hit)==1 && hit.a_node==2,
          "source back subtree after disabled front face");
    bh_geometry_set_flags(&a.geometry,1,0);
    /* Parent's source face misses its plane test; descendant uses that face
       against a different destination subtree. Ancestor marking must suppress it. */
    an[0].front_leaf=0;an[0].plane=100;al[1].face=0;
    CHECK(bh_collision_query_run(&q,0,0,&relative,&world,&hit)==1 && hit.a_node==2,
          "ancestor source face suppressed in child and sibling remains available");
    for(i=0;i<3;++i) CHECK(!q.suppressed[i],"successful unwind restores suppression counts");
    an[0].plane=0;an[0].front_node=an[0].back_node=-1;al[1].face=1;
    /* 90 degree world rotation maps +X to +Y; no in-place transform alias. */
    world.m[0][0]=world.m[1][1]=0;world.m[0][1]=-1;world.m[1][0]=1;
    CHECK(bh_collision_query_run(&q,0,0,&relative,&world,&hit)==1 &&
          hit.normal.x==0 && hit.normal.y==1 && hit.contact.point.y>0,
          "world rotation transforms normal from unmodified input vector");
    identity(&world);
    /* a-to-b swaps x and y with a sign, retaining a spanning polygon. */
    relative.m[0][0]=relative.m[1][1]=0;relative.m[0][1]=-1;relative.m[1][0]=1;
    CHECK(bh_collision_query_run(&q,0,0,&relative,&world,&hit)==1 && hit.contact.raw_point.y==0,
          "rotated source polygon retains all original coordinate components");
    identity(&relative);
    saved=hit;relative.m[0][3]=100;
    CHECK(bh_collision_query_run(&q,0,0,&relative,&world,&hit)==0 && !memcmp(&hit,&saved,sizeof(hit)),
          "separated node spheres prune before narrow phase");
    identity(&relative);an[0].plane=100;
    CHECK(bh_collision_query_node(&q,0,0,&relative,&world,&hit)==0,"source plane versus destination radius prune");
    an[0].plane=0;
    bn[0].front_leaf=99;
    CHECK(bh_collision_query_run(&q,0,0,&relative,&world,&hit)==-1 && !q.busy &&
          !memcmp(&hit,&saved,sizeof(hit)),"foreign leaf rejects without output mutation");
    bn[0].front_leaf=0;bl[0].next=0;bh_geometry_set_flags(&b.geometry,0,1);
    CHECK(bh_collision_query_run(&q,0,0,&relative,&world,&hit)==-1 && !q.busy,
          "cyclic leaf list bounded and reset");
    bl[0].next=-1;bh_geometry_set_flags(&b.geometry,0,0);
    an[0].front_node=0;an[0].front_leaf=-1;
    CHECK(bh_collision_query_run(&q,0,0,&relative,&world,&hit)==-1 && !q.busy,
          "source child cycle bounded by stack capacity");
    an[0].front_node=-1;an[0].front_leaf=0;
    CHECK(bh_collision_query_run(&q,0,0,&relative,&world,&hit)==1,"query reusable after malformed topology");
    CHECK(bh_collision_query_node(&q,-1,0,&relative,&world,&hit)==0,"negative root sentinel misses");
    CHECK(bh_collision_query_node(&q,99,0,&relative,&world,&hit)==-1,"foreign node rejects");
    CHECK(bh_collision_query_dispose(&q),"query workspace released");
    CHECK(bh_collision_model_dispose(&a) && bh_collision_model_dispose(&b),"borrowers released before model owners");
}
int bh_collision_query_checkpoint(unsigned long *count) {
    checks=0;failed=0;cases();*count=checks;
    printf("{\"scenario\":\"collision-traversal\",\"checks\":%lu,\"failures\":%d,\"original_code_required\":false}\n",checks,failed);
    return failed!=0;
}
