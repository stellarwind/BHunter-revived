#include "bh_scene_view.h"
#include "bh_blocks.h"
#include "bh_geometry.h"
#include "bh_rotation.h"
#include "bh_fp.h"
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
/* Source-only static checkpoint. Deliberate diagnostic colors/camera; no
   script/entity construction, material fidelity or gameplay admission. */
int main(int argc,char **argv) {
    BHBlockLibrary library;BHSceneLevel level;BHGeometry *geometry=NULL;unsigned char *shell=NULL;
    BHSceneView view;BHMathEnvironment environment;BHVFile *file=NULL;
    BHVec3 *polygon=NULL,local,world;BHFaceInfo face;BHMat34 transform;
    BHVec3 camera={-60,-17,60};unsigned long i,j,k,initialized=0,faces=0,max_vertices=0,shells=0;
    long model;int ok=0,fp=0;float angle;unsigned long color;
    if(argc!=3){printf("usage: bhunter-scene-snapshot archive output.bmp\n");return 2;}
    memset(&view,0,sizeof(view));bh_blocks_init(&library,NULL);bh_scene_level_init(&level,NULL);
    fp=bh_math_environment_enter(&environment);if(!fp)goto done;
    if(!bh_vfile_mount(argv[1]) || !bh_blocks_load(&library,"BSP\\BLOCKS.TXT","BSP\\",BH_BSP_PREFIX_ONLY))goto done;
    file=bh_vfile_open("TEST.LEV","rb");if(!file || !bh_scene_level_read(&level,file))goto done;
    bh_vfile_close(file);file=NULL;
    geometry=(BHGeometry *)calloc(library.count,sizeof(*geometry));if(!geometry)goto done;
    shell=(unsigned char *)calloc(library.count,1);if(!shell)goto done;
    for(i=0;i<library.count;++i) {
        if(!bh_geometry_init(geometry+i,&library.blocks[i].bsp,NULL))goto done;++initialized;
        memset(&face,0,sizeof(face));
        for(j=0;j<library.blocks[i].bsp.counts[3];++j) {
            if(!bh_geometry_face(geometry+i,j,&face))goto done;
            if(!_stricmp(face.texture,"PORTAL.GIF"))shell[i]=1;
            if(max_vertices<face.vertices)max_vertices=face.vertices;
        }
    }
    if(max_vertices>1024 || max_vertices<3)goto done;
    polygon=(BHVec3 *)malloc(max_vertices*sizeof(*polygon));if(!polygon)goto done;
    if(!bh_scene_view_init(&view,640,480) || !bh_scene_view_begin(&view))goto done;
    for(i=0;i<level.placement_count;++i) {
        BHScenePlacement *placement=level.placements+i;
        model=bh_blocks_find(&library,placement->name);if(model<0)goto done;
        /* This first view draws navigation shell models only. Non-shell
           placements include invisible sensor/script proxy meshes as well as
           visible props; their actual draw ownership remains unrecovered. */
        if(!shell[model])continue;++shells;
        memset(&transform,0,sizeof(transform));transform.m[0][0]=transform.m[1][1]=transform.m[2][2]=1;
        bh_mat34_translate_local((float)placement->x,(float)placement->y,(float)placement->z,&transform);
        angle=(float)((placement->flags&3UL)*1.5707963267948966);bh_mat34_rotate_y(angle,&transform);
        memset(&face,0,sizeof(face));
        for(j=0;j<library.blocks[model].bsp.counts[3];++j) {
            if(!bh_geometry_face(geometry+model,j,&face))goto done;
            /* A portal aperture is a connection, not a visible solid wall. */
            if(!_stricmp(face.texture,"PORTAL.GIF") || face.vertices<3)continue;
            for(k=0;k<face.vertices;++k) {
                if(!bh_geometry_vertex(geometry+model,j,k,&local))goto done;
                bh_mat34_transform_point(&transform,&local,&world);
                polygon[k].x=camera.x-world.x;polygon[k].y=world.y-camera.y;polygon[k].z=camera.z-world.z;
            }
            /* Material-index colors make this explicitly an untextured
               geometry diagnostic rather than invented original lighting. */
            color=library.blocks[model].bsp.faces[j].material;
            color=0x303030UL|(((color*53+37)&127)<<16)|(((color*97+71)&127)<<8)|((color*29+19)&127);
            if(!bh_scene_view_polygon(&view,polygon,face.vertices,color))goto done;++faces;
        }
    }
    if(!bh_scene_view_end(&view,argv[2]) || !view.triangles || !view.pixels)goto done;
    ok=1;
done:
    printf("{\"scenario\":\"source-scene-snapshot\",\"passed\":%s,\"models\":%lu,\"placements\":%lu,\"shell_placements\":%lu,\"faces\":%lu,\"triangles\":%lu,\"nonblack_pixels\":%lu,\"hresult\":%ld,\"unapplied_scripts\":%lu,\"camera\":[-60,-17,60],\"forward\":[0,0,-1],\"textured\":false,\"original_code_required\":false}\n",
        ok?"true":"false",library.count,level.placement_count,shells,faces,view.triangles,view.pixels,(long)view.error,library.unapplied_scripts);
    bh_scene_view_dispose(&view);free(polygon);free(shell);
    for(i=0;i<initialized;++i)bh_geometry_dispose(geometry+i);free(geometry);
    if(file)bh_vfile_close(file);bh_scene_level_dispose(&level);bh_blocks_dispose(&library);bh_vfile_shutdown();
    if(fp)bh_math_environment_leave(&environment);return ok?0:1;
}
