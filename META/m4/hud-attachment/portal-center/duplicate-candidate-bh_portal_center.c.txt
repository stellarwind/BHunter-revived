#include "bh_portal_center.h"
#include "bh_round.h"
#include <float.h>
#include <stddef.h>
typedef char center_portal_stride[sizeof(BHPortal)==60?1:-1];
typedef char center_scratch_offset[offsetof(BHPortal,scratch)==40?1:-1];
int __cdecl bh_portal_scratch_center(const BHPortalNode *node,unsigned long slot,BHVec3 *out) {
    const BHPortal *p;double raw_u,raw_v;float u,plane;
    if(!node||slot>=6||!out)return 0;
    p=node->portals+slot;
    if(!_finite(p->scratch[0])||!_finite(p->scratch[1])||!_finite(p->scratch[2])||!_finite(p->scratch[3]))return 0;
    raw_u=(double)p->scratch[1]+p->scratch[0];
    raw_v=((double)p->scratch[3]+p->scratch[2])*.5;
    bh_round_float(&u,raw_u*.5);bh_round_float(&plane,(double)p->plane);
    switch(p->direction) {
    case 0:case 4:out->x=plane;out->y=u;bh_round_float(&out->z,raw_v);return 1;
    case 1:case 5:out->x=u;out->y=plane;bh_round_float(&out->z,raw_v);return 1;
    case 2:case 6:out->x=u;out->z=plane;bh_round_float(&out->y,raw_v);return 1;
    default:return -1;
    }
}
