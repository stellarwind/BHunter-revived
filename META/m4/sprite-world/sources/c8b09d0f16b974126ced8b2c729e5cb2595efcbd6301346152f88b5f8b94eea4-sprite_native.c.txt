static unsigned long checks;static int failures;
#define CHECK(x,s) check((x)!=0,s,__LINE__)
static void check(int ok,const char *s,int line){++checks;if(!ok){++failures;if(failures<20)printf("FAIL sprite native %d: %s\n",line,s);}}
static unsigned long pixel(const char *name,unsigned long x,unsigned long y){
 FILE *f=fopen(name,"rb");BITMAPFILEHEADER h;BITMAPINFOHEADER info;unsigned char rgb[3];unsigned long stride;
 if(!f)return 0xffffffffUL;
 if(fread(&h,sizeof(h),1,f)!=1 || fread(&info,sizeof(info),1,f)!=1 || h.bfType!=0x4d42 || info.biBitCount!=24){fclose(f);return 0xffffffffUL;}
 stride=(info.biWidth*3+3)&~3UL;
 if(fseek(f,h.bfOffBits+(info.biHeight-1-y)*stride+x*3,SEEK_SET) || fread(rgb,1,3,f)!=3){fclose(f);return 0xffffffffUL;}
 fclose(f);return ((unsigned long)rgb[2]<<16)|((unsigned long)rgb[1]<<8)|rgb[0];
}
int bh_sprite_native_checkpoint(unsigned long *count){
 BHMathEnvironment fp;BHSceneView view;BHTextureContext textures;BHTextureImage *image=NULL;BHRenderMaterial material;BHRenderState state;BHNativePrimitive native;
 BHMainCamera camera;BHScreenClipContext screen;BHAttributeClipContext attribute;BHSpriteBillboardContext billboard;BHSpriteOrientedContext oriented;BHSpriteOrientedVertex path[8];
 BHDrawResource resource,lod;BHDrawResourceContext draw;BHPcx pcx;unsigned char pixels[1024];unsigned long i,mode,expected,queued;int texture_live=0;
 const char *names[]={"sprite-native-billboard.bmp","sprite-native-half.bmp","sprite-native-ribbon.bmp","sprite-native-lod-blank.bmp"};
 checks=0;failures=0;if(!bh_math_environment_enter(&fp))return 2;
 memset(&view,0,sizeof(view));memset(&textures,0,sizeof(textures));memset(&material,0,sizeof(material));memset(&camera,0,sizeof(camera));memset(&native,0,sizeof(native));
 CHECK(bh_scene_view_init(&view,64,64),"real offscreen native scene");if(!view.device)goto done;
 CHECK(bh_texture_context_init(&textures,view.draw,view.device,NULL),"real texture lifetime");if(!textures.device)goto done;texture_live=1;
 textures.slot_caps_override=DDSCAPS_TEXTURE|DDSCAPS_SYSTEMMEMORY;
 CHECK(bh_texture_formats(&textures),"real native texture formats");
 memset(&pcx,0,sizeof(pcx));memset(pixels,1,sizeof(pixels));pcx.width=pcx.height=32;pcx.pixel_bytes=1024;pcx.pixels=pixels;pcx.palette[3]=pcx.palette[4]=pcx.palette[5]=255;
 image=bh_texture_stage(&textures,"SPRITEWHITE.PCX",&pcx,0,0);CHECK(image!=NULL,"real white texture staging");if(!image)goto done;image->flags=1;
 CHECK(!bh_render_material_initialize(&material,&textures,view.d3d) && !material.failures && bh_render_state_init(&state,&textures),"real material and state owners");
 CHECK(bh_view_config_init(&view.projection,64,64) && bh_main_camera_settings(&view.projection,60,1,100) && bh_main_camera_pose(&view.projection,&camera,0,0,0,0,0,0),"recovered main camera");view.original_depth=1;
 CHECK(bh_screen_clip_init(&screen,1,100,0,64,0,64) && bh_attribute_clip_init(&attribute,1,100,0,64,0,64),"real screen/attribute clippers");
 native.textures=&textures;native.near_z=1;
 memset(&billboard,0,sizeof(billboard));billboard.camera=&camera;billboard.config=&view.projection;billboard.clip=&screen;billboard.emit=bh_native_primitive_emit;billboard.emit_user=&native;billboard.base_color=0xff000000UL;
 memset(&oriented,0,sizeof(oriented));memset(path,0,sizeof(path));oriented.path=path;oriented.capacity=8;oriented.near_z=1;oriented.projection_scale=camera.focal;oriented.clip=&attribute;oriented.color=0xff000000UL;oriented.emit=bh_native_attribute_emit;oriented.emit_user=&native;
 memset(&draw,0,sizeof(draw));draw.camera=&camera;draw.config=&view.projection;draw.billboard=&billboard;draw.oriented=&oriented;
 for(mode=0;mode<4;++mode){
  memset(&resource,0,sizeof(resource));resource.tag=BH_DRAW_BILLBOARD;resource.data.billboard.image=image;resource.data.billboard.width=resource.data.billboard.height=2;
  if(mode==2){memset(&resource,0,sizeof(resource));resource.tag=BH_DRAW_RIBBON;resource.data.ribbon.image=image;resource.data.ribbon.local_end.x=2;resource.data.ribbon.start_width=resource.data.ribbon.end_width=.5f;}
  memset(&lod,0,sizeof(lod));lod.tag=BH_DRAW_LOD;lod.data.lod.radius=1;lod.data.lod.count=1;lod.data.lod.thresholds[0]=3;lod.data.lod.children[0]=&resource;
  CHECK(bh_scene_view_begin(&view),"begin real native source composition");
  CHECK(bh_draw_resource(&draw,mode==3?&lod:&resource,mode==2?-1:0,0,4,0,0,0,mode==1?.5f:1),"recursive resource owner reaches actual sprite geometry and native emitter");
  queued=image->vertex_count;expected=mode==3?0:mode==1?0x7f7f7fUL:0xffffffUL;
  CHECK(mode==3?!queued:queued>=4,"real native queues are populated or LOD-rejected");
  for(i=0;i<queued;++i){CHECK(image->vertices[i].sz==.25f && image->vertices[i].rhw==.25f,"native queue preserves original near/z and reciprocal z");CHECK((image->vertices[i].color&0xffffffUL)==expected,"native queue color agrees with original signed grayscale conversion");}
  CHECK(bh_render_mode(&state,0) && !image->vertex_count && !image->run_count,"real texture queue flush/residency and DrawPrimitive");
  CHECK(bh_scene_view_end(&view,names[mode]),"read back actual native pixels");
  {unsigned long actual=pixel(names[mode],32,32);
   if(mode==1)CHECK((actual&255)>=122 && (actual&255)<=132 && ((actual>>8)&255)>=122 && ((actual>>8)&255)<=132 && ((actual>>16)&255)>=122 && ((actual>>16)&255)<=132,"half modulation center agrees within one native RGB565 quantization step");
   else CHECK(actual==expected,"independent center pixel white or LOD blank");}
  CHECK(pixel(names[mode],2,2)==0,"independent exterior pixel stays blank");
  CHECK(mode==3?!view.pixels:view.pixels>50,"visible source geometry reaches real rasterization");
 }
 /* Per-vertex modulation is independently checked in the actual queue. */
 {BHClipVertex v[3];float scalar[3]={0,.5f,1};unsigned long before=image->vertex_count;
  memset(v,0,sizeof(v));for(i=0;i<3;++i){v[i].point.z=4;v[i].sx=(float)(i*4+8);v[i].sy=(float)(i==1?16:8);v[i].next=v+(i+1)%3;v[i].previous=v+(i+2)%3;memcpy(&v[i].unknown28,scalar+i,4);}
  CHECK(bh_native_attribute_emit(v,image,0xff000000UL,&native),"attribute native adapter reaches real queue with varying modulation");
  CHECK(image->vertices[before].color==0xff000000UL && image->vertices[before+1].color==0xff7f7f7fUL && image->vertices[before+2].color==0xffffffffUL,"each original scalar produces its own queue color");
  before=image->vertex_count;textures.software=1;CHECK(!bh_native_attribute_emit(v,image,0,&native) && image->vertex_count==before,"unfinished software backend explicitly rejected without queue publication");textures.software=0;
 }
done:
 bh_render_material_dispose(&material);if(texture_live)bh_texture_context_dispose(&textures);bh_scene_view_dispose(&view);
 CHECK(bh_math_environment_supported(),"native composition retains PC53 profile");bh_math_environment_leave(&fp);*count=checks;
 printf("{\"scenario\":\"sprite-native\",\"checks\":%lu,\"failures\":%d,\"original_code_required\":false}\n",checks,failures);return failures!=0;
}
