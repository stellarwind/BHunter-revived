static unsigned long checks;static int failures;
#define CHECK(x,s) check((x)!=0,s,__LINE__)
static void check(int ok,const char *s,int line){++checks;if(!ok){++failures;if(failures<20)printf("FAIL resource draw %d: %s\n",line,s);}}
typedef struct Fixture {
    BHDrawResourceContext draw;BHMainCamera camera;BHViewConfig config;
    BHSpriteBillboardContext billboard;BHScreenClipContext clip;
    BHSpriteOrientedContext oriented;BHAttributeClipContext attribute_clip;
    BHSpriteOrientedVertex path[8];
    BHTextureImage image;unsigned long calls,color;float scalar;int reject;
} Fixture;
static int __cdecl capture(BHClipVertex *ring,BHTextureImage *image,unsigned long color,void *user){
    Fixture *f=(Fixture *)user;
    CHECK(ring && ring->next && image==&f->image,"real billboard image and ring");
    memcpy(&f->scalar,&ring->unknown28,4);
    ++f->calls;f->color=color;return !f->reject;
}
static void setup(Fixture *f){memset(f,0,sizeof(*f));
    f->camera.projected.m[0][0]=f->camera.projected.m[1][1]=f->camera.projected.m[2][2]=1;
    f->camera.planes[0].normal.z=1;f->camera.focal=100;
    f->config.near_z=1;f->config.far_z=100;
    f->config.left=0;f->config.right=200;f->config.top=0;f->config.bottom=200;
    bh_screen_clip_init(&f->clip,1,100,0,200,0,200);
    f->billboard.camera=&f->camera;f->billboard.config=&f->config;
    f->billboard.clip=&f->clip;f->billboard.emit=capture;
    f->billboard.emit_user=f;f->billboard.base_color=0xff000000UL;
    bh_attribute_clip_init(&f->attribute_clip,1,100,-10,10,-10,10);
    f->oriented.path=f->path;f->oriented.capacity=8;f->oriented.near_z=1;
    f->oriented.projection_scale=2;f->oriented.clip=&f->attribute_clip;
    f->oriented.emit=capture;f->oriented.emit_user=f;f->oriented.color=0xff000000UL;
    f->draw.camera=&f->camera;f->draw.config=&f->config;f->draw.billboard=&f->billboard;
    f->draw.oriented=&f->oriented;
}
static void routing(void){Fixture f;BHDrawResource leaf,lod;setup(&f);
    memset(&leaf,0,sizeof(leaf));leaf.tag=BH_DRAW_BILLBOARD;
    leaf.data.billboard.image=&f.image;leaf.data.billboard.width=4;
    leaf.data.billboard.height=2;
    memset(&lod,0,sizeof(lod));lod.tag=BH_DRAW_LOD;
    lod.data.lod.radius=2;lod.data.lod.count=2;
    lod.data.lod.thresholds[0]=10;lod.data.lod.thresholds[1]=50;
    lod.data.lod.children[1]=&leaf;
    CHECK(bh_draw_resource(&f.draw,&lod,1000,1000,20,0,0,0,.5f),"LOD routes second interval through real billboard owner");
    CHECK(f.calls==1 && f.draw.dispatched==1 && f.draw.culled==0 && f.color==0xff7f7f7fUL,
        "selected child carries pose and final scalar");
    CHECK(bh_draw_resource(&f.draw,&lod,1000,1000,51,0,0,0,1),"past last threshold is a no-draw success");
    CHECK(f.calls==1 && f.draw.dispatched==0 && f.draw.culled==1,"no child beyond LOD range");
    CHECK(!bh_draw_resource(&f.draw,&lod,1000,1000,-1,0,0,0,1) &&
        f.draw.error==BH_DRAW_RESOURCE_TOPOLOGY,"near tangent selects first child");
    CHECK(bh_draw_resource(&f.draw,&lod,1000,1000,-1.01,0,0,0,1) && f.draw.culled==1,
        "outside expanded near plane is culled");
    lod.data.lod.children[0]=&leaf;
    CHECK(bh_draw_resource(&f.draw,&lod,1000,1000,10,0,0,0,1),"threshold equality selects first child");
    CHECK(f.calls==2,"threshold equality emitted exactly one child");
    CHECK(bh_draw_resource(&f.draw,&lod,1000,1000,-1.0-1.0/67108864.0,0,0,0,1) &&
        f.draw.dispatched==0 && f.draw.culled==1,
        "near plane compares retained double even when float depth rounds to tangent");
    lod.data.lod.thresholds[1]=150;
    CHECK(bh_draw_resource(&f.draw,&lod,1000,1000,102.0+1.0/67108864.0,0,0,0,1) &&
        f.draw.dispatched==1,
        "far plane compares stored float when double depth exceeds expanded far plane");
    f.reject=1;
    CHECK(!bh_draw_resource(&f.draw,&leaf,1000,1000,10,0,0,0,1) &&
        f.draw.error==BH_DRAW_RESOURCE_BILLBOARD,"primitive failure propagates through dispatch");
}
static void guards(void){Fixture f;BHDrawResource lod,ribbon;BHSpriteOrientedContext oriented;BHSpriteOrientedVertex path[1];setup(&f);
    memset(&lod,0,sizeof(lod));lod.tag=BH_DRAW_LOD;lod.data.lod.radius=1;
    lod.data.lod.count=6;
    CHECK(!bh_draw_resource(&f.draw,&lod,0,0,10,0,0,0,1) &&
        f.draw.error==BH_DRAW_RESOURCE_ARGUMENT,"original fixed five threshold slots bounded");
    lod.data.lod.count=1;lod.data.lod.thresholds[0]=20;lod.data.lod.children[0]=&lod;
    CHECK(!bh_draw_resource(&f.draw,&lod,0,0,10,0,0,0,1) &&
        f.draw.error==BH_DRAW_RESOURCE_TOPOLOGY,"recursive LOD cycle bounded");
    memset(&ribbon,0,sizeof(ribbon));ribbon.tag=BH_DRAW_RIBBON;
    ribbon.data.ribbon.image=&f.image;
    memset(&oriented,0,sizeof(oriented));oriented.path=path;oriented.capacity=1;
    f.draw.oriented=&oriented;
    CHECK(!bh_draw_resource(&f.draw,&ribbon,0,0,10,0,0,0,1) &&
        f.draw.error==BH_DRAW_RESOURCE_CAPACITY && oriented.used==0,
        "two-point ribbon requires real path capacity before mutation");
    lod.tag=(BHDrawResourceTag)4;
    CHECK(bh_draw_resource(&f.draw,&lod,0,0,10,0,0,0,1) &&
        f.draw.error==BH_DRAW_RESOURCE_OK && !f.draw.dispatched && !f.draw.culled && !f.calls,
        "unknown positive tag falls through without draw");
    lod.tag=(BHDrawResourceTag)-1;
    CHECK(bh_draw_resource(&f.draw,&lod,0,0,10,0,0,0,1) &&
        f.draw.error==BH_DRAW_RESOURCE_OK && !f.draw.dispatched && !f.draw.culled && !f.calls,
        "negative tag bit pattern follows unsigned default case");
}
static void ribbon(void){Fixture f;BHDrawResource resource;setup(&f);memset(&resource,0,sizeof(resource));
    resource.tag=BH_DRAW_RIBBON;resource.data.ribbon.image=&f.image;
    resource.data.ribbon.local_end.x=2;resource.data.ribbon.start_width=1;
    resource.data.ribbon.end_width=1;
    CHECK(bh_draw_resource(&f.draw,&resource,0,0,2,0,0,0,.25f),
        "resource appends path and invokes real oriented renderer");
    CHECK(f.calls==1 && f.oriented.polygons_emitted==1 && f.oriented.used==0 &&
        f.draw.dispatched==1,"single source-built ribbon polygon drains path");
    CHECK(f.scalar==.25f && f.oriented.image==&f.image &&
        f.path[0].texture_start==.25f && f.path[1].texture_start==.25f,
        "image and modulation carried through two vertices and emit ring");
    CHECK(f.path[0].next==f.path+1 && f.path[1].previous==f.path &&
        f.path[0].texture_end==0 && f.path[1].texture_end==1,
        "original two-point chain and longitudinal indices");
}
int bh_draw_resource_checkpoint(unsigned long *count){BHMathEnvironment fp;
    checks=0;failures=0;if(!bh_math_environment_enter(&fp))return 2;
    routing();guards();ribbon();CHECK(bh_math_environment_supported(),"caller PC53 profile retained");
    bh_math_environment_leave(&fp);*count=checks;
    printf("{\"scenario\":\"draw-resource\",\"checks\":%lu,\"failures\":%d,\"original_code_required\":false}\n",checks,failures);
    return failures!=0;
}
#ifdef BH_DRAW_RESOURCE_STANDALONE
int main(void){unsigned long count;return bh_draw_resource_checkpoint(&count);}
#endif
