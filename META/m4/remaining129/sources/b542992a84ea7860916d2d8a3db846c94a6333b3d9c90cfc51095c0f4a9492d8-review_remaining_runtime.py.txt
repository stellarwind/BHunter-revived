"""One-library bounded full-body attribution, with known-free positive control.

Only actual COFF relocation operands are masked. No source recovery claim.
Do not infer ownership from an address or a decompiler-assigned name.
"""
import hashlib,json,struct,sys
from pathlib import Path
ROOT=Path(__file__).resolve().parents[1]
sys.path.insert(0,str(ROOT/'analysis/astra-review'))
sys.path.insert(0,str(ROOT/'tools'))
from targeted_probes import parse_coff
from record_pickup_basic import reference,read,write

def main():
    ledger,raw=reference();catalog=read(ROOT/'META/m1/role-catalog.json')
    library=Path('D:/Decomp/MSCVC/MSVC5/SP3_EXTRACTED/VS97_SP3/all/vc/lib/libc.lib')
    lib=library.read_bytes();assert lib[:8]==b'!<arch>\n'
    candidates=[r for r in catalog['records'] if r['role']=='unknown' and int(r['address'],16)>=0x47db50]
    candidates.append(dict(address='0047db90',body_ranges=['0047db90-0047dbdf'],positive_control=True))
    hits={r['address']:[] for r in candidates};pos=8;members=0;longnames=b'';covered=[]
    previous=read(ROOT/'analysis/crt-comparison.json')[0]['matches']
    while pos+60<=len(lib):
        head=lib[pos:pos+60];size=int(head[48:58].strip(b'\0 '));obj=lib[pos+60:pos+60+size];pos+=60+size+(size&1)
        member=head[:16].decode().strip()
        if member=='//':longnames=obj
        elif member.startswith('/') and member[1:].isdigit():member=longnames[int(member[1:]):].split(b'\0')[0].split(b'\n')[0].decode().rstrip('/')
        if obj[:2]!=b'\x4c\x01':continue
        members+=1;symbols,sections=parse_coff(obj)
        for prior in previous:
            if prior['member']!=member:continue
            section=sections[prior['section']+1];code=section['data'];original=raw(int(prior['address'],16),len(code));mask=bytearray(b'\xff'*len(code));relocations=[]
            for rel in section['relocations']:
                off=rel['offset'];assert rel['kind'] in (6,20) and off+4<=len(code)
                mask[off:off+4]=b'\0'*4
                operand=struct.unpack_from('<I',original,off)[0];at=int(prior['address'],16)+off
                resolved=(at+4+operand)&0xffffffff if rel['kind']==20 else operand
                relocations.append(dict(address=f'{at:08x}',symbol=rel['symbol']['name'],original_target=f'{resolved:08x}'))
            assert len(code)==prior['length'] and all((a&m)==(b&m) for a,b,m in zip(code,original,mask))
            covered.append(dict(start=prior['address'],bytes=len(code),member=member,object_sha256=hashlib.sha256(obj).hexdigest(),original_hex=original.hex(),library_hex=code.hex(),mask_hex=mask.hex(),relocations=relocations,symbols=prior['symbols']))
        for sym in symbols.values():
            if sym['section']<=0 or sym['storage'] not in (2,3) or sym['name'].startswith('.') :continue
            sh=20+(sym['section']-1)*40
            if not (struct.unpack_from('<I',obj,sh+36)[0]&0x20):continue
            section=sections[sym['section']];start=sym['value'];code=section['data']
            for target in candidates:
                entry=int(target['address'],16);ranges=[];ok=True;unmasked=0;relocations=[]
                for span in target['body_ranges']:
                    lo,hi=[int(v,16) for v in span.split('-')];hi+=1;off=start+lo-entry
                    if off<0 or off+hi-lo>len(code):ok=False;break
                    original=raw(lo,hi-lo);body=code[off:off+len(original)];mask=bytearray(b'\xff'*len(body))
                    for rel in section['relocations']:
                        at=rel['offset']-off
                        if at>=len(body) or at+4<=0:continue
                        if rel['kind'] not in (6,20) or at<0 or at+4>len(body):ok=False;break
                        mask[at:at+4]=b'\0'*4
                        operand=struct.unpack_from('<I',original,at)[0]
                        resolved=(lo+at+4+operand)&0xffffffff if rel['kind']==20 else operand
                        relocations.append(dict(address=f'{lo+at:08x}',kind=rel['kind'],symbol=rel['symbol']['name'],original_target=f'{resolved:08x}'))
                    if not ok or any((a&m)!=(b&m) for a,b,m in zip(original,body,mask)):ok=False;break
                    unmasked+=sum(bool(m) for m in mask)
                    ranges.append(dict(start=f'{lo:08x}',end_exclusive=f'{hi:08x}',original_hex=original.hex(),library_hex=body.hex(),mask_hex=mask.hex(),sha256=hashlib.sha256(original).hexdigest()))
                if ok and unmasked>=2:
                    # Corrupt an unmasked byte; this same matcher must reject it.
                    control=next(r for r in ranges if 'ff' in r['mask_hex']);m=bytes.fromhex(control['mask_hex']);b=bytearray.fromhex(control['original_hex']);i=next(i for i,v in enumerate(m) if v);b[i]^=1
                    assert any((a&v)!=(z&v) for a,z,v in zip(b,bytes.fromhex(control['library_hex']),m))
                    hits[target['address']].append(dict(symbol=sym['name'],member_header=head[:16].decode().strip(),object_sha256=hashlib.sha256(obj).hexdigest(),section=sym['section'],offset=start,ranges=ranges,relocations=relocations,unmasked_bytes=unmasked,negative_control_rejected=True))
    assert any(h['symbol']=='_free' for h in hits['0047db90']),'positive control failed'
    coverage={}
    for target in candidates:
        owners=[]
        for span in target['body_ranges']:
            lo,hi=[int(v,16) for v in span.split('-')];hi+=1
            found=next((s for s in covered if int(s['start'],16)<=lo and hi<=int(s['start'],16)+s['bytes']),None)
            if found is None:break
            owners.append(found['start'])
        else:coverage[target['address']]=owners
    report=dict(passed=True,section_coverage=coverage,verified_sections=covered,library=str(library),library_sha256=hashlib.sha256(lib).hexdigest(),members=members,positive_control='0047db90 _free',comparison='Every observed body range, masking only actual COFF DIR32/REL32 operands; no byte-match/source recovery claim',matches={a:h for a,h in hits.items() if h},unmatched=[a for a,h in hits.items() if not h and a not in coverage])
    write(ROOT/'META/m4/remaining129/runtime-review.json',report)
    print(json.dumps(dict(verified_candidates=len(report['matches'])-1,unmatched=len(report['unmatched']),positive_control=True)))
if __name__=='__main__':main()
