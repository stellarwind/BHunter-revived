"""Audit the six source owners and fifty runtime/import exclusions separately."""
import hashlib,json,sys
from pathlib import Path
from record_pickup_basic import ROOT,reference,read,write,sha,active,native_dependencies
sys.path.insert(0,str(ROOT/'analysis/astra-review'))
from targeted_probes import parse_coff
HERE=ROOT/'META/m4/remaining129'

def verify_runtime(raw):
    report=read(HERE/'runtime-review.json');library=Path(report['library']);body=library.read_bytes()
    assert sha(library)==report['library_sha256'];objects={};pos=8
    while pos+60<=len(body):
        size=int(body[pos+48:pos+58].strip(b'\0 '));obj=body[pos+60:pos+60+size];pos+=60+size+(size&1)
        if obj[:2]==b'\x4c\x01':objects[hashlib.sha256(obj).hexdigest()]=parse_coff(obj)
    def mask_for(section,off,n):
        mask=bytearray(b'\xff'*n)
        for rel in section['relocations']:
            at=rel['offset']-off
            if at>=n or at+4<=0:continue
            assert rel['kind'] in (6,20) and at>=0 and at+4<=n
            mask[at:at+4]=b'\0'*4
        return bytes(mask)
    for address,hits in report['matches'].items():
        for hit in hits:
            syms,sections=objects[hit['object_sha256']];section=sections[hit['section']]
            assert any(s['name']==hit['symbol'] and s['section']==hit['section'] and s['value']==hit['offset'] for s in syms.values())
            for span in hit['ranges']:
                lo=int(span['start'],16);n=int(span['end_exclusive'],16)-lo;off=hit['offset']+lo-int(address,16)
                assert section['data'][off:off+n]==bytes.fromhex(span['library_hex'])
                mask=mask_for(section,off,n);original=raw(lo,n)
                assert original.hex()==span['original_hex'] and mask.hex()==span['mask_hex']
                assert all((a&m)==(b&m) for a,b,m in zip(original,section['data'][off:off+n],mask))
                assert hashlib.sha256(original).hexdigest()==span['sha256']
    for item in report['verified_sections']:
        syms,sections=objects[item['object_sha256']];code=bytes.fromhex(item['library_hex']);mask=bytes.fromhex(item['mask_hex'])
        assert any(s['data']==code and mask_for(s,0,len(code))==mask for s in sections.values())
        original=raw(int(item['start'],16),len(code));assert original.hex()==item['original_hex']
        assert all((a&m)==(b&m) for a,b,m in zip(original,code,mask))
    classification=read(HERE/'runtime-classification.json');catalog={r['address']:r for r in read(ROOT/'META/m1/role-catalog.json')['records']}
    assert len(classification['entries'])==50
    for entry in classification['entries']:
        assert catalog[entry['address']]['role']==entry['role'] and entry['proof']==catalog[entry['address']]['behavior']
    assert raw(0x4882d0,6)==bytes.fromhex('ff25f0904800')
    return len(classification['entries'])

def main():
    ledger,raw=reference();real=read(ROOT/'build/m0/result.json');clean_report=read(HERE/'clean606/clean-checkout.json');clean=clean_report['result']
    assert real['passed'] and clean['passed'] and clean_report['exit_code']==0
    assert not clean_report['original_assets_present'] and not clean_report['compiler_in_checkout']
    assert real['sources']==clean['sources'];native_dependencies(real,clean)
    current=active(ledger);assert len(current)==len({r['original_va'] for r in current})==606
    for row in current:assert row['source_sha256']==real['sources'][row['source']]==sha(ROOT/row['source']),row['id']
    added=[r for r in current if r['id'].endswith('-remaining129')];assert len(added)==6
    for row in added:
        boundary=row['validation']['boundary'];body=raw(int(boundary['start'],16),boundary['bytes'])
        assert hashlib.sha256(body).hexdigest()==boundary['sha256']
        assert len(body)==int(boundary['end_exclusive'],16)-int(boundary['start'],16)
        assert not row['validation']['raw_byte_match'] and not row['validation']['original_code_executed']
    for name in ['force-feedback-update-spec.json','remote-render-spec.json','source-files-spec.json','remote-query-spec.json']:
        for item in read(HERE/name).get('constants',[]):assert raw(int(item['address'],16),len(bytes.fromhex(item['hex'])))==bytes.fromhex(item['hex'])
    scenarios={'impact-ring':99,'force-feedback-update':51,'remote-render':777,'source-files':71,'remote-query':51}
    for build in (real,clean):
        assert not build['original_code_required'] and build['test']['failures']==0
        assert len([c for c in build['commands'] if c['label'].startswith('compile-')])==550
        controls=[v for k,v in build.items() if k.startswith('missing_')];assert len(controls)==269 and all('LNK2001' in v for v in controls)
        for name,n in scenarios.items():
            found=[s for s in build['scenarios'] if s.get('scenario')==name];assert len(found)==1 and found[0]['checks']==n and found[0]['failures']==0
    excluded=verify_runtime(raw)
    catalog=read(ROOT/'META/m1/role-catalog.json')['records'];addresses={r['original_va'] for r in current}
    remaining=[r for r in catalog if r['role'] in ('game','engine_candidate','unknown') and r['address'] not in addresses]
    assert len(remaining)==73
    report=dict(passed=True,admitted_original_owners=606,new_whole_source_owners=6,runtime_or_import_entries_excluded=excluded,remaining_catalog_entries=73,translation_units=550,missing_module_controls=269,real_checks=real['test']['checks'],clean_checks=clean['test']['checks'],compiled_source_inputs_identical=True,reference_files_unchanged=True,game_launches=0,original_code_executed=False,request_complete=False)
    write(HERE/'final-audit.json',report);print(json.dumps(report))
if __name__=='__main__':main()
