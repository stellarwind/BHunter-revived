#include "bh_collision_model.h"
#include <string.h>

int __cdecl bh_collision_model_init(BHCollisionModel *model, const BHSceneAllocator *allocator) {
    BHCollisionModel next;
    if (!model)
        return 0;
    memset(&next, 0, sizeof(next));
    if (!bh_bsp_init(&next.bsp, allocator))
        return 0;
    *model = next;
    return 1;
}
int __cdecl bh_collision_model_dispose(BHCollisionModel *model) {
    if (!model || model->geometry.enumerating)
        return 0;
    if (model->geometry.bsp)
        bh_geometry_dispose(&model->geometry);
    bh_bsp_dispose(&model->bsp);
    return 1;
}
static int prepare(BHCollisionModel *model) {
    BHBspFace *faces = (BHBspFace *)model->bsp.faces;
    unsigned long i;
    if (!bh_geometry_init(&model->geometry, &model->bsp, &model->bsp.allocator)) {
        model->bsp.error = BH_BSP_MEMORY;
        return 0;
    }
    /* These bytes belong to the newly decoded owner, never to caller input. */
    for (i = 0; i < model->bsp.counts[3]; ++i)
        faces[i].runtime = 0;
    return 1;
}
static int commit(BHCollisionModel *model, BHCollisionModel *next, int ok) {
    BHBspError error;
    if (ok)
        ok = prepare(next);
    if (!ok) {
        error = next->bsp.error;
        bh_collision_model_dispose(next);
        model->bsp.error = error;
        return 0;
    }
    bh_collision_model_dispose(model);
    *model = *next;
    model->geometry.bsp = &model->bsp;
    return 1;
}
int __cdecl bh_collision_model_decode(BHCollisionModel *model, const void *bytes,
                                      unsigned long size) {
    BHCollisionModel next;
    int ok;
    if (!model || model->geometry.enumerating ||
        !bh_collision_model_init(&next, &model->bsp.allocator))
        return 0;
    ok = bh_bsp_decode(&next.bsp, bytes, size);
    return commit(model, &next, ok);
}
int __cdecl bh_collision_model_load(BHCollisionModel *model, const char *name, const char *prefix,
                                    BHBspSearch search) {
    BHCollisionModel next;
    int ok;
    if (!model || model->geometry.enumerating ||
        !bh_collision_model_init(&next, &model->bsp.allocator))
        return 0;
    ok = bh_bsp_load(&next.bsp, name, prefix, search);
    return commit(model, &next, ok);
}

/* Face-pair narrow phase, 0044cc20. Explicit inputs replace the original
   shared query globals; decoded BSP owners supply the actual vertex/edge data. */
#include "bh_fp.h"
#include "bh_round.h"
#include <float.h>

static double face_dot(const BHVec3 *n, const BHVec3 *p) {
    return ((double)n->x * p->x + (double)n->y * p->y) + (double)n->z * p->z;
}
static int face_vertex(const BHBsp *b, long edge, const BHMat34 *transform, BHVec3 *p) {
    long v;
    if (edge < 0 || (unsigned long)edge >= b->counts[2]) return 0;
    v = b->edges[edge].vertex;
    if (v < 0 || (unsigned long)v >= b->counts[1]) return 0;
    p->x = b->vertices[v].x; p->y = b->vertices[v].y; p->z = b->vertices[v].z;
    if (!_finite(p->x) || !_finite(p->y) || !_finite(p->z)) return 0;
    if (transform) bh_mat34_transform_point(transform, p, p);
    return _finite(p->x) && _finite(p->y) && _finite(p->z);
}
static double cut_distance(const BHVec3 *n, const BHVec3 *p, float plane, int first) {
    /* The first plane evaluation adds x+y, then z; the second uses x+y+z. */
    if (first) return ((double)n->z*p->z +
        ((double)n->x*p->x + (double)n->y*p->y)) + plane;
    return face_dot(n,p) + plane;
}
static int face_cut(const BHBsp *b, long face, const BHMat34 *transform,
                    const BHVec3 *n, float plane, int first, BHVec3 result[2]) {
    BHVec3 start, previous, current;
    long edge, next;
    unsigned long steps = 0, found = 0, axis;
    float previous_distance;
    double d, inverse;
    if (!b || !b->faces || !b->edges || !b->vertices || face < 0 ||
        (unsigned long)face >= b->counts[3]) return -1;
    edge = b->faces[face].first_edge;
    if (!face_vertex(b, edge, transform, &start)) return -1;
    previous = start;
    bh_round_float(&previous_distance, cut_distance(n,&previous,plane,first));
    do {
        if (++steps > b->counts[2]) return -1;
        next = b->edges[edge].next;
        if (next < 0) current = start;
        else if (!face_vertex(b, next, transform, &current)) return -1;
        d = cut_distance(n,&current,plane,first);
        if (d * previous_distance < 0) {
            inverse = 1.0 / (d - previous_distance);
            for (axis=0; axis<3; ++axis)
                bh_round_float(((float *)&result[found])+axis,
                    (d*((float *)&previous)[axis] -
                     (double)previous_distance*((float *)&current)[axis])*inverse);
            if (++found == 2) return 1;
        }
        previous = current;
        bh_round_float(&previous_distance,d);
        edge = next;
    } while (edge >= 0);
    return 0;
}
int __cdecl bh_collision_face_contact(const BHBsp *a, long a_face,
    const BHMat34 *a_to_b, const BHBsp *b, long b_face, long b_node,
    const BHMat34 *b_to_world, const BHVec3 *world_normal, BHFaceContact *out) {
    BHVec3 first[2], second[2], direction, side, midpoint, plane_normal;
    const BHVec3 *low, *high;
    BHFaceContact hit;
    float side_plane, offset, extent, x, y;
    unsigned long i;
    int result;
    if (!a || !b || !a_to_b || !b_to_world || !world_normal || !out ||
        !b->nodes || b_node < 0 || (unsigned long)b_node >= b->counts[4] ||
        !bh_math_environment_supported()) return -1;
    for (i=0;i<12;++i)
        if (!_finite(((const float *)a_to_b)[i]) ||
            !_finite(((const float *)b_to_world)[i])) return -1;
    for (i=0;i<3;++i) {
        ((float *)&plane_normal)[i]=b->nodes[b_node].normal[i];
        if (!_finite(((const float *)&plane_normal)[i]) ||
            !_finite(((const float *)world_normal)[i])) return -1;
    }
    if (!_finite(b->nodes[b_node].plane)) return -1;
    result=face_cut(a,a_face,a_to_b,&plane_normal,b->nodes[b_node].plane,1,first);
    if (result != 1) return result;
    for (i=0;i<3;++i) bh_round_float(((float *)&direction)+i,
        (double)((float *)&first[1])[i]-((float *)&first[0])[i]);
    bh_vec3_cross(&direction,&plane_normal,&side);
    bh_round_float(&side_plane,-face_dot(&side,&first[0]));
    bh_round_float(&offset,-face_dot(&direction,&first[0]));
    bh_round_float(&extent,face_dot(&direction,&first[1])+offset);
    result=face_cut(b,b_face,NULL,&side,side_plane,0,second);
    if (result != 1) return result;
    bh_round_float(&x,face_dot(&direction,&second[0])+offset);
    bh_round_float(&y,face_dot(&direction,&second[1])+offset);
    if (y <= x) {
        if (x <= 0 || extent <= y) return 0;
        high = extent <= x ? &first[1] : &second[0];
        low = y <= 0 ? &first[0] : &second[1];
    } else {
        if (y <= 0 || extent <= x) return 0;
        high = extent <= y ? &first[1] : &second[1];
        low = x <= 0 ? &first[0] : &second[0];
    }
    for(i=0;i<3;++i) bh_round_float(((float *)&midpoint)+i,
        ((double)((const float *)low)[i]+((const float *)high)[i])*.5);
    bh_mat34_transform_point(b_to_world,&midpoint,&hit.raw_point);
    for(i=0;i<3;++i) {
        bh_round_float(((float *)&hit.point)+i,
            (double)((float *)&hit.raw_point)[i]-((const float *)world_normal)[i]*-0.05);
        if (!_finite(((float *)&hit.point)[i])) return -1;
    }
    *out=hit;
    return 1;
}
