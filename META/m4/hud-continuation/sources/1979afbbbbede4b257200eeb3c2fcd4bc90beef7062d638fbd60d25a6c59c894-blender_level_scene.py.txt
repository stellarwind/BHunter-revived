"""Run inside Blender: author and verify a packed static BHunter level."""
import bpy
import json
import math
from pathlib import Path
import sys
from mathutils import Quaternion, Vector

out = Path(sys.argv[sys.argv.index('--')+1])
data = json.loads((out/'level-data.json').read_text())
bpy.ops.object.select_all(action='SELECT')
bpy.ops.object.delete(use_global=False)
for mesh in list(bpy.data.meshes):
    if mesh.users == 0: bpy.data.meshes.remove(mesh)
scene = bpy.context.scene
world = bpy.data.collections.new('01 World geometry')
props = bpy.data.collections.new('02 Props and scripted placements')
helpers = bpy.data.collections.new('03 Portals and proxy geometry (hidden)')
presentation = bpy.data.collections.new('04 Cameras and lights')
for collection in [world, props, helpers, presentation]: scene.collection.children.link(collection)
helpers.hide_render = True
helpers.hide_viewport = True
images, materials = {}, {}
for key, item in data['textures'].items():
    image = bpy.data.images.load(str(out/item['file']), check_existing=True)
    image.name = key
    image.pack()
    image.filepath = '//'+item['file']
    images[key] = image


def material(item):
    key = (item['texture'] or '__HELPER__'+item['name'].upper(), tuple(item['attributes']))
    if key in materials: return materials[key]
    mat = bpy.data.materials.new(item['name'] or 'Unnamed helper')
    mat.use_nodes = True
    mat['BHunter_material_bytes'] = item['attributes']
    nodes = mat.node_tree.nodes; nodes.clear()
    result = nodes.new('ShaderNodeOutputMaterial')
    shader = nodes.new('ShaderNodeBsdfPrincipled')
    mat.node_tree.links.new(shader.outputs['BSDF'], result.inputs['Surface'])
    shader.inputs['Roughness'].default_value = 1
    if item['texture']:
        tex = nodes.new('ShaderNodeTexImage'); tex.image = images[item['texture']]
        tex.interpolation = 'Closest'; tex.extension = 'REPEAT'
        mat.node_tree.links.new(tex.outputs['Color'], shader.inputs['Base Color'])
        mat.node_tree.links.new(tex.outputs['Color'], shader.inputs['Emission Color'])
        shader.inputs['Emission Strength'].default_value = .65
        mat.node_tree.links.new(tex.outputs['Alpha'], shader.inputs['Alpha'])
        mat.surface_render_method = 'DITHERED'
        mat.diffuse_color = (.6, .6, .6, 1)
    else:
        shader.inputs['Base Color'].default_value = (1, .1, .5, 1)
        mat.diffuse_color = (1, .1, .5, 1)
    materials[key] = mat
    return mat


models = {}
for source in data['models']:
    parts = []
    for helper in [False, True]:
        selected = [i for i, mat in enumerate(source['face_materials'])
                    if (source['materials'][mat]['texture'] is None) == helper]
        if not selected: parts.append(None); continue
        mesh = bpy.data.meshes.new(source['name']+(' helpers' if helper else ''))
        mesh.from_pydata([(x, z, -y) for x, y, z in source['vertices']], [], [source['faces'][i] for i in selected])
        mesh.update()
        for item in source['materials']: mesh.materials.append(material(item))
        uv = mesh.uv_layers.new(name='Original BSP UV')
        original_faces = mesh.attributes.new(name='BHunter_face_index', type='INT', domain='FACE')
        for polygon, original in zip(mesh.polygons, selected):
            polygon.material_index = source['face_materials'][original]
            original_faces.data[polygon.index].value = source['face_ids'][original]
            for loop, point in zip(polygon.loop_indices, source['uvs'][original]): uv.data[loop].uv = point
        mesh['BHunter_BSP'] = source['name']
        mesh['BHunter_original_vertex_count'] = len(source['vertices'])
        mesh['BHunter_portal_shell'] = source['shell']
        assert len(mesh.uv_layers[0].data) == len(mesh.loops)
        parts.append(mesh)
    models[source['name'].lower()] = (parts, source['shell'])

counts = {'world_objects': 0, 'prop_objects': 0, 'helper_objects': 0}
for placement in data['placements']:
    parts, shell = models[placement['name'].lower()]
    x, y, z = placement['position']
    for helper, mesh in enumerate(parts):
        if mesh is None: continue
        obj = bpy.data.objects.new('%04d %s%s'%(placement['index'], placement['name'], ' [helper]' if helper else ''), mesh)
        collection = helpers if helper else world if shell else props
        collection.objects.link(obj)
        obj.location = (x, z, -y)
        obj.rotation_euler.z = -(placement['flags']&3)*math.pi/2
        obj['BHunter_placement_index'] = placement['index']
        obj['BHunter_flags'] = placement['flags']
        obj['BHunter_game_position'] = placement['position']
        obj['BHunter_source_model'] = placement['name']
        counts['helper_objects' if helper else 'world_objects' if shell else 'prop_objects'] += 1

scene.unit_settings.system = 'NONE'
scene['BHunter_source_level'] = 'TEST.LEV'
scene['BHunter_archive_sha256'] = data['archive_sha256']
scene['BHunter_coordinate_conversion'] = data['coordinate_conversion']
scene['BHunter_scope'] = 'Static BSP/LEV data. Scripts are metadata; gameplay is not reconstructed by this export.'
text = bpy.data.texts.new('README — BHunter level')
readme = '''BHUNTER — TEST.LEV static level export

Open in Material Preview to see the original textures. All textures are packed
inside this blend and also supplied in the textures folder.

01 World geometry: portal-bearing room/corridor meshes.
02 Props and scripted placements: static raw placement meshes.
03 Portals and proxy geometry: hidden by default; enable to inspect helpers.
Meshes are shared between placements to preserve instancing and keep editing fast.
Object names retain placement IDs and BSP names; custom properties retain flags.
Original per-corner UVs and source face IDs are retained.

Coordinates: Blender(x,y,z)=game(x,z,-y), original unit scale unchanged.
No scripts are executed and no original game code is loaded. Dynamic objects,
game lighting, visibility and mission behavior are not recreated here.
'''
text.write(readme)
(out/'README.txt').write_text(readme)

# Set a practical overview and two cameras without altering the imported data.
visible = list(world.objects)+list(props.objects)
bpy.context.view_layer.update()
points = [obj.matrix_world @ Vector(corner) for obj in world.objects for corner in obj.bound_box]
minimum = Vector(tuple(min(p[i] for p in points) for i in range(3)))
maximum = Vector(tuple(max(p[i] for p in points) for i in range(3)))
center = (minimum+maximum)/2
extent = max(maximum-minimum)
for name, location, target, ortho in [
    ('Overview camera', center+Vector((extent*.6, -extent*.8, extent*.85)), center, True),
    ('Starting area camera', Vector((-970, 1460, 60)), Vector((-920, 1460, 60)), False),
]:
    cam = bpy.data.cameras.new(name); obj = bpy.data.objects.new(name, cam); presentation.objects.link(obj)
    obj.location = location; obj.rotation_euler = (target-location).to_track_quat('-Z', 'Y').to_euler()
    cam.clip_start = .05; cam.clip_end = 100000
    if ortho: cam.type = 'ORTHO'; cam.ortho_scale = extent*1.5; scene.camera = obj
    else: cam.lens = 32
sun = bpy.data.lights.new('Inspection sun', 'SUN'); sun.energy = 2
sun_obj = bpy.data.objects.new('Inspection sun', sun); presentation.objects.link(sun_obj)
sun_obj.rotation_euler = (.3, -.4, -.5)
scene.world.color = (.15, .15, .15)
scene.render.engine = 'BLENDER_EEVEE'
scene.render.resolution_x = 1400; scene.render.resolution_y = 1000; scene.render.resolution_percentage = 100
for screen in bpy.data.screens:
    for area in screen.areas:
        if area.type == 'VIEW_3D':
            space = area.spaces.active
            space.clip_end = 100000
            space.shading.type = 'MATERIAL'
            space.region_3d.view_location = center
            space.region_3d.view_distance = extent*1.1
            space.region_3d.view_rotation = Quaternion((.8205, .4247, .1759, .3399))
            space.overlay.show_floor = False
scene.render.filepath = str(out/'level-overview.png')
bpy.context.preferences.filepaths.save_version = 0
bpy.ops.wm.save_as_mainfile(filepath=str(out/'BHunter_TEST.blend'))
# Save a small inspectable preview without invoking the game or its renderer.
scene.render.resolution_percentage = 60
bpy.ops.render.render(write_still=True)
report = dict(passed=True, meshes=len(bpy.data.meshes), objects=len(visible),
              packed_textures=sum(bool(image.packed_file) for image in images.values()),
              materials=len(materials), bounds=[list(minimum), list(maximum)], **counts)
assert report['packed_textures'] == len(data['textures'])
assert counts['world_objects'] > 0 and len(data['placements']) == 5815
(out/'blender-validation.json').write_text(json.dumps(report, indent=2)+'\n')
print(json.dumps(report))
