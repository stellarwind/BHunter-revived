"""Audit linked HUD continuation evidence without executing any game code."""
import hashlib
import json
from pathlib import Path
ROOT=Path(__file__).resolve().parents[1];HERE=ROOT/'META/m4/hud-continuation'
def read(p):return json.loads(p.read_text(encoding='utf-8-sig'))
def sha(p):return hashlib.sha256(p.read_bytes()).hexdigest()
def main():
    v=read(HERE/'validation.json');real=read(HERE/'integration-real.json');report=read(HERE/'clean/clean-checkout.json');clean=report['result']
    owners=read(HERE/'owners.json');ledger=read(ROOT/'META/function-status.json')
    assert v['passed'] and real['passed'] and clean['passed'] and real['sources']==clean['sources']
    assert not report['original_assets_present'] and not report['compiler_in_checkout']
    assert len(owners)==v['new_complete_functions']==22
    for name,h in v['sources'].items():assert sha(ROOT/name)==sha(ROOT/v['source_snapshots'][name])==h,name
    for build,prefix in [(real,'real'),(clean,'clean')]:
        assert build['test']==v[prefix+'_total'] and not build['test']['failures'] and not build['original_code_required']
        assert sum(c['label'].startswith('compile-') for c in build['commands'])==v['translation_units']
        controls=[r for k,r in build.items() if k.startswith('missing_')]
        assert len(controls)==v['missing_module_controls'] and all('LNK2001' in r for r in controls)
        assert sum(r['checks'] for r in build['scenarios'] if 'checks' in r and r.get('scenario')!='m2-foundations')==build['test']['checks']
        for scenario,row in v[prefix+'_focused'].items():
            assert next(r for r in build['scenarios'] if r.get('scenario')==scenario)==row
            assert not row['failures'] and (row['checks']>0 or (prefix=='clean' and row.get('archive_skipped')))
    for owner in owners:
        record=next(r for r in ledger['records'] if r['id']==owner['address']+'-functional-c-hud-continuation')
        assert record['source_sha256']==real['sources'][record['source']]
        assert record['implementation']=='func' and record['integration']=='linked_harness' and not record['game_admitted']
        assert not record['validation']['raw_byte_match']
        for name in ['link-map.txt','clean/link-map.txt']:assert ('_'+record['function']).encode() in (HERE/name).read_bytes()
    active={r['original_va'] for r in ledger['records'] if r['implementation'] in ('func','bm') and r['integration']=='linked_harness' and not r.get('historical_variant') and not r.get('validation',{}).get('superseded_by')}
    assert len(active)==v['active_original_functions']==322
    factory=next(r for r in ledger['records'] if r['id']=='00433ea0-construction-registry-adapter')
    assert factory['implementation']=='adapter'
    for entry in read(ROOT/'META/reference-manifest.json')['files']:
        if entry['root']=='repo':assert sha(ROOT/entry['path'])==entry['sha256']
    assert not v['original_code_executed'] and not v['game_launches'] and not v['gameplay_executed']
    print(json.dumps(dict(passed=True,complete_new_owners=22,active_functions=322,real_checks=real['test']['checks'],clean_checks=clean['test']['checks'])))
if __name__=='__main__':main()
