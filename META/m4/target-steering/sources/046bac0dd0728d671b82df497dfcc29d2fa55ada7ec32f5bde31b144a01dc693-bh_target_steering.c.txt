#include "bh_target_steering.h"
#include <stddef.h>
#include <string.h>
#include <float.h>

typedef char steering_payload_offset[offsetof(BHEntity,payload)==0xe4 ? 1:-1];
static float bound(const BHEntity *actor,unsigned long offset) {
    float result;memcpy(&result,actor->payload+offset,4);return result;
}
void __cdecl bh_actor_clamp_point(const BHEntity *actor,BHVec3 *point) {
    float low=bound(actor,0xcc),high;
    if(_isnan((double)point->x) || _isnan((double)low) || point->x<low)point->x=low;
    else {
        high=bound(actor,0xd8);
        if(point->x>high)point->x=high;
    }
    low=bound(actor,0xd0);
    if(low>point->y)point->y=low;
    else {
        high=bound(actor,0xdc);
        if(_isnan((double)high) || _isnan((double)point->y) || high<point->y)point->y=high;
    }
    low=bound(actor,0xd4);
    if(low>point->z)point->z=low;
    else {
        high=bound(actor,0xe0);
        if(_isnan((double)high) || _isnan((double)point->z) || high<point->z)point->z=high;
    }
}
