static unsigned long checks,failures;
#define CHECK(x,s) do {++checks;if(!(x)){++failures;printf("FAIL route-curve-length: %s\n",s);}} while(0)
static unsigned long bits(float f) {unsigned long b;memcpy(&b,&f,4);return b;}
/* Algebraically expanded quartic weights, independent of the real evaluator's
   smoothstep/line-pair implementation. Integer controls below64 make all
   quarter-sample coordinates exact binary32 rationals. */
static double polynomial(double t,double a,double b,double c,double d) {
    double t2=t*t,t3=t2*t,t4=t3*t;
    return a*(1-t-3*t2+5*t3-2*t4)+b*(t-3*t3+2*t4)+c*(3*t2-5*t3+2*t4)+d*(3*t3-2*t4);
}
static float independent_length(const BHVec3 *p) {
    unsigned long j,k;double previous[3],next[3],x,y,z;volatile float total=0;
    for(k=0;k<3;++k)previous[k]=((const float*)p)[k];
    for(j=1;j<4;++j) {
        for(k=0;k<3;++k)next[k]=polynomial((double)j/4,((const float*)p)[k],((const float*)(p+1))[k],
            ((const float*)(p+2))[k],((const float*)(p+3))[k]);
        x=next[0]-previous[0];y=next[1]-previous[1];z=next[2]-previous[2];
        total=(float)(sqrt((x*x+y*y)+z*z)+total);memcpy(previous,next,sizeof(next));
    }
    return total;
}
static void fixtures(void) {
    BHVec3 p[4]={{0,0,0},{12,16,0},{0,0,0},{12,16,0}},before[4];float result;
    unsigned long i,j,k;long value;
    CHECK(bh_route_curve_length(p)==15,"three 3-4-5 chords omit final t=1 endpoint (full line length20)");
    for(i=0;i<4;++i)p[i].x=p[i].y=p[i].z=7;
    CHECK(bits(bh_route_curve_length(p))==0,"constant controls produce positive zero length");
    p[0].x=p[0].y=p[0].z=0;p[1].x=p[1].y=p[1].z=10;
    p[2].x=p[2].y=p[2].z=100;p[3].x=p[3].y=p[3].z=200;
    result=bh_route_curve_length(p);
    CHECK(fabs((double)result-148.828125*sqrt(3))<0.00003,"known quarter samples compose monotonic diagonal distance");
    for(i=0;i<128;++i) {
        for(j=0;j<4;++j)for(k=0;k<3;++k) {
            value=(long)((i*37+j*19+k*11)%127)-63;((float*)(p+j))[k]=(float)value;
        }
        memcpy(before,p,sizeof(p));
        CHECK(bits(bh_route_curve_length(p))==bits(independent_length(p)),
            "independent expanded quartic exact quarter samples and rounded chord sum");
        CHECK(!memcmp(before,p,sizeof(p)),"all input controls remain unchanged");
    }
}
int bh_route_curve_length_checkpoint(unsigned long *count) {
    BHMathEnvironment env;checks=failures=0;
    CHECK(bh_math_environment_enter(&env),"real FP environment");fixtures();bh_math_environment_leave(&env);
    if(count)*count=checks;
    printf("{\"scenario\":\"route-curve-length\",\"checks\":%lu,\"failures\":%lu,\"original_code_required\":false}\n",checks,failures);
    return failures!=0;
}
