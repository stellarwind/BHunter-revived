static unsigned long checks,failures;
#define CHECK(x,s) do {++checks;if(!(x)){++failures;printf("FAIL actor-route: %s\n",s);}} while(0)
static void link(BHEntity *a,unsigned long offset,BHEntity *p){memcpy((unsigned char*)a+offset,&p,4);}
static BHEntity *linked(BHEntity *a,unsigned long offset){BHEntity *p;memcpy(&p,(unsigned char*)a+offset,4);return p;}
static unsigned long next_state(unsigned long state){unsigned long p=state*58199UL;return ((p<<16)|(p>>16))^state;}
static float stored(double x){float f;bh_round_float(&f,x);return f;}
static unsigned long bits(float f){unsigned long b;memcpy(&b,&f,4);return b;}
typedef struct Fixture {
    BHWorldGeometry worlds[3];BHGeometryModel model;
    BHPickupRoom rooms[3];BHActorRouteContext context;BHActorRouteState state;
} Fixture;
static void fixture(Fixture *f,BHEntity *actor,unsigned long flags,unsigned long axis) {
    unsigned long i;BHPortal *p;long kind=2;
    memset(f,0,sizeof(*f));f->model.portal_count=2;
    for(i=0;i<3;++i){f->rooms[i].world=f->worlds+i;f->worlds[i].model=&f->model;f->worlds[i].navigation.count=2;}
    f->worlds[1].query_flags=flags;
    f->worlds[1].minimum.x=f->worlds[1].minimum.y=f->worlds[1].minimum.z=-20;
    f->worlds[1].maximum.x=f->worlds[1].maximum.y=f->worlds[1].maximum.z=20;
    ((float*)&f->worlds[1].minimum)[axis]=0;((float*)&f->worlds[1].maximum)[axis]=100;
    for(i=0;i<2;++i){p=f->worlds[1].navigation.portals+i;p->direction=axis+(i?0:4);p->plane=i?100:0;
        p->scratch[0]=p->scratch[2]=-20;p->scratch[1]=p->scratch[3]=20;
        p->neighbor=&f->worlds[i?2:0].navigation;}
    f->context.rooms=f->rooms;f->context.room_count=3;f->context.actor_limit=16;f->context.tick=0xbeef;
    f->state.current=f->worlds;f->state.next=f->worlds+1;f->state.speed=10;f->state.speed_scale=1;
    f->state.u=f->state.v=.5;f->state.random=0x12345678;f->state.sound=BH_SOUND_INVALID;f->state.phase=7;
    f->rooms[0].actors=actor;link(actor,0xf4,NULL);link(actor,0xf8,NULL);memcpy(actor->payload+24,&kind,4);
    CHECK(bh_actor_route_write(actor,&f->state),"real actor field-view binding");
}
static void promotion(BHEntity *actor,BHEntity *peer) {
    Fixture f;BHActorRouteState s;unsigned long axis,straight,i;BHEntity before;BHWorldGeometry *owner;
    for(axis=0;axis<3;++axis)for(straight=0;straight<2;++straight) {
        fixture(&f,actor,straight?8:0,axis);f.rooms[1].actors=peer;link(peer,0xf4,NULL);link(peer,0xf8,NULL);
        CHECK(bh_actor_route_promote(&f.context,actor,0,1),"complete route promotion");bh_actor_route_read(actor,&s);
        CHECK(s.current==f.worlds+1 && s.next==f.worlds+2,"world owner promotion and actual portal next world");
        CHECK(f.rooms[0].actors==NULL && f.rooms[1].actors==actor && linked(actor,0xf4)==peer &&
            linked(actor,0xf8)==NULL && linked(peer,0xf8)==actor,"actual old/new room head and reciprocal f4/f8 ownership");
        memcpy(&owner,actor->opaque2b,4);CHECK(owner==f.worlds+1,"actual actor+c4 owner pointer updated");
        CHECK(s.controls[0].x==0 && s.controls[0].y==0 && s.controls[0].z==0 &&
            ((float*)&s.controls[3])[axis]==100 && ((float*)&s.portal_center)[axis]==100,"axis-specific endpoint and center mapping");
        CHECK(s.phase==0 && s.duration==(straight?20:10) && s.duration_ticks==(straight?600:300) && s.tick==0xbeef,
            "exact straight/curved duration,30Hz frame count and explicit clock binding");
        CHECK(((float*)&s.controls[1])[axis]==(straight?100:65) && ((float*)&s.controls[2])[axis]==(straight?0:35),
            "real dominant-axis tangent owner and original negative1.3 coefficient");
        for(i=0;i<3;++i)if(i!=axis)CHECK(((float*)&s.controls[1])[i]==0 && ((float*)&s.controls[2])[i]==0,"other tangent axes preserved");
    }
    fixture(&f,actor,0,0);
    CHECK(bh_actor_route_promote(&f.context,actor,0,0),"same portal detour route");bh_actor_route_read(actor,&s);
    CHECK(s.next==f.worlds && s.controls[1].x==65 && s.controls[1].y==-2 && s.controls[1].z==50 &&
        s.controls[2].x==65 && s.controls[2].y==2 && s.controls[2].z==-50,"same portal detour uses snapped delta and asymmetric vertical offsets");
    fixture(&f,actor,0,0);
    for(i=0;i<2;++i){BHPortal *p=f.worlds[1].navigation.portals+i;p->scratch[0]=p->scratch[2]=-10;p->scratch[1]=p->scratch[3]=10;}
    CHECK(bh_actor_route_promote(&f.context,actor,0,1),"narrow aperture duration branch");bh_actor_route_read(actor,&s);
    CHECK(bits(s.duration)==bits(stored(10.0/.75)) && s.duration_ticks==400,"narrow duration uses stored base but retained quotient for30Hz conversion");
    fixture(&f,actor,0,0);before=*actor;f.context.actor_limit=0;
    CHECK(!bh_actor_route_promote(&f.context,actor,0,1) && !memcmp(actor,&before,sizeof(before)),"missing bound rejects without actor mutation");
    f.context.actor_limit=16;
    CHECK(!bh_actor_route_promote(&f.context,actor,-1,1) && !memcmp(actor,&before,sizeof(before)),"invalid incoming rejects before mutation");
    CHECK(!bh_actor_route_promote(&f.context,actor,0,2) && !memcmp(actor,&before,sizeof(before)),"invalid outgoing rejects before mutation");
    f.worlds[1].navigation.portals[1].neighbor=NULL;
    CHECK(!bh_actor_route_promote(&f.context,actor,0,1) && !memcmp(actor,&before,sizeof(before)),"missing real next neighbor rejected");
    fixture(&f,actor,0,0);f.state.speed=0;bh_actor_route_write(actor,&f.state);before=*actor;
    CHECK(!bh_actor_route_promote(&f.context,actor,0,1) && !memcmp(actor,&before,sizeof(before)),"invalid positive speed contract rejects before publication");
    CHECK(!bh_actor_route_read(NULL,&s) && !bh_actor_route_write(NULL,&s),"field view null guards");
}
static void list_positions(BHEntity **nodes) {
    Fixture f;unsigned long position,i;BHEntity *actor,*expected[2];
    for(position=0;position<3;++position) {
        actor=nodes[position];fixture(&f,actor,0,0);f.rooms[0].actors=nodes[0];
        for(i=0;i<3;++i){link(nodes[i],0xf8,i?nodes[i-1]:NULL);link(nodes[i],0xf4,i<2?nodes[i+1]:NULL);}
        CHECK(bh_actor_route_promote(&f.context,actor,0,1),"promote from each actual room-list position");
        i=0;while(i<3 && nodes[i]==actor)++i;expected[0]=nodes[i++];while(i<3 && nodes[i]==actor)++i;expected[1]=nodes[i];
        CHECK(f.rooms[0].actors==expected[0] && linked(expected[0],0xf8)==NULL && linked(expected[0],0xf4)==expected[1] &&
            linked(expected[1],0xf8)==expected[0] && linked(expected[1],0xf4)==NULL,"head/middle/tail unlink retains both neighbor links");
    }
}
static void advance(BHEntity *actor) {
    Fixture f;BHActorRouteState s;unsigned long guided,expected,first,second;BHSoundQueue sound;
    BHSoundResource *resource;BHSoundVoice *voice;unsigned long handle;unsigned short generation;
    for(guided=0;guided<2;++guided) {
        fixture(&f,actor,guided?8:0,0);expected=f.state.random;
        if(!guided)expected=next_state(expected);first=next_state(expected);second=next_state(first);
        f.state.jitter=.5f;bh_actor_route_write(actor,&f.state);
        CHECK(bh_actor_route_advance(&f.context,actor),"full random/guided caller advances real room route");bh_actor_route_read(actor,&s);
        CHECK(s.random==second && s.current==f.worlds+1 && s.next==f.worlds+2,"guided consumes2 and random consumes3 exact RNG transitions");
        CHECK(bits(s.u)==bits(stored(.25+(double)(first&32767)/65536)) &&
            bits(s.v)==bits(stored(.25+(double)(second&32767)/65536)),"aperture bias smoothing uses real signed coefficient and exact random draws");
    }
    CHECK(bh_sound_queue_init(&sound,NULL,NULL,NULL),"real source sound queue owner");
    resource=bh_sound_resource_allocate(&sound);CHECK(resource!=NULL,"real source streaming sound resource");
    if(resource) {
        resource->streaming=1;resource->base_frequency=44100;
        handle=bh_sound_request(&sound,resource->index,1,1,NULL,NULL);voice=bh_sound_voice(&sound,handle);
        CHECK(handle!=BH_SOUND_INVALID && voice!=NULL,"real source voice request without playback");
        if(voice) {
            generation=voice->generation;fixture(&f,actor,0,0);f.context.sound=&sound;f.state.sound=handle;f.state.sound_frames=1;
            bh_actor_route_write(actor,&f.state);
            CHECK(bh_actor_route_advance(&f.context,actor),"route caller executes real sound expiry owner then route update");bh_actor_route_read(actor,&s);
            CHECK(s.sound==BH_SOUND_INVALID && s.sound_frames==0 && resource->stream_head==NULL &&
                voice->generation==(unsigned short)(generation+1) && bh_sound_voice(&sound,handle)==NULL,
                "actual voice lifetime invalidated and streaming list unlinked, no fake stop callback");
        }
    }
    CHECK(bh_sound_queue_dispose(&sound),"real sound queue disposal");
    fixture(&f,actor,0,0);f.state.sound_frames=0xffffffffUL;bh_actor_route_write(actor,&f.state);
    CHECK(bh_actor_route_advance(&f.context,actor),"negative original countdown word preserves wrapping decrement");bh_actor_route_read(actor,&s);
    CHECK(s.sound_frames==0xfffffffeUL,"unsigned storage preserves original DEC bits");
    fixture(&f,actor,0,0);f.state.sound_frames=1;f.state.sound=123;bh_actor_route_write(actor,&f.state);
    CHECK(!bh_actor_route_advance(&f.context,actor) && f.context.error==BH_ACTOR_ROUTE_ARGUMENT,"missing actual sound owner rejects expiry branch");
}
int bh_actor_route_checkpoint(unsigned long *count) {
    BHEntityPool pool;BHEntity *nodes[4];unsigned long i;BHMathEnvironment env;
    checks=failures=0;CHECK(bh_math_environment_enter(&env),"real numerical environment");
    CHECK(bh_entities_init(&pool,4,NULL),"real source actor allocation owner");
    for(i=0;i<4;++i)nodes[i]=bh_entity_acquire(&pool);
    CHECK(nodes[0] && nodes[1] && nodes[2] && nodes[3],"four actual acquired actor slots");
    if(nodes[0] && nodes[1] && nodes[2] && nodes[3]){promotion(nodes[0],nodes[3]);list_positions(nodes);advance(nodes[0]);}
    CHECK(bh_entities_valid(&pool),"pool-owned links remain valid independently from actual game f4/f8 lists");
    CHECK(bh_entities_dispose(&pool),"real actor pool disposal");bh_math_environment_leave(&env);
    if(count)*count=checks;
    printf("{\"scenario\":\"actor-route\",\"checks\":%lu,\"failures\":%lu,\"original_code_required\":false}\n",checks,failures);
    return failures!=0;
}
