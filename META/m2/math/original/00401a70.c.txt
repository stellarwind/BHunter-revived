
/* Recovered layout and behavior tested against original isolated machine code. See src/bh_math.c
   and analysis/math/RECOVERY.md. Not byte-matched; aliasing and floating-point limits documented
   there. */

void __cdecl bh_mat34_inverse_rigid(BHMat34 *m,BHMat34 *out)

{
  float fVar1;
  float fVar2;
  float fVar3;
  float fVar4;
  
  out->m[0][0] = m->m[0][0];
  out->m[0][1] = m->m[1][0];
  out->m[0][2] = m->m[2][0];
  out->m[1][0] = m->m[0][1];
  out->m[1][1] = m->m[1][1];
  out->m[1][2] = m->m[2][1];
  out->m[2][0] = m->m[0][2];
  out->m[2][1] = m->m[1][2];
  fVar1 = out->m[0][1];
  out->m[2][2] = m->m[2][2];
  fVar2 = out->m[1][0];
  fVar3 = out->m[1][1];
  out->m[0][3] = (-(m->m[0][3] * out->m[0][0]) - fVar1 * m->m[1][3]) - m->m[2][3] * out->m[0][2];
  fVar1 = out->m[2][0];
  fVar4 = out->m[2][1];
  out->m[1][3] = (-(fVar2 * m->m[0][3]) - fVar3 * m->m[1][3]) - m->m[2][3] * out->m[1][2];
  out->m[2][3] = (-(m->m[0][3] * fVar1) - m->m[1][3] * fVar4) - m->m[2][3] * out->m[2][2];
  return;
}
