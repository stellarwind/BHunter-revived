
/* Recovered layout and behavior tested against original isolated machine code. See src/bh_math.c
   and analysis/math/RECOVERY.md. Not byte-matched; aliasing and floating-point limits documented
   there. */

void __cdecl bh_mat34_multiply(BHMat34 *a,BHMat34 *b,BHMat34 *out)

{
  out->m[0][0] = b->m[0][0] * a->m[0][0] + a->m[0][1] * b->m[1][0] + a->m[0][2] * b->m[2][0];
  out->m[0][1] = a->m[0][1] * b->m[1][1] + b->m[2][1] * a->m[0][2] + b->m[0][1] * a->m[0][0];
  out->m[0][2] = b->m[2][2] * a->m[0][2] + a->m[0][1] * b->m[1][2] + b->m[0][2] * a->m[0][0];
  out->m[0][3] = b->m[2][3] * a->m[0][2] + a->m[0][1] * b->m[1][3] + b->m[0][3] * a->m[0][0] +
                 a->m[0][3];
  out->m[1][0] = b->m[1][0] * a->m[1][1] + b->m[2][0] * a->m[1][2] + a->m[1][0] * b->m[0][0];
  out->m[1][1] = a->m[1][1] * b->m[1][1] + b->m[2][1] * a->m[1][2] + b->m[0][1] * a->m[1][0];
  out->m[1][2] = a->m[1][0] * b->m[0][2] + b->m[1][2] * a->m[1][1] + b->m[2][2] * a->m[1][2];
  out->m[1][3] = b->m[1][3] * a->m[1][1] + a->m[1][2] * b->m[2][3] + b->m[0][3] * a->m[1][0] +
                 a->m[1][3];
  out->m[2][0] = a->m[2][0] * b->m[0][0] + a->m[2][2] * b->m[2][0] + b->m[1][0] * a->m[2][1];
  out->m[2][1] = a->m[2][2] * b->m[2][1] + b->m[1][1] * a->m[2][1] + b->m[0][1] * a->m[2][0];
  out->m[2][2] = a->m[2][2] * b->m[2][2] + b->m[0][2] * a->m[2][0] + b->m[1][2] * a->m[2][1];
  out->m[2][3] = a->m[2][1] * b->m[1][3] + b->m[2][3] * a->m[2][2] + b->m[0][3] * a->m[2][0] +
                 a->m[2][3];
  return;
}
