
/* Recovered layout and behavior tested against original isolated machine code. See src/bh_math.c
   and analysis/math/RECOVERY.md. Not byte-matched; aliasing and floating-point limits documented
   there. */

void __cdecl bh_mat34_inverse_transform_point(BHMat34 *m,BHVec3 *v,BHVec3 *out)

{
  float fVar1;
  float fVar2;
  float fVar3;
  
  fVar2 = v->x - m->m[0][3];
  fVar1 = v->y - m->m[1][3];
  fVar3 = v->z - m->m[2][3];
  out->x = fVar2 * m->m[0][0] + m->m[2][0] * fVar3 + m->m[1][0] * fVar1;
  out->y = m->m[2][1] * fVar3 + m->m[1][1] * fVar1 + m->m[0][1] * fVar2;
  out->z = m->m[2][2] * fVar3 + m->m[1][2] * fVar1 + m->m[0][2] * fVar2;
  return;
}
