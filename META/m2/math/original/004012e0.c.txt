
/* Recovered layout and behavior tested against original isolated machine code. See src/bh_math.c
   and analysis/math/RECOVERY.md. Not byte-matched; aliasing and floating-point limits documented
   there. */

void __cdecl bh_mat34_transform_point(BHMat34 *m,BHVec3 *v,BHVec3 *out)

{
  out->x = v->x * m->m[0][0] + v->z * m->m[0][2] + v->y * m->m[0][1] + m->m[0][3];
  out->y = m->m[1][0] * v->x + m->m[1][2] * v->z + v->y * m->m[1][1] + m->m[1][3];
  out->z = m->m[2][0] * v->x + m->m[2][2] * v->z + v->y * m->m[2][1] + m->m[2][3];
  return;
}
