/* Shared deterministic inputs, ABI dispatch and representation-aware checks.
   No oracle code or original image mapping is present in this translation unit. */
#include "math_cases.h"
#include <string.h>
#include <float.h>

void bh_math_source_ops(BHMathOps *ops) {
    ops->multiply=bh_mat34_multiply;ops->cross=bh_vec3_cross;
    ops->normalize=bh_vec3_normalize;ops->normalize_to=bh_vec3_normalize_to;
    ops->point=bh_mat34_transform_point;ops->direction=bh_mat34_transform_direction;
    ops->inverse_point=bh_mat34_inverse_transform_point;
    ops->world=bh_mat34_translate_world;ops->local=bh_mat34_translate_local;
    ops->inverse=bh_mat34_inverse_rigid;
}

static unsigned long random_word(unsigned long *seed) {
    *seed=*seed*1664525UL+1013904223UL;return *seed;
}

void bh_math_case(unsigned long id,BHMathCase *input) {
    static const unsigned long edge[]={0,0x80000000UL,1,0x80000001UL,0x007fffffUL,0x00800000UL,
        0x3f000000UL,0x3f800000UL,0xbf800000UL,0x3f800001UL,0x3f7fffffUL,
        0x7f7fffffUL,0xff7fffffUL,0x00800001UL,0x7f800000UL,0xff800000UL,
        0x7fc12345UL,0xffc54321UL,0x7f812345UL};
    unsigned long seed=id^0x19990911UL,word;
    float values[30];
    int i;
    for (i=0;i<30;i++) {
        word=random_word(&seed);
        if (id<2048) values[i]=(float)((long)((word>>8)&65535)-32768)/1024.0f;
        else if (id<10240) {
            word=(word&0x807fffffUL) | (((random_word(&seed)>>16)%255)<<23);
            memcpy(&values[i],&word,4);
        } else {
            word=edge[((id-10240)*7+i*3)%(sizeof(edge)/sizeof(edge[0]))];
            memcpy(&values[i],&word,4);
        }
    }
    memcpy(&input->a,values,48);memcpy(&input->b,values+12,48);
    memcpy(&input->v,values+24,12);memcpy(&input->w,values+27,12);
    /* Deterministic cancellation, zero and extreme normalization cases. */
    if (id<32) {
        float magnitude;
        word=(unsigned long)(id%2 ? 250 : 4)<<23;memcpy(&magnitude,&word,4);
        input->v.x=magnitude;input->v.y=-magnitude;input->v.z=1.0f;
        input->a.m[0][0]=1.0f;input->a.m[0][1]=1.0f;input->a.m[0][2]=1.0f;
        if (id==0) input->v.x=input->v.y=input->v.z=0;
    }
}

void bh_math_eval(const BHMathOps *ops,int function,int alias,const BHMathCase *input,BHMathResult *result) {
    struct {unsigned long before;BHMat34 value;unsigned long after;} matrix;
    struct {unsigned long before;BHVec3 value;unsigned long after;} vector;
    BHMat34 a=input->a,b=input->b;
    BHVec3 v=input->v,w=input->w;
    BHVec3 *destination=alias==1 ? &v : alias==2 ? &w : &vector.value;
    const BHVec3 *second=alias==3 ? &v : &w;
    unsigned int control=_control87(0,0);
    int is_matrix=function==0 || function>=7;
    memset(&matrix.value,0x3f,sizeof(matrix.value));
    vector.value.x=vector.value.y=vector.value.z=123.0f;
    matrix.before=matrix.after=vector.before=vector.after=0x1badb002UL;
    result->status=0;
    _clearfp();
    switch (function) {
    case 0:ops->multiply(&a,&b,&matrix.value);break;
    case 1:ops->cross(&v,second,destination);break;
    case 2:result->status=ops->normalize(&v);destination=&v;break;
    case 3:result->status=ops->normalize_to(&v,destination);break;
    case 4:ops->point(&a,&v,destination);break;
    case 5:ops->direction(&a,&v,destination);break;
    case 6:ops->inverse_point(&a,&v,destination);break;
    case 7:matrix.value=a;ops->world(v.x,v.y,v.z,&matrix.value);break;
    case 8:matrix.value=a;ops->local(v.x,v.y,v.z,&matrix.value);break;
    case 9:ops->inverse(&a,&matrix.value);break;
    }
    result->fp_status=_status87();
    result->control_ok=(_control87(0,0)&(_MCW_PC|_MCW_RC|_MCW_EM))==(control&(_MCW_PC|_MCW_RC|_MCW_EM));
    result->guards_ok=matrix.before==0x1badb002UL && matrix.after==0x1badb002UL &&
                      vector.before==0x1badb002UL && vector.after==0x1badb002UL;
    result->count=is_matrix ? 12 : 3;
    memset(result->words,0,sizeof(result->words));
    memcpy(result->words,is_matrix ? (void*)&matrix.value : (void*)destination,result->count*4);
}

/* 0 finite nonzero, 1 signed zero, 2 infinity, 3 NaN. */
int bh_math_word_class(unsigned long word) {
    unsigned long magnitude=word&0x7fffffffUL;
    if (magnitude>0x7f800000UL) return 3;
    if (magnitude==0x7f800000UL) return 2;
    return magnitude==0 ? 1 : 0;
}

unsigned long bh_math_ulp(unsigned long a,unsigned long b) {
    unsigned long x=(a&0x80000000UL) ? ~a : a^0x80000000UL;
    unsigned long y=(b&0x80000000UL) ? ~b : b^0x80000000UL;
    return x>y ? x-y : y-x;
}
